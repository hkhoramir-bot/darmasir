    37→async function api(path, opts = {}) {
    38→  const headers = { "Content-Type": "application/json", ...(opts.headers || {}) };
    39→  const token = localStorage.getItem(TOKEN_KEY);
    40→  if (token) headers.Authorization = `Bearer ${token}`;
    41→  const res = await fetch(apiPath(path), {
    42→    method: opts.method || "GET",
    43→    headers,
    44→    body: opts.body ? JSON.stringify(opts.body) : undefined,
    45→  });
    46→  const json = await res.json().catch(() => ({ ok: false, error: { code: "PARSE", message: "پاسخ نامعتبر سرور" } }));
    47→  if (!json.ok) {
    48→    if (res.status === 401 && !path.startsWith("auth/")) { logoutLocal(); throw new Error(json.error?.message || "نشست منقضی شده"); }
    49→    throw new Error(json.error?.message || "خطای ناشناخته");
    50→  }
    51→  return json.data;
    52→}

    96→function csvExport(filename, headers, rows) {
    97→  const line = (arr) => arr.map((v) => `"${String(v ?? "").replace(/"/g, '""')}"`).join(",");
    98→  const csv = "\uFEFF" + line(headers) + "\r\n" + rows.map(line).join("\r\n");
    99→  const blob = new Blob([csv], { type: "text/csv; charset=utf-8" });
   100→  const a = document.createElement("a");
   101→  a.href = URL.createObjectURL(blob);
   102→  a.download = filename;
   103→  a.click();
   104→  URL.revokeObjectURL(a.href);
   105→  toast("فایل CSV دانلود شد", "ok2");
   106→}
