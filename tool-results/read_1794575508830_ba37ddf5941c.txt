    50→function computeVisibleStats(hidden: Set<string>) {
    51→  const lines = MAP_LINES.filter((ml) => !hidden.has(ml.lineId));
    52→  const stationIds = new Set<string>();
    53→  const count = new Map<string, number>();
    54→  const color = new Map<string, string>();
    55→  for (const ml of lines) {
    56→    for (const id of ml.stations) {
    57→      stationIds.add(id);
    58→      count.set(id, (count.get(id) ?? 0) + 1);
    59→      if (!color.has(id)) color.set(id, ml.color);
    60→    }
    61→  }
    62→  const inter = new Set([...count.entries()].filter(([, n]) => n > 1).map(([id]) => id));
    63→  return { lines, stationIds, inter, color };
    64→}

    66→/* ---------- شبکه اشغال «بدنه خطوط» برای جلوگیری از افتادن متن روی خط ----------
    67→   سلول‌هایی که مرکزشان کمتر از GRID_PAD از هر قطعه خط فاصله دارد علامت
    68→   می‌خورند؛ برچسب‌ها پیش از جای‌گذاری با این شبکه چک می‌شوند. */
    69→const GRID_CELL = 12;
    70→const GRID_PAD = 8;
    71→
    72→function buildStrokeGrid(lines: { stations: string[] }[]) {
    73→  const gw = Math.ceil(MAP_CANVAS.w / GRID_CELL);
    74→  const gh = Math.ceil(MAP_CANVAS.h / GRID_CELL);
    75→  const grid = new Uint8Array(gw * gh);
    76→  const ptSegDist = (
    77→    px: number, py: number,
    78→    ax: number, ay: number, bx: number, by: number
    79→  ): number => {
    80→    const dx = bx - ax, dy = by - ay;
    81→    const len2 = dx * dx + dy * dy || 1;
    82→    const t = Math.max(0, Math.min(1, ((px - ax) * dx + (py - ay) * dy) / len2));
    83→    return Math.hypot(px - (ax + dx * t), py - (ay + dy * t));
    84→  };
    85→  for (const ml of lines) {
    86→    const pts: { x: number; y: number }[] = [];
    87→    for (const id of ml.stations) {
    88→      const s = MAP_STATION_BY_ID.get(id);
    89→      if (s) pts.push(s);
    90→    }
    91→    for (let i = 0; i < pts.length - 1; i++) {
    92→      const a = pts[i];
    93→      const b = pts[i + 1];
    94→      const c0x = Math.max(0, Math.floor((Math.min(a.x, b.x) - GRID_PAD) / GRID_CELL));
    95→      const c1x = Math.min(gw - 1, Math.floor((Math.max(a.x, b.x) + GRID_PAD) / GRID_CELL));
    96→      const c0y = Math.max(0, Math.floor((Math.min(a.y, b.y) - GRID_PAD) / GRID_CELL));
    97→      const c1y = Math.min(gh - 1, Math.floor((Math.max(a.y, b.y) + GRID_PAD) / GRID_CELL));
    98→      for (let cy = c0y; cy <= c1y; cy++) {
    99→        for (let cx = c0x; cx <= c1x; cx++) {
   100→          const px = cx * GRID_CELL + GRID_CELL / 2;
   101→          const py = cy * GRID_CELL + GRID_CELL / 2;
   102→          if (ptSegDist(px, py, a.x, a.y, b.x, b.y) <= GRID_PAD) grid[cy * gw + cx] = 1;
   103→        }
   104→      }
   105→    }
   106→  }
   107→  return { grid, gw, gh };
   108→}

   121→/* ---------------- اندازه‌گیری واقعی عرض متن (برای جلوگیری از همپوشانی) ---------------- */
   122→let measureCtx: CanvasRenderingContext2D | null = null;
   123→const widthCache = new Map<string, number>();
   124→
   125→function textWidthPx(text: string, weight: number, px: number, epoch: number): number {
   126→  const key = `${epoch}|${weight}|${px}|${text}`;
   127→  const hit = widthCache.get(key);
   128→  if (hit !== undefined) return hit;
   129→  let w: number;
   130→  try {
   131→    if (!measureCtx) {
   132→      measureCtx = document.createElement("canvas").getContext("2d");
   133→    }
   134→    if (measureCtx) {
   135→      measureCtx.font = `${weight} ${px}px Vazirmatn, sans-serif`;
   136→      w = measureCtx.measureText(text).width;
   137→    } else {
   138→      w = text.length * px * 0.6;
   139→    }
   140→  } catch {
   141→    w = text.length * px * 0.6;
   142→  }
   143→  widthCache.set(key, w);
   144→  return w;
   145→}

   185→export default function MetroMap({
   186→  onStationSelect,
   187→  highlightStations,
   188→  hiddenLineIds,
   189→  onToggleLine,
   190→  className = "",
   191→}: MetroMapProps) {
   192→  const [t, setT] = useState<Transform>({ scale: 1, tx: 0, ty: 0 });
   193→  const [legendOpen, setLegendOpen] = useState(false);
   194→  const [locate, setLocate] = useState<LocateState>({ kind: "idle" });
   195→  const [box, setBox] = useState({ w: 0, h: 0 });
   196→  const [fontsTick, setFontsTick] = useState(0);
   197→
   198→  /* ---------- خطوط نمایان (کلید چشم) + شبکه اشغال خطوط ---------- */
   199→  const hiddenSet = useMemo(() => new Set(hiddenLineIds ?? []), [hiddenLineIds]);
   200→  const visible = useMemo(() => computeVisibleStats(hiddenSet), [hiddenSet]);
   201→  const strokeGrid = useMemo(() => buildStrokeGrid(visible.lines), [visible.lines]);
   202→
   203→  /* ---------- آینه همزمان ترنسفورم — جلوگیری از رقابت با React ---------- */
   204→  const tRef = useRef<Transform>({ scale: 1, tx: 0, ty: 0 });
   205→  const svgRef = useRef<SVGSVGElement>(null);
   206→
   207→  /* اشاره‌گرهای فعال — پایه ژست چندلمسی (t = آخرین زمان دیدن اشاره‌گر) */
   208→  const pointersRef = useRef<Map<number, { x: number; y: number; t: number }>>(new Map());
   209→  const panRef = useRef<{ startX: number; startY: number; tx: number; ty: number } | null>(null);
   210→  const pinchRef = useRef<{
   211→    dist0: number;
   212→    vx0: number;
   213→    vy0: number;
   214→    s0: number;
   215→    tx0: number;
   216→    ty0: number;
   217→  } | null>(null);
   218→  const movedRef = useRef(false); // بعد از جابجایی، تپ ایستگاه نادیده گرفته می‌شود
   219→
   220→  /* ---------- اندازه واقعی بوم برای مقیاس ثابت متن ---------- */
   221→  useEffect(() => {
   222→    const el = svgRef.current;
   223→    if (!el) return;
   224→    const update = () => {
   225→      const r = el.getBoundingClientRect();
   226→      if (r.width < 2 || r.height < 2) return;
   227→      setBox((b) =>
   228→        Math.abs(b.w - r.width) > 0.5 || Math.abs(b.h - r.height) > 0.5
   229→          ? { w: r.width, h: r.height }
   230→          : b
   231→      );
   232→    };
   233→    update();
   234→    const ro = new ResizeObserver(update);
   235→    ro.observe(el);
   236→    return () => ro.disconnect();
   237→  }, []);
   238→
   239→  /* ---------- بعد از لود کامل فونت، عرض متن‌ها دوباره اندازه‌گیری می‌شود ---------- */
   240→  useEffect(() => {
   241→    let alive = true;
   242→    try {
   243→      document.fonts?.ready
   244→        .then(() => {
   245→          if (!alive) return;
   246→          widthCache.clear();
   247→          setFontsTick((v) => v + 1);
   248→        })
   249→        .catch(() => {});
   250→    } catch {
   251→      /* مرورگرهای قدیمی */
   252→    }
   253→    return () => {
   254→      alive = false;
   255→    };
   256→  }, []);
   257→
   258→  const clamp = (v: number, min: number, max: number) => Math.min(max, Math.max(min, v));
   259→
   260→  /** ثبت ترنسفورم جدید — همیشه از طریق این تابع */
   261→  const commit = useCallback((next: Transform) => {
   262→    const scale = clamp(next.scale, MIN_SCALE, MAX_SCALE);
   263→    /* پنجره دید واقعی بر حسب واحدهای viewBox — با احتساب letterbox حالت meet.
   264→       clamp قبلی فقط مرکز بوم را مبنا می‌گرفت و در بزرگ‌نمایی اجازه می‌داد
   265→       کل نقشه از پنجره دید خارج شود (کاربر حس می‌کرد pan خراب است). */
   266→    let wx0 = 0, wy0 = 0, winW = MAP_CANVAS.w, winH = MAP_CANVAS.h;
   267→    const svg = svgRef.current;
   268→    if (svg) {
   269→      const rect = svg.getBoundingClientRect();
   270→      if (rect.width > 2 && rect.height > 2) {
   271→        const f = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   272→        wx0 = -((rect.width - MAP_CANVAS.w * f) / 2) / f;
   273→        wy0 = -((rect.height - MAP_CANVAS.h * f) / 2) / f;
   274→        winW = rect.width / f;
   275→        winH = rect.height / f;
   276→      }
   277→    }
   278→    const OVER = 60; // اجازه سرریز اندک در لبه‌ها
   279→    // لبه محتوا نباید از پنجره دید بیرون بزند — در هر بزرگ‌نمایی کل نقشه قابل رسیدن است
   280→    const loX = wx0 + winW - scale * MAP_CANVAS.w - OVER;
   281→    const hiX = wx0 + OVER;
   282→    const loY = wy0 + winH - scale * MAP_CANVAS.h - OVER;
   283→    const hiY = wy0 + OVER;
   284→    const tx = loX <= hiX ? clamp(next.tx, loX, hiX) : (loX + hiX) / 2;
   285→    const ty = loY <= hiY ? clamp(next.ty, loY, hiY) : (loY + hiY) / 2;
   286→    const fixed = { scale, tx, ty };
   287→    tRef.current = fixed;
   288→    setT(fixed);
   289→  }, []);
   290→
   291→  /** تبدیل مختصات صفحه به مختصات viewBox (با احتساب letterbox وضعیت meet) */
   292→  const viewPoint = useCallback((clientX: number, clientY: number) => {
   293→    const svg = svgRef.current;
   294→    if (!svg) return { vx: 0, vy: 0 };
   295→    const rect = svg.getBoundingClientRect();
   296→    const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   297→    const ox = (rect.width - MAP_CANVAS.w * fit) / 2;
   298→    const oy = (rect.height - MAP_CANVAS.h * fit) / 2;
   299→    return {
   300→      vx: (clientX - rect.left - ox) / (fit * tRef.current.scale) - tRef.current.tx / tRef.current.scale,
   301→      vy: (clientY - rect.top - oy) / (fit * tRef.current.scale) - tRef.current.ty / tRef.current.scale,
   302→    };
   303→  }, []);
   304→
   305→  /** بزرگ‌نمایی حول یک نقطه صفحه (پیش‌فرض: مرکز) — لنگر ثابت می‌ماند */
   306→  const zoomAt = useCallback(
   307→    (factor: number, clientX?: number, clientY?: number) => {
   308→      const t0 = tRef.current;
   309→      const svg = svgRef.current;
   310→      if (!svg) {
   311→        commit({ ...t0, scale: t0.scale * factor });
   312→        return;
   313→      }
   314→      const rect = svg.getBoundingClientRect();
   315→      const cx = clientX ?? rect.left + rect.width / 2;
   316→      const cy = clientY ?? rect.top + rect.height / 2;
   317→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   318→      const ox = (rect.width - MAP_CANVAS.w * fit) / 2;
   319→      const oy = (rect.height - MAP_CANVAS.h * fit) / 2;
   320→      // نقطه لنگر در مختصات viewBox فعلی
   321→      const ax = (cx - rect.left - ox) / fit;
   322→      const ay = (cy - rect.top - oy) / fit;
   323→      const scale = clamp(t0.scale * factor, MIN_SCALE, MAX_SCALE);
   324→      commit({
   325→        scale,
   326→        tx: ax - ((ax - t0.tx) * scale) / t0.scale,
   327→        ty: ay - ((ay - t0.ty) * scale) / t0.scale,
   328→      });
   329→    },
   330→    [commit]
   331→  );
   332→
   333→  /* ---------- چرخ ماوس (غیر-Passive تا preventDefault کار کند) ---------- */
   334→  useEffect(() => {
   335→    const el = svgRef.current;
   336→    if (!el) return;
   337→    const onWheel = (e: WheelEvent) => {
   338→      e.preventDefault();
   339→      zoomAt(e.deltaY < 0 ? 1.15 : 0.87, e.clientX, e.clientY);
   340→    };
   341→    el.addEventListener("wheel", onWheel, { passive: false });
   342→    return () => el.removeEventListener("wheel", onWheel);
   343→  }, [zoomAt]);
   344→
   345→  /* ---------- تور ایمنی سراسری برای اشاره‌گرهای گم‌شده ----------
   346→     اگر pointerup/pointercancel انگشتی به SVG نرسد (مثلاً انگشت روی کارت
   347→     راهنما، دکمه‌های شناور یا بیرون نقشه بالا آمده باشد، یا برچسبِ زیر
   348→     انگشت در میانه ژست از DOM حذف شود)، اشاره‌گر «مرده» در حافظه می‌ماند
   349→     و ژستِ بعدی به‌جای جابجایی، پینچِ خراب با لنگر مرده اجرا می‌شد —
   350→     همین باعث «ناتوانی در pan بعد از zoom» بود. این شنوندهٔ سراسری
   351→     (فاز capture) هر اشاره‌گر گم‌شده را بلافاصله پاک می‌کند. */
   352→  useEffect(() => {
   353→    const release = (e: PointerEvent) => {
   354→      if (!pointersRef.current.has(e.pointerId)) return;
   355→      pointersRef.current.delete(e.pointerId);
   356→      if (pointersRef.current.size === 1) {
   357→        // یک انگشت باقی مانده — ادامه با جابجایی از موقعیت فعلی او
   358→        pinchRef.current = null;
   359→        const [p] = [...pointersRef.current.values()];
   360→        panRef.current = { startX: p.x, startY: p.y, tx: tRef.current.tx, ty: tRef.current.ty };
   361→      } else if (pointersRef.current.size === 0) {
   362→        panRef.current = null;
   363→        pinchRef.current = null;
   364→      }
   365→    };
   366→    const resetAll = () => {
   367→      pointersRef.current.clear();
   368→      panRef.current = null;
   369→      pinchRef.current = null;
   370→    };
   371→    const onVisibility = () => {
   372→      if (document.hidden) resetAll();
   373→    };
   374→    window.addEventListener("pointerup", release, true);
   375→    window.addEventListener("pointercancel", release, true);
   376→    window.addEventListener("blur", resetAll);
   377→    document.addEventListener("visibilitychange", onVisibility);
   378→    return () => {
   379→      window.removeEventListener("pointerup", release, true);
   380→      window.removeEventListener("pointercancel", release, true);
   381→      window.removeEventListener("blur", resetAll);
   382→      document.removeEventListener("visibilitychange", onVisibility);
   383→    };
   384→  }, []);
   385→
   386→  /* ---------- اشاره‌گرها ---------- */
   387→  const onPointerDown = (e: React.PointerEvent) => {
   388→    /* پاک‌سازی اشاره‌گرهای مرده — pointerup آن‌ها هرگز به نقشه نرسیده
   389→       (انگشت روی کارت HTML یا بیرون نقشه بالا آمده بود). همین اشاره‌گرهای
   390→       مرده ژستِ بعدی را به پینچ خراب تبدیل می‌کردند. */
   391→    const now = performance.now();
   392→    for (const [id, p] of pointersRef.current) {
   393→      if (now - p.t > STALE_POINTER_MS) pointersRef.current.delete(id);
   394→    }
   395→    pointersRef.current.set(e.pointerId, { x: e.clientX, y: e.clientY, t: now });
   396→
   397→    if (pointersRef.current.size === 1) {
   398→      // شروع جابجایی تک‌انگشتی — کپچر عمداً فعال نمی‌شود تا تپ روی
   399→      // برچسب/ایستگاه با click واقعی روی خود عنصر کار کند؛
   400→      // کپچر روی SVG به‌محض عبور از آستانه درگ فعال می‌شود.
   401→      movedRef.current = false;
   402→      panRef.current = { startX: e.clientX, startY: e.clientY, tx: tRef.current.tx, ty: tRef.current.ty };
   403→      pinchRef.current = null;
   404→    } else if (pointersRef.current.size === 2) {
   405→      // شروع پینچ — کپچر هر دو انگشت روی خود SVG (عنصر پایدار که هرگز
   406→      // unmount نمی‌شود؛ برخلاف برچسب‌ها که در میانه ژست حذف می‌شوند)
   407→      try {
   408→        svgRef.current?.setPointerCapture?.(e.pointerId);
   409→        for (const id of pointersRef.current.keys()) {
   410→          if (id !== e.pointerId) svgRef.current?.setPointerCapture?.(id);
   411→        }
   412→      } catch {
   413→        /* برخی محیط‌ها (رویدادهای ساختگی/WebView) خطای InvalidPointerId می‌دهند */
   414→      }
   415→      const [p1, p2] = [...pointersRef.current.values()];
   416→      const t0 = tRef.current;
   417→      const svg = svgRef.current;
   418→      if (!svg) return;
   419→      const rect = svg.getBoundingClientRect();
   420→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   421→      const ox = (rect.width - MAP_CANVAS.w * fit) / 2;
   422→      const oy = (rect.height - MAP_CANVAS.h * fit) / 2;
   423→      const mx = (p1.x + p2.x) / 2;
   424→      const my = (p1.y + p2.y) / 2;
   425→      pinchRef.current = {
   426→        dist0: Math.hypot(p1.x - p2.x, p1.y - p2.y) || 1,
   427→        vx0: (mx - rect.left - ox) / fit,
   428→        vy0: (my - rect.top - oy) / fit,
   429→        s0: t0.scale,
   430→        tx0: t0.tx,
   431→        ty0: t0.ty,
   432→      };
   433→      panRef.current = null;
   434→    }
   435→  };
   436→
   437→  const onPointerMove = (e: React.PointerEvent) => {
   438→    if (!pointersRef.current.has(e.pointerId)) return;
   439→    pointersRef.current.set(e.pointerId, { x: e.clientX, y: e.clientY, t: performance.now() });
   440→
   441→    if (pinchRef.current && pointersRef.current.size >= 2) {
   442→      // پینچ: مقیاس + حفظ لنگر زیر انگشتان
   443→      const [p1, p2] = [...pointersRef.current.values()];
   444→      const p0 = pinchRef.current;
   445→      const dist = Math.hypot(p1.x - p2.x, p1.y - p2.y) || 1;
   446→      const svg = svgRef.current;
   447→      if (!svg) return;
   448→      const rect = svg.getBoundingClientRect();
   449→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   450→      const ox = (rect.width - MAP_CANVAS.w * fit) / 2;
   451→      const oy = (rect.height - MAP_CANVAS.h * fit) / 2;
   452→      const mx = (p1.x + p2.x) / 2;
   453→      const my = (p1.y + p2.y) / 2;
   454→      const vxm = (mx - rect.left - ox) / fit;
   455→      const vym = (my - rect.top - oy) / fit;
   456→      const scale = clamp((p0.s0 * dist) / p0.dist0, MIN_SCALE, MAX_SCALE);
   457→      // نقطه میانی اولیه باید زیر انگشتان بماند
   458→      const cx = (p0.vx0 - p0.tx0) / p0.s0;
   459→      const cy = (p0.vy0 - p0.ty0) / p0.s0;
   460→      commit({
   461→        scale,
   462→        tx: vxm - scale * cx,
   463→        ty: vym - scale * cy,
   464→      });
   465→      movedRef.current = true;
   466→      return;
   467→    }
   468→
   469→    if (panRef.current && pointersRef.current.size === 1) {
   470→      const p0 = panRef.current;
   471→      const svg = svgRef.current;
   472→      if (!svg) return;
   473→      const rect = svg.getBoundingClientRect();
   474→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   475→      const dx = (e.clientX - p0.startX) / fit;
   476→      const dy = (e.clientY - p0.startY) / fit;
   477→      if (Math.abs(e.clientX - p0.startX) > TAP_SLOP_PX || Math.abs(e.clientY - p0.startY) > TAP_SLOP_PX) {
   478→        if (!movedRef.current) {
   479→          // گذر از آستانه درگ — کپچر روی SVG تا pointerup حتی روی کارت‌های
   480→          // HTML یا بیرون نقشه هم تضمین‌شده به همین عنصر برسد
   481→          try {
   482→            svgRef.current?.setPointerCapture?.(e.pointerId);
   483→          } catch {
   484→            /* محیط‌های ساختگی */
   485→          }
   486→        }
   487→        movedRef.current = true;
   488→      }
   489→      commit({ scale: tRef.current.scale, tx: p0.tx + dx, ty: p0.ty + dy });
   490→    }
   491→  };
   492→
   493→  const endPointer = (e: React.PointerEvent) => {
   494→    try {
   495→      svgRef.current?.releasePointerCapture?.(e.pointerId);
   496→    } catch {
   497→      /* قبلاً آزاد شده */
   498→    }
   499→    pointersRef.current.delete(e.pointerId);
   500→    if (pointersRef.current.size === 1) {
   501→      // یک انگشت باقی مانده — پینچ تمام شده، ادامه با جابجایی
   502→      pinchRef.current = null;
   503→      const [p] = [...pointersRef.current.values()];
   504→      panRef.current = { startX: p.x, startY: p.y, tx: tRef.current.tx, ty: tRef.current.ty };
   505→    } else if (pointersRef.current.size === 0) {
   506→      panRef.current = null;
   507→      pinchRef.current = null;
   508→    }
   509→  };
   510→
   511→  /* ---------- مرکز کردن دید روی یک ایستگاه ---------- */
   512→  const zoomToStation = useCallback(
   513→    (sx: number, sy: number) => {
   514→      const svg = svgRef.current;
   515→      if (!svg) return;
   516→      const rect = svg.getBoundingClientRect();
   517→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   518→      const scale = 2.4;
   519→      commit({
   520→        scale,
   521→        tx: rect.width / (2 * fit) - scale * sx,
   522→        ty: rect.height / (2 * fit) - scale * sy,
   523→      });
   524→    },
   525→    [commit]
   526→  );
   527→
   528→  /* ---------- موقعیت‌یاب GPS ---------- */
   529→  const locateMe = useCallback(() => {
   530→    if (locate.kind === "busy") return;
   531→    /* احترام به کلید GPS در تنظیمات */
   532→    try {
   533→      if (localStorage.getItem("mj-gps-enabled") === "false") {
   534→        setLocate({ kind: "error", message: "دسترسی موقعیت مکانی در تنظیمات غیرفعال است" });
   535→        return;
   536→      }
   537→    } catch {
   538→      /* حافظه در دسترس نیست — ادامه */
   539→    }
   540→    setLocate({ kind: "busy" });
   541→    void (async () => {
   542→      // روی اندروید مجوز لحظه‌ای با دیالوگ سیستم گرفته می‌شود
   543→      const perm = await requestGeoPermission();
   544→      if (perm === "denied") {
   545→        setLocate({ kind: "error", message: "دسترسی موقعیت مکانی داده نشد" });
   546→        return;
   547→      }
   548→      if (perm === "unavailable") {
   549→        setLocate({ kind: "error", message: "مرورگر از موقعیت‌یابی پشتیبانی نمی‌کند" });
   550→        return;
   551→      }
   552→      const fix = await getCurrentGeo(15000);
   553→      if (!fix) {
   554→        setLocate({ kind: "error", message: "موقعیت مکانی در دسترس نیست" });
   555→        return;
   556→      }
   557→      const near = nearestStationTo(fix.lat, fix.lng);
   558→      if (!near) {
   559→        setLocate({ kind: "far" });
   560→        return;
   561→      }
   562→      const st = MAP_STATION_BY_ID.get(near.id);
   563→      if (!st) {
   564→        setLocate({ kind: "far" });
   565→        return;
   566→      }
   567→      setLocate({
   568→        kind: "ok",
   569→        stationId: near.id,
   570→        stationName: near.name,
   571→        distanceM: near.distanceM,
   572→        accuracyM: fix.accuracy,
   573→      });
   574→      // بزرگ‌نمایی روی ایستگاه
   575→      zoomToStation(st.x, st.y);
   576→    })();
   577→  }, [locate.kind, zoomToStation]);
   578→
   579→  const resetView = useCallback(() => {
   580→    commit({ scale: 1, tx: 0, ty: 0 });
   581→    setLocate({ kind: "idle" });
   582→  }, [commit]);
   583→
   584→  const highlighted = useMemo(() => {
   585→    const set = new Set<string>();
   586→    for (const id of highlightStations ?? []) {
   587→      // ایستگاه‌های خطوط خاموش رندر نمی‌شوند
   588→      if (visible.stationIds.has(id)) set.add(id);
   589→    }
   590→    if (locate.kind === "ok" && visible.stationIds.has(locate.stationId)) set.add(locate.stationId);
   591→    return set;
   592→  }, [highlightStations, locate, visible.stationIds]);
   593→
   594→  /* ============================================================
   595→     موتور برچسب‌ها — مقیاس «فضای صفحه»:
   596→     برخلاف گذشته که متن‌ها داخل گروه بزرگ‌نمایی‌شونده بودند (و در
   597→     بزرگ‌نمایی ۱ روی هم ریخته دیده می‌شدند)، اینجا هر برچسب:
   598→     ۱. با اندازه ثابت روی صفحه رندر می‌شود (مستقل از زوم)
   599→     ۲. زنده از مختصات ایستگاه پیروی می‌کند
   600→     ۳. فقط اگر با برچسب مهم‌تری همپوشانی نداشته باشد نمایش
   601→        داده می‌شود — با نزدیک‌تر شدن (زوم بیشتر) برچسب‌های
   602→        بیشتری جا می‌گیرند و ظاهر می‌شوند
   603→     ۴. روی «بدنه خطوط» نمی‌افتد (شبکه اشغال خطوط) و در صورت
   604→        کمبود جا، بالای/پایین ایستگاه هم امتحان می‌شود
   605→     ============================================================ */
   606→  const fit = box.w > 0 && box.h > 0 ? Math.min(box.w / MAP_CANVAS.w, box.h / MAP_CANVAS.h) : 0;
   607→
   608→  /** چک اشغال خطوط — جعبه برچسب (فضای صفحه) به فضای بوم برگردانده می‌شود */
   609→  const hitsStroke = useCallback(
   610→    (b: Box) => {
   611→      const { grid, gw, gh } = strokeGrid;
   612→      const cx0 = (b.x0 - t.tx) / t.scale;
   613→      const cy0 = (b.y0 - t.ty) / t.scale;
   614→      const cx1 = (b.x1 - t.tx) / t.scale;
   615→      const cy1 = (b.y1 - t.ty) / t.scale;
   616→      const c0x = Math.max(0, Math.floor(cx0 / GRID_CELL));
   617→      const c1x = Math.min(gw - 1, Math.floor(cx1 / GRID_CELL));
   618→      const c0y = Math.max(0, Math.floor(cy0 / GRID_CELL));
   619→      const c1y = Math.min(gh - 1, Math.floor(cy1 / GRID_CELL));
   620→      for (let cy = c0y; cy <= c1y; cy++) {
   621→        for (let cx = c0x; cx <= c1x; cx++) {
   622→          if (grid[cy * gw + cx]) return true;
   623→        }
   624→      }
   625→      return false;
   626→    },
   627→    [strokeGrid, t.scale, t.tx, t.ty]
   628→  );
   629→
   630→  const labelLayout = useMemo(() => {
   631→    const empty: { labels: LabelItem[]; badges: BadgeItem[] } = { labels: [], badges: [] };
   632→    if (fit <= 0) return empty;
   633→    const toVb = (px: number) => px / fit;
   634→
   635→    const boxes: Box[] = [];
   636→    const overlaps = (b: Box) =>
   637→      boxes.some((o) => b.x0 < o.x1 && b.x1 > o.x0 && b.y0 < o.y1 && b.y1 > o.y0);
   638→
   639→    /* ---------- ناحیه کارت‌های HTML روی نقشه (راهنما + کنترل‌ها) —
   640→       برچسب و نشان نباید زیر این کارت‌ها پنهان شوند ---------- */
   641→    const ox = (box.w - MAP_CANVAS.w * fit) / 2;
   642→    const oy = (box.h - MAP_CANVAS.h * fit) / 2;
   643→    const htmlZone = (left: number, top: number, w: number, h: number): Box | null => {
   644→      const x0 = Math.max(0, (left - ox) / fit);
   645→      const y0 = Math.max(0, (top - oy) / fit);
   646→      const x1 = Math.min(MAP_CANVAS.w, (left + w - ox) / fit);
   647→      const y1 = Math.min(MAP_CANVAS.h, (top + h - oy) / fit);
   648→      if (x1 <= x0 || y1 <= y0) return null;
   649→      return { x0, y0, x1, y1 };
   650→    };
   651→    const legendZone = htmlZone(10, 10, 158, legendOpen ? (onToggleLine ? 240 : 185) : 40);
   652→    if (legendZone) boxes.push(legendZone);
   653→    // ستون دکمه‌های شناور پایین-چپ (۴ دکمه ۳۶px با فاصله)
   654→    const controlsZone = htmlZone(8, box.h - 196, 44, 188);
   655→    if (controlsZone) boxes.push(controlsZone);
   656→
   657→    /* ---------- نشان شماره خطوط (اندازه ثابت، پشته‌ای در ایستگاه مشترک) ---------- */
   658→    const badges: BadgeItem[] = [];
   659→    const badgeN = new Map<string, number>();
   660→    for (const ml of visible.lines) {
   661→      for (const id of [ml.stations[0], ml.stations[ml.stations.length - 1]]) {
   662→        const s = MAP_STATION_BY_ID.get(id);
   663→        if (!s) continue;
   664→        const n = badgeN.get(id) ?? 0;
   665→        badgeN.set(id, n + 1);
   666→        const r = toVb(8);
   667→        const side = s.x > MAP_CANVAS.w / 2 ? -1 : 1;
   668→        const bx = s.x * t.scale + t.tx + side * toVb(14);
   669→        const by = s.y * t.scale + t.ty - toVb(15) - n * toVb(19);
   670→        if (bx < -r || bx > MAP_CANVAS.w + r || by < -r || by > MAP_CANVAS.h + r) continue;
   671→        const bBox: Box = {
   672→          x0: bx - r - toVb(1),
   673→          y0: by - r - toVb(1),
   674→          x1: bx + r + toVb(1),
   675→          y1: by + r + toVb(1),
   676→        };
   677→        // نشان زیر کارت‌های HTML پنهان نمی‌شود
   678→        if (overlaps(bBox)) continue;
   679→        badges.push({
   680→          key: `${ml.lineId}-${id}-${n}`,
   681→          x: bx,
   682→          y: by,
   683→          r,
   684→          stroke: toVb(1.5),
   685→          font: toVb(9.5),
   686→          num: toFa(TEHRAN_LINES.find((l) => l.id === ml.lineId)?.number ?? 0),
   687→          color: ml.color,
   688→          ink: ml.lineId === "l4" ? "#1a1a1a" : "#ffffff",
   689→        });
   690→        boxes.push(bBox);
   691→      }
   692→    }
   693→
   694→    /* ---------- ناحیه حفاظتی نشان بزرگ ایستگاه‌های تبادلی ---------- */
   695→    const interExcl: { id: string; box: Box }[] = [];
   696→    for (const s of MAP_STATIONS) {
   697→      if (!visible.inter.has(s.id)) continue;
   698→      const half = 6.4 * t.scale + 3 / fit;
   699→      const X = s.x * t.scale + t.tx;
   700→      const Y = s.y * t.scale + t.ty;
   701→      interExcl.push({
   702→        id: s.id,
   703→        box: { x0: X - half, y0: Y - half, x1: X + half, y1: Y + half },
   704→      });
   705→    }
   706→    const hitsInterchange = (b: Box, selfId: string) =>
   707→      interExcl.some(
   708→        (e) => e.id !== selfId && b.x0 < e.box.x1 && b.x1 > e.box.x0 && b.y0 < e.box.y1 && b.y1 > e.box.y0
   709→      );
   710→
   711→    /* ---------- برچسب‌ها به ترتیب اولویت (تبادلی ← پرچم‌دار ← عادی) ----------
   712→       فقط ایستگاه‌های خطوط نمایان */
   713→    const cands = MAP_STATIONS.filter((s: MapStation) => visible.stationIds.has(s.id)).map((s: MapStation) => {
   714→      const isInter = visible.inter.has(s.id);
   715→      const prio = highlighted.has(s.id)
   716→        ? 1000
   717→        : isInter && s.label
   718→          ? 500
   719→          : isInter
   720→            ? 400
   721→            : s.label
   722→              ? 300
   723→              : 200;
   724→      return { s, prio, major: isInter || s.label };
   725→    }).sort((a, b) => b.prio - a.prio);
   726→
   727→    const labels: LabelItem[] = [];
   728→    for (const { s, prio, major } of cands) {
   729→      void prio;
   730→      const X = s.x * t.scale + t.tx;
   731→      const Y = s.y * t.scale + t.ty;
   732→      const margin = toVb(26);
   733→      if (X < -margin || X > MAP_CANVAS.w + margin || Y < -margin || Y > MAP_CANVAS.h + margin) continue;
   734→
   735→      const fontPx = major ? 10.5 : 9.5;
   736→      const weight = major ? 800 : 600;
   737→      const wVb = toVb(textWidthPx(s.name, weight, fontPx, fontsTick) + 7);
   738→      const hVb = toVb(fontPx * 1.1 + 5);
   739→      const gapVb = toVb(10);
   740→      // جهت متن فارسی rtl است: anchor «end» یعنی لبه چپ متن در x
   741→      const prefRight = s.x <= MAP_CANVAS.w / 2;
   742→      const sides: ("right" | "left")[] = prefRight ? ["right", "left"] : ["left", "right"];
   743→      // اگر در دو طرفِ هم‌سطح جا نشد: بالای و پایین ایستگاه هم امتحان می‌شود
   744→      const dyOptions = [0, hVb + 6, -(hVb + 6)];
   745→
   746→      let placed = false;
   747→      for (const dy of dyOptions) {
   748→        for (const side of sides) {
   749→          const bx0 = side === "right" ? X + gapVb : X - gapVb - wVb;
   750→          const by0 = Y + dy - hVb / 2;
   751→          const b: Box = { x0: bx0, y0: by0, x1: bx0 + wVb, y1: by0 + hVb };
   752→          if (b.x0 < 1 || b.x1 > MAP_CANVAS.w - 1 || b.y0 < 1 || b.y1 > MAP_CANVAS.h - 1) continue;
   753→          if (overlaps(b) || hitsInterchange(b, s.id) || hitsStroke(b)) continue;
   754→          boxes.push(b);
   755→          labels.push({
   756→            id: s.id,
   757→            name: s.name,
   758→            x: side === "right" ? X + gapVb : X - gapVb,
   759→            y: Y + dy + toVb(fontPx * 0.35),
   760→            anchor: side === "right" ? "end" : "start",
   761→            fontSize: toVb(fontPx),
   762→            weight,
   763→            halo: toVb(major ? 3.4 : 2.8),
   764→          });
   765→          placed = true;
   766→          break;
   767→        }
   768→        if (placed) break;
   769→      }
   770→    }
   771→
   772→    return { labels, badges };
   773→  }, [t, fit, fontsTick, highlighted, legendOpen, box.w, box.h, visible, hitsStroke, onToggleLine]);
   774→
   775→  return (
   776→    <div className={`relative overflow-hidden rounded-2xl ${className}`}>
   777→      {/* لایه SVG نقشه */}
   778→      <svg
   779→        ref={svgRef}
   780→        viewBox={`0 0 ${MAP_CANVAS.w} ${MAP_CANVAS.h}`}
   781→        className="w-full h-full map-viewport select-none"
   782→        style={{ touchAction: "none" }}
   783→        onPointerDown={onPointerDown}
   784→        onPointerMove={onPointerMove}
   785→        onPointerUp={endPointer}
   786→        onPointerCancel={endPointer}
   787→        role="img"
   788→        aria-label="نقشه شماتیک شبکه متروی تهران — با دو انگشت بزرگ‌نمایی کنید"
   789→      >
   790→        <defs>
   791→          <pattern id="map-grid" width="30" height="30" patternUnits="userSpaceOnUse">
   792→            <path d="M 30 0 L 0 0 0 30" fill="none" stroke="var(--mq-map-grid)" strokeWidth="0.7" />
   793→          </pattern>
   794→        </defs>
   795→        <rect width={MAP_CANVAS.w} height={MAP_CANVAS.h} fill="var(--mq-map-bg)" />
   796→        <rect width={MAP_CANVAS.w} height={MAP_CANVAS.h} fill="url(#map-grid)" />
   797→
   798→        <g transform={`translate(${t.tx} ${t.ty}) scale(${t.scale})`}>
   799→          {/* خطوط — فقط خطوط نمایان (کلید چشم)؛ هر خط با زیرلایه روشن */}
   800→          {visible.lines.map((ml) => (
   801→            <g key={ml.lineId}>
   802→              <polyline
   803→                points={linePoints(ml.stations)}
   804→                fill="none"
   805→                stroke="var(--mq-map-bg)"
   806→                strokeWidth={12}
   807→                strokeLinecap="round"
   808→                strokeLinejoin="round"
   809→              />
   810→              <polyline
   811→                points={linePoints(ml.stations)}
   812→                fill="none"
   813→                stroke={ml.color}
   814→                strokeWidth={8}
   815→                strokeLinecap="round"
   816→                strokeLinejoin="round"
   817→              />
   818→            </g>
   819→          ))}
   820→
   821→          {/* ایستگاه‌ها — فقط ایستگاه‌های خطوط نمایان؛ حلقه با رنگ خط (سبک نقشه کلاسیک) */}
   822→          {MAP_STATIONS.map((s) => {
   823→            if (!visible.stationIds.has(s.id)) return null;
