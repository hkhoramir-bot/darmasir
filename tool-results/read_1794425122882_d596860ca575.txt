   137→/* ------------------------- صفحه اصلی ------------------------- */
   138→export default function StationDetailPage() {
   139→  const params = useParams<{ id: string }>();
   140→  const id = params.id;
   141→  const station = useMemo(() => getStation(id), [id]);
   142→
   143→  if (!station) return <NotFound />;
   144→  return <StationDetail key={id} station={station} />;
   145→}
   146→
   147→function StationDetail({ station }: { station: StationDef }) {
   148→  const router = useRouter();
   149→  const { toast } = useToast();
   150→  const announcer = useAnnouncer();
   151→  const id = station.id;
   152→
   153→  const [tab, setTab] = useState<TabKey>("amenities");
   154→  const [isFav, setIsFav] = useState(false);
   155→  const [favPending, setFavPending] = useState(false);
   156→  const [arrivals, setArrivals] = useState<ArrivalRow[] | null>(null);
   157→  const [beforeStations, setBeforeStations] = useState<BeforeStations>(2);
   158→  const [alertOn, setAlertOn] = useState(false);
   159→
   160→  /* فرم گزارش */
   161→  const [eqType, setEqType] = useState<EquipmentType | null>(null);
   162→  const [status, setStatus] = useState<ReportStatus | null>(null);
   163→  const [note, setNote] = useState("");
   164→  const [submitting, setSubmitting] = useState(false);
   165→  const [reports, setReports] = useState<ReportRow[]>([]);
   166→
   167→  const subtitle =
   168→    station.lines.length > 1
   169→      ? `تقاطع ${station.lines.map((l) => toFa(getLine(l.lineId).number)).join(" و ")}`
   170→      : `خط ${toFa(getLine(station.lines[0].lineId).number)}`;
   171→
   172→  const amenities = useMemo(() => amenitiesLabels(station.amenities), [station.amenities]);
   173→  const exits = useMemo(() => exitsFor(station.id), [station.id]);
   174→  const landmarks = useMemo(() => landmarksFor(station.id), [station.id]);
   175→  const equipment = useMemo(() => equipmentFor(station.id), [station.id]);
   176→
   177→  /* ---------- تابلوی قطارهای بعدی — محاسبه محلی بر اساس جدول حرکت رسمی ----------
   178→     (با ساعت خودِ دستگاه کاربر محاسبه می‌شود تا دقیق باشد؛ هر ۳۰ ثانیه بروزرسانی) */
   179→  useEffect(() => {
   180→    let alive = true;
   181→    const build = () => {
   182→      const now = Date.now();
   183→      const d = new Date(now);
   184→      const nowMin = d.getHours() * 60 + d.getMinutes();
   185→      const rows = station.lines.map((ls) => {
   186→        const line = getLine(ls.lineId);
   187→        return {
   188→          lineId: line.id,
   189→          lineName: line.name,
   190→          lineNumber: line.number,
   191→          color: line.color,
   192→          textOnColor: line.textOnColor,
   193→          headwayLabel: toFa(headwayAt(line, nowMin)),
   194→          directions: nextArrivalsBoth(ls.lineId, station.id, now).map((dir) => ({
   195→            to: dir.to,
   196→            nextMin: dir.minutes[0] ?? null,
   197→            next2Min: dir.minutes[1] ?? null,
   198→            nextAt: dir.at[0] ?? null,
   199→            next2At: dir.at[1] ?? null,
   200→            outOfService: dir.outOfService,
   201→            crowd: dir.crowd,
   202→            crowdLabel: dir.crowdLabel,
   203→          })),
   204→          firstTrain: line.firstTrain,
   205→          lastTrain: line.lastTrain,
   206→        };
   207→      });
   208→      if (alive) setArrivals(rows);
   209→    };
   210→    build();
   211→    const t = setInterval(build, 30000);
   212→    return () => {
   213→      alive = false;
   214→      clearInterval(t);
   215→    };
   216→  }, [station]);
   217→
   218→  /* ---------- وضعیت نشان ---------- */
   219→  useEffect(() => {
   220→    let alive = true;
   221→    fetch("/api/favorites")
   222→      .then((r) => r.json())
   223→      .then((d: { ids?: string[] }) => {
   224→        if (alive) setIsFav(!!d.ids?.includes(id));
   225→      })
   226→      .catch(() => {});
   227→    return () => {
   228→      alive = false;
   229→    };
   230→  }, [id]);
   231→
   232→  const toggleFav = async () => {
   233→    if (favPending) return;
   234→    setFavPending(true);
   235→    const next = !isFav;
   236→    setIsFav(next);
   237→    try {
   238→      if (next) {
   239→        const res = await fetch("/api/favorites", {
   240→          method: "POST",
   241→          headers: { "Content-Type": "application/json" },
   242→          body: JSON.stringify({ stationId: id }),
   243→        });
   244→        if (!res.ok) throw new Error();
   245→        toast({ title: "به نشان‌ها اضافه شد", description: `${station.name} به فهرست نشان‌شده‌ها اضافه شد` });
   246→      } else {
   247→        const res = await fetch(`/api/favorites?stationId=${id}`, { method: "DELETE" });
   248→        if (!res.ok) throw new Error();
   249→        toast({ title: "از نشان‌ها حذف شد", description: `${station.name} از فهرست نشان‌شده‌ها حذف شد` });
   250→      }
   251→    } catch {
   252→      setIsFav(!next);
   253→      toast({ title: "خطا در ارتباط با سرور", description: "لطفاً دوباره تلاش کنید" });
   254→    } finally {
   255→      setFavPending(false);
   256→    }
   257→  };
   258→
   259→  /* ---------- اعلام صوتی ---------- */
   260→  const speak = useCallback(() => {
   261→    const first = arrivals?.[0]?.directions?.[0];
   262→    if (!first || first.outOfService || first.nextMin === null) {
   263→      announcer.announce(`ایستگاه ${station.name}. در حال حاضر قطاری در سرویس نیست`).catch(() => {});
   264→      return;
   265→    }
   266→    const mins = first.nextMin > 90 ? `ساعت ${first.nextAt ?? ""}` : `${toFa(first.nextMin)} دقیقه`;
   267→    announcer
   268→      .announce(`ایستگاه ${station.name}. قطار بعدی به سمت ${first.to}، ${mins}`)
   269→      .catch(() => {});
   270→  }, [arrivals, announcer, station.name]);
   271→
   272→  /* ---------- هشدار مقصد (بیدارباش هوشمند) ---------- */
   273→  useEffect(() => {
   274→    try {
   275→      const raw = localStorage.getItem("metrojo-alert");
   276→      if (!raw) return;
   277→      const a = JSON.parse(raw) as { stationId?: string; beforeStations?: number };
   278→      if (a.stationId === id) {
   279→        setAlertOn(true);
   280→        if (a.beforeStations === 2 || a.beforeStations === 3 || a.beforeStations === 4) {
   281→          setBeforeStations(a.beforeStations);
   282→        }
   283→      }
   284→    } catch {}
   285→  }, [id]);
   286→
   287→  const saveAlert = (on: boolean, before: BeforeStations) => {
   288→    try {
   289→      if (on) {
   290→        localStorage.setItem(
   291→          "metrojo-alert",
   292→          JSON.stringify({ stationId: id, stationName: station.name, beforeStations: before })
   293→        );
   294→      } else {
   295→        localStorage.removeItem("metrojo-alert");
   296→      }
   297→    } catch {}
   298→  };
   299→
   300→  const pickBefore = (n: BeforeStations) => {
   301→    setBeforeStations(n);
   302→    if (alertOn) {
   303→      saveAlert(true, n);
   304→      toast({ title: "تنظیم هشدار بروزرسانی شد", description: `${toFa(n)} ایستگاه مانده به مقصد بیدار می‌شوید` });
   305→    }
   306→  };
   307→
   308→  const toggleAlert = () => {
   309→    const next = !alertOn;
   310→    setAlertOn(next);
   311→    saveAlert(next, beforeStations);
   312→    if (next) {
   313→      toast({ title: "هشدار مقصد فعال شد", description: `${toFa(beforeStations)} ایستگاه مانده به ${station.name} بیدار می‌شوید` });
   314→      announcer
   315→        .announce(`هشدار مقصد فعال شد. ${toFa(beforeStations)} ایستگاه مانده به مقصد بیدار می‌شوید.`)
   316→        .catch(() => {});
   317→    } else {
   318→      toast({ title: "هشدار مقصد غیرفعال شد", description: "می‌توانید هر زمان دوباره فعال کنید" });
   319→    }
   320→  };
   321→
   322→  /* ---------- گزارش تجهیزات ---------- */
   323→  useEffect(() => {
   324→    let alive = true;
   325→    fetch(`/api/reports?station=${id}`)
   326→      .then((r) => r.json())
   327→      .then((d: { reports?: ReportRow[] }) => {
   328→        if (alive) setReports(d.reports ?? []);
   329→      })
   330→      .catch(() => {});
   331→    return () => {
   332→      alive = false;
   333→    };
   334→  }, [id]);
   335→
   336→  const submitReport = async () => {
   337→    if (!eqType || !status || submitting) return;
   338→    setSubmitting(true);
   339→    try {
   340→      const res = await fetch("/api/reports", {
   341→        method: "POST",
   342→        headers: { "Content-Type": "application/json" },
   343→        body: JSON.stringify({
   344→          stationId: id,
   345→          equipmentType: eqType,
   346→          status,
   347→          note: note.trim() ? note.trim() : undefined,
   348→        }),
   349→      });
   350→      if (!res.ok) throw new Error();
   351→      toast({ title: "گزارش شما ثبت شد. سپاس از همکاری" });
   352→      setEqType(null);
   353→      setStatus(null);
   354→      setNote("");
   355→      const r = await fetch(`/api/reports?station=${id}`);
   356→      const d = (await r.json()) as { reports?: ReportRow[] };
   357→      setReports(d.reports ?? []);
   358→    } catch {
   359→      toast({ title: "ثبت گزارش ناموفق بود", description: "اتصال اینترنت را بررسی و دوباره تلاش کنید" });
   360→    } finally {
   361→      setSubmitting(false);
   362→    }
   363→  };
   364→
   365→  return (
   366→    <>
   367→      <PageHeader
   368→        title={station.name}
   369→        subtitle={subtitle}
   370→        actions={
   371→          <button
   372→            type="button"
   373→            aria-label={isFav ? "حذف از نشان‌ها" : "افزودن به نشان‌ها"}
   374→            aria-pressed={isFav}
   375→            onClick={toggleFav}
   376→            className={`w-10 h-10 flex items-center justify-center rounded-full active:scale-95 transition-all ${
   377→              isFav ? "text-amber-500" : "text-outline"
   378→            }`}
   379→          >
   380→            <MI name="star" size={22} fill={isFav} />
   381→          </button>
   382→        }
   383→      />
   384→
   385→      <main className="flex-1 w-full px-4 pt-3 pb-28 space-y-4">
   386→        {/* ================= کارت اطلاعات و تابلوی زنده ================= */}
   387→        <section className="bg-surface-container-lowest rounded-2xl p-4 border border-outline-variant/40 shadow-sm space-y-3">
   388→          <div className="flex items-center justify-between gap-2">
   389→            <div className="flex flex-wrap items-center gap-1.5">
   390→              {station.lines.map((ls) => {
   391→                const line = getLine(ls.lineId);
   392→                return (
   393→                  <span
   394→                    key={ls.lineId}
   395→                    className="rounded-lg px-1.5 py-1 text-[10px] font-bold shadow-sm"
   396→                    style={{ backgroundColor: line.color, color: line.textOnColor }}
   397→                  >
   398→                    خط {toFa(line.number)}
   399→                  </span>
   400→                );
   401→              })}
   402→            </div>
   403→            <button
   404→              type="button"
   405→              onClick={speak}
   406→              aria-label="اعلام صوتی اطلاعات ایستگاه"
   407→              className="w-9 h-9 rounded-full bg-primary/10 text-primary flex items-center justify-center active:scale-95 transition-all shrink-0"
   408→            >
   409→              <MI name="volume_up" size={18} />
   410→            </button>
   411→          </div>
   412→
   413→          {/* تابلوی قطارهای بعدی — بر اساس جدول حرکت رسمی */}
   414→          <div className="flex items-center justify-between">
   415→            <div className="flex items-center gap-2">
   416→              <MI name="departure_board" size={20} className="text-primary" />
   417→              <h3 className="font-bold text-xs text-on-surface">قطارهای بعدی</h3>
   418→            </div>
   419→            <span className="text-[10px] font-bold text-tertiary bg-tertiary-fixed/40 px-2 py-0.5 rounded-full flex items-center gap-1">
   420→              <MI name="schedule" size={12} />
   421→              بر اساس جدول حرکت
   422→            </span>
   423→          </div>
   424→
   425→          {arrivals === null ? (
   426→            <div className="space-y-2" aria-hidden>
   427→              <div className="bg-surface-container-low rounded-xl h-12 animate-pulse" />
   428→              <div className="bg-surface-container-low rounded-xl h-12 animate-pulse" />
   429→            </div>
   430→          ) : (
   431→            <div className="divide-y divide-outline-variant/30">
   432→              {arrivals.map((a) =>
   433→                a.directions.map((d) => {
   434→                  const c = CROWD_COLOR[d.crowd];
   435→                  return (
   436→                    <div key={`${a.lineId}-${d.to}`} className="py-2.5 space-y-1.5 first:pt-0 last:pb-0">
   437→                      <div className="flex items-center justify-between gap-2">
   438→                        <div className="flex items-center gap-1.5 min-w-0">
   439→                          <span
   440→                            className="rounded px-1 py-0.5 text-[9px] font-bold shrink-0"
   441→                            style={{ backgroundColor: a.color, color: a.textOnColor }}
   442→                          >
   443→                            {toFa(a.lineNumber)}
   444→                          </span>
   445→                          <span className="text-[11px] font-bold text-on-surface truncate">
   446→                            به سمت {d.to}
   447→                          </span>
   448→                        </div>
   449→                        <span
   450→                          className="text-[9px] font-bold px-1.5 py-0.5 rounded-full flex items-center gap-1 shrink-0"
   451→                          style={{ color: c, backgroundColor: `${c}14` }}
   452→                        >
   453→                          <span className="w-1 h-1 rounded-full shrink-0" style={{ backgroundColor: c }} />
   454→                          {d.crowdLabel}
   455→                        </span>
   456→                      </div>
   457→                      {d.outOfService || d.nextMin === null ? (
   458→                        <p className="text-[10px] text-outline font-medium">
   459→                          سرویس امروز پایان یافته — اولین قطار فردا ساعت {toFa(a.firstTrain)}
   460→                        </p>
   461→                      ) : (
   462→                        <div className="flex items-center gap-1.5">
   463→                          <span className="bg-surface-container-low rounded-lg px-2 py-1 text-[10px] font-bold text-primary">
   464→                            {d.nextMin > 90 ? `ساعت ${toFa(d.nextAt ?? "")}` : `${toFa(d.nextMin)} دقیقه`}
   465→                          </span>
   466→                          {d.next2Min !== null && (
   467→                            <span className="bg-surface-container-low rounded-lg px-2 py-1 text-[10px] font-bold text-on-surface-variant">
   468→                              {d.next2Min > 90 ? `ساعت ${toFa(d.next2At ?? "")}` : `${toFa(d.next2Min)} دقیقه`}
   469→                            </span>
   470→                          )}
   471→                          <span className="text-[9px] text-outline ms-auto">هر {toFa(a.headwayLabel)} دقیقه یک قطار</span>
   472→                        </div>
   473→                      )}
   474→                    </div>
   475→                  );
   476→                })
   477→              )}
   478→            </div>
   479→          )}
   480→        </section>
   481→
   482→        {/* ================= تب‌ها ================= */}
   483→        <div className="bg-surface-container-low p-1 rounded-full flex items-center" role="tablist" aria-label="بخش‌های ایستگاه">
   484→          {TABS.map((t) => (
   485→            <button
   486→              key={t.key}
   487→              type="button"
   488→              role="tab"
   489→              aria-selected={tab === t.key}
   490→              onClick={() => setTab(t.key)}
   491→              className={`flex-1 text-center py-1.5 rounded-full text-[11px] active:scale-95 transition-all ${
   492→                tab === t.key
