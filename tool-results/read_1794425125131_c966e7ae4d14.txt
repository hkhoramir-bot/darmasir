   146→   146→
   147→   147→function StationDetail({ station }: { station: StationDef }) {
   148→   148→  const router = useRouter();
   149→   149→  const { toast } = useToast();
   150→   150→  const announcer = useAnnouncer();
   151→   151→  const id = station.id;
   152→   152→
   153→   153→  const [tab, setTab] = useState<TabKey>("amenities");
   154→   154→  const [isFav, setIsFav] = useState(false);
   155→   155→  const [favPending, setFavPending] = useState(false);
   156→   156→  const [arrivals, setArrivals] = useState<ArrivalRow[] | null>(null);
   157→   157→  const [beforeStations, setBeforeStations] = useState<BeforeStations>(2);
   158→   158→  const [alertOn, setAlertOn] = useState(false);
   159→   159→
   160→   160→  /* فرم گزارش */
   161→   161→  const [eqType, setEqType] = useState<EquipmentType | null>(null);
   162→   162→  const [status, setStatus] = useState<ReportStatus | null>(null);
   163→   163→  const [note, setNote] = useState("");
   164→   164→  const [submitting, setSubmitting] = useState(false);
   165→   165→  const [reports, setReports] = useState<ReportRow[]>([]);
   166→   166→
   167→   167→  const subtitle =
   168→   168→    station.lines.length > 1
   169→   169→      ? `تقاطع ${station.lines.map((l) => toFa(getLine(l.lineId).number)).join(" و ")}`
   170→   170→      : `خط ${toFa(getLine(station.lines[0].lineId).number)}`;
   171→   171→
   172→   172→  const amenities = useMemo(() => amenitiesLabels(station.amenities), [station.amenities]);
   173→   173→  const exits = useMemo(() => exitsFor(station.id), [station.id]);
   174→   174→  const landmarks = useMemo(() => landmarksFor(station.id), [station.id]);
   175→   175→  const equipment = useMemo(() => equipmentFor(station.id), [station.id]);
   176→   176→
   177→   177→  /* ---------- تابلوی قطارهای بعدی — محاسبه محلی بر اساس جدول حرکت رسمی ----------
   178→   178→     (با ساعت خودِ دستگاه کاربر محاسبه می‌شود تا دقیق باشد؛ هر ۳۰ ثانیه بروزرسانی) */
   179→   179→  useEffect(() => {
   180→   180→    let alive = true;
   181→   181→    const build = () => {
   182→   182→      const now = Date.now();
   183→   183→      const d = new Date(now);
   184→   184→      const nowMin = d.getHours() * 60 + d.getMinutes();
   185→   185→      const rows = station.lines.map((ls) => {
   186→   186→        const line = getLine(ls.lineId);
   187→   187→        return {
   188→   188→          lineId: line.id,
   189→   189→          lineName: line.name,
   190→   190→          lineNumber: line.number,
   191→   191→          color: line.color,
   192→   192→          textOnColor: line.textOnColor,
   193→   193→          headwayLabel: toFa(headwayAt(line, nowMin)),
   194→   194→          directions: nextArrivalsBoth(ls.lineId, station.id, now).map((dir) => ({
   195→   195→            to: dir.to,
   196→   196→            nextMin: dir.minutes[0] ?? null,
   197→   197→            next2Min: dir.minutes[1] ?? null,
   198→   198→            nextAt: dir.at[0] ?? null,
   199→   199→            next2At: dir.at[1] ?? null,
   200→   200→            outOfService: dir.outOfService,
   201→   201→            crowd: dir.crowd,
   202→   202→            crowdLabel: dir.crowdLabel,
   203→   203→          })),
   204→   204→          firstTrain: line.firstTrain,
   205→   205→          lastTrain: line.lastTrain,
   206→   206→        };
   207→   207→      });
   208→   208→      if (alive) setArrivals(rows);
   209→   209→    };
   210→   210→    build();
   211→   211→    const t = setInterval(build, 30000);
   212→   212→    return () => {
   213→   213→      alive = false;
   214→   214→      clearInterval(t);
   215→   215→    };
   216→   216→  }, [station]);
   217→   217→
   218→   218→  /* ---------- وضعیت نشان ---------- */
   219→   219→  useEffect(() => {
   220→   220→    let alive = true;
   221→   221→    fetch("/api/favorites")
   222→   222→      .then((r) => r.json())
   223→   223→      .then((d: { ids?: string[] }) => {
   224→   224→        if (alive) setIsFav(!!d.ids?.includes(id));
   225→   225→      })
   226→   226→      .catch(() => {});
   227→   227→    return () => {
   228→   228→      alive = false;
   229→   229→    };
   230→   230→  }, [id]);
   231→   231→
   232→   232→  const toggleFav = async () => {
   233→   233→    if (favPending) return;
   234→   234→    setFavPending(true);
   235→   235→    const next = !isFav;
   236→   236→    setIsFav(next);
   237→   237→    try {
   238→   238→      if (next) {
   239→   239→        const res = await fetch("/api/favorites", {
   240→   240→          method: "POST",
   241→   241→          headers: { "Content-Type": "application/json" },
   242→   242→          body: JSON.stringify({ stationId: id }),
   243→   243→        });
   244→   244→        if (!res.ok) throw new Error();
   245→   245→        toast({ title: "به نشان‌ها اضافه شد", description: `${station.name} به فهرست نشان‌شده‌ها اضافه شد` });
   246→   246→      } else {
   247→   247→        const res = await fetch(`/api/favorites?stationId=${id}`, { method: "DELETE" });
   248→   248→        if (!res.ok) throw new Error();
   249→   249→        toast({ title: "از نشان‌ها حذف شد", description: `${station.name} از فهرست نشان‌شده‌ها حذف شد` });
   250→   250→      }
   251→   251→    } catch {
   252→   252→      setIsFav(!next);
   253→   253→      toast({ title: "خطا در ارتباط با سرور", description: "لطفاً دوباره تلاش کنید" });
   254→   254→    } finally {
   255→   255→      setFavPending(false);
   256→   256→    }
   257→   257→  };
   258→   258→
   259→   259→  /* ---------- اعلام صوتی ---------- */
   260→   260→  const speak = useCallback(() => {
   261→   261→    const first = arrivals?.[0]?.directions?.[0];
   262→   262→    if (!first || first.outOfService || first.nextMin === null) {
   263→   263→      announcer.announce(`ایستگاه ${station.name}. در حال حاضر قطاری در سرویس نیست`).catch(() => {});
   264→   264→      return;
   265→   265→    }
   266→   266→    const mins = first.nextMin > 90 ? `ساعت ${first.nextAt ?? ""}` : `${toFa(first.nextMin)} دقیقه`;
   267→   267→    announcer
   268→   268→      .announce(`ایستگاه ${station.name}. قطار بعدی به سمت ${first.to}، ${mins}`)
   269→   269→      .catch(() => {});
   270→   270→  }, [arrivals, announcer, station.name]);
   271→   271→
   272→   272→  /* ---------- هشدار مقصد (بیدارباش هوشمند) ---------- */
   273→   273→  useEffect(() => {
   274→   274→    try {
   275→   275→      const raw = localStorage.getItem("metrojo-alert");
   276→   276→      if (!raw) return;
   277→   277→      const a = JSON.parse(raw) as { stationId?: string; beforeStations?: number };
   278→   278→      if (a.stationId === id) {
   279→   279→        setAlertOn(true);
   280→   280→        if (a.beforeStations === 2 || a.beforeStations === 3 || a.beforeStations === 4) {
   281→   281→          setBeforeStations(a.beforeStations);
   282→   282→        }
   283→   283→      }
   284→   284→    } catch {}
   285→   285→  }, [id]);
   286→   286→
   287→   287→  const saveAlert = (on: boolean, before: BeforeStations) => {
   288→   288→    try {
   289→   289→      if (on) {
   290→   290→        localStorage.setItem(
   291→   291→          "metrojo-alert",
   292→   292→          JSON.stringify({ stationId: id, stationName: station.name, beforeStations: before })
   293→   293→        );
   294→   294→      } else {
   295→   295→        localStorage.removeItem("metrojo-alert");
   296→   296→      }
   297→   297→    } catch {}
   298→   298→  };
   299→   299→
   300→   300→  const pickBefore = (n: BeforeStations) => {
   301→   301→    setBeforeStations(n);
   302→   302→    if (alertOn) {
   303→   303→      saveAlert(true, n);
   304→   304→      toast({ title: "تنظیم هشدار بروزرسانی شد", description: `${toFa(n)} ایستگاه مانده به مقصد بیدار می‌شوید` });
   305→   305→    }
   306→   306→  };
   307→   307→
   308→   308→  const toggleAlert = () => {
   309→   309→    const next = !alertOn;
   310→   310→    setAlertOn(next);
   311→   311→    saveAlert(next, beforeStations);
   312→   312→    if (next) {
   313→   313→      toast({ title: "هشدار مقصد فعال شد", description: `${toFa(beforeStations)} ایستگاه مانده به ${station.name} بیدار می‌شوید` });
   314→   314→      announcer
   315→   315→        .announce(`هشدار مقصد فعال شد. ${toFa(beforeStations)} ایستگاه مانده به مقصد بیدار می‌شوید.`)
   316→   316→        .catch(() => {});
   317→   317→    } else {
   318→   318→      toast({ title: "هشدار مقصد غیرفعال شد", description: "می‌توانید هر زمان دوباره فعال کنید" });
   319→   319→    }
   320→   320→  };
   321→   321→
   322→   322→  /* ---------- گزارش تجهیزات ---------- */
   323→   323→  useEffect(() => {
   324→   324→    let alive = true;
   325→   325→    fetch(`/api/reports?station=${id}`)
   326→   326→      .then((r) => r.json())
   327→   327→      .then((d: { reports?: ReportRow[] }) => {
   328→   328→        if (alive) setReports(d.reports ?? []);
   329→   329→      })
   330→   330→      .catch(() => {});
   331→   331→    return () => {
   332→   332→      alive = false;
   333→   333→    };
   334→   334→  }, [id]);
   335→   335→
   336→   336→  const submitReport = async () => {
   337→   337→    if (!eqType || !status || submitting) return;
   338→   338→    setSubmitting(true);
   339→   339→    try {
   340→   340→      const res = await fetch("/api/reports", {
   341→   341→        method: "POST",
   342→   342→        headers: { "Content-Type": "application/json" },
   343→   343→        body: JSON.stringify({
   344→   344→          stationId: id,
   345→   345→          equipmentType: eqType,
   346→   346→          status,
   347→   347→          note: note.trim() ? note.trim() : undefined,
   348→   348→        }),
   349→   349→      });
   350→   350→      if (!res.ok) throw new Error();
   351→   351→      toast({ title: "گزارش شما ثبت شد. سپاس از همکاری" });
   352→   352→      setEqType(null);
   353→   353→      setStatus(null);
   354→   354→      setNote("");
   355→   355→      const r = await fetch(`/api/reports?station=${id}`);
   356→   356→      const d = (await r.json()) as { reports?: ReportRow[] };
   357→   357→      setReports(d.reports ?? []);
   358→   358→    } catch {
   359→   359→      toast({ title: "ثبت گزارش ناموفق بود", description: "اتصال اینترنت را بررسی و دوباره تلاش کنید" });
   360→   360→    } finally {
   361→   361→      setSubmitting(false);
   362→   362→    }
   363→   363→  };
   364→   364→
   365→   365→  return (
   366→   366→    <>
   367→   367→      <PageHeader
   368→   368→        title={station.name}
   369→   369→        subtitle={subtitle}
   370→   370→        actions={
   371→   371→          <button
   372→   372→            type="button"
   373→   373→            aria-label={isFav ? "حذف از نشان‌ها" : "افزودن به نشان‌ها"}
   374→   374→            aria-pressed={isFav}
   375→   375→            onClick={toggleFav}
   376→   376→            className={`w-10 h-10 flex items-center justify-center rounded-full active:scale-95 transition-all ${
   377→   377→              isFav ? "text-amber-500" : "text-outline"
   378→   378→            }`}
   379→   379→          >
   380→   380→            <MI name="star" size={22} fill={isFav} />
   381→   381→          </button>
   382→   382→        }
   383→   383→      />
   384→   384→
   385→   385→      <main className="flex-1 w-full px-4 pt-3 pb-28 space-y-4">
   386→   386→        {/* ================= کارت اطلاعات و تابلوی زنده ================= */}
   387→   387→        <section className="bg-surface-container-lowest rounded-2xl p-4 border border-outline-variant/40 shadow-sm space-y-3">
   388→   388→          <div className="flex items-center justify-between gap-2">
   389→   389→            <div className="flex flex-wrap items-center gap-1.5">
   390→   390→              {station.lines.map((ls) => {
   391→   391→                const line = getLine(ls.lineId);
   392→   392→                return (
   393→   393→                  <span
   394→   394→                    key={ls.lineId}
   395→   395→                    className="rounded-lg px-1.5 py-1 text-[10px] font-bold shadow-sm"
   396→   396→                    style={{ backgroundColor: line.color, color: line.textOnColor }}
   397→   397→                  >
   398→   398→                    خط {toFa(line.number)}
   399→   399→                  </span>
   400→   400→                );
   401→   401→              })}
   402→   402→            </div>
   403→   403→            <button
   404→   404→              type="button"
   405→   405→              onClick={speak}
   406→   406→              aria-label="اعلام صوتی اطلاعات ایستگاه"
   407→   407→              className="w-9 h-9 rounded-full bg-primary/10 text-primary flex items-center justify-center active:scale-95 transition-all shrink-0"
   408→   408→            >
   409→   409→              <MI name="volume_up" size={18} />
   410→   410→            </button>
   411→   411→          </div>
   412→   412→
   413→   413→          {/* تابلوی قطارهای بعدی — بر اساس جدول حرکت رسمی */}
   414→   414→          <div className="flex items-center justify-between">
   415→   415→            <div className="flex items-center gap-2">
   416→   416→              <MI name="departure_board" size={20} className="text-primary" />
   417→   417→              <h3 className="font-bold text-xs text-on-surface">قطارهای بعدی</h3>
   418→   418→            </div>
   419→   419→            <span className="text-[10px] font-bold text-tertiary bg-tertiary-fixed/40 px-2 py-0.5 rounded-full flex items-center gap-1">
   420→   420→              <MI name="schedule" size={12} />
   421→   421→              بر اساس جدول حرکت
   422→   422→            </span>
   423→   423→          </div>
   424→   424→
