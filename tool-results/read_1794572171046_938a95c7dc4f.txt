    58→/* ------------------------------ کمکی‌ها ------------------------------ */

    60→function clockFa(d: Date): string {
    61→  return toFa(`${String(d.getHours()).padStart(2, "0")}:${String(d.getMinutes()).padStart(2, "0")}`);
    62→}
    63→
    64→function dayDiff(iso: string): number {
    65→  const d = new Date(iso);
    66→  if (Number.isNaN(d.getTime())) return 99;
    67→  const now = new Date();
    68→  const d0 = new Date(d.getFullYear(), d.getMonth(), d.getDate()).getTime();
    69→  const n0 = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
    70→  return Math.round((n0 - d0) / 86400000);
    71→}
    72→
    73→/** «امروز ساعت ۱۲:۳۰» / «دیروز ساعت ۱۸:۱۵» / «۳ روز پیش» */
    74→function relTime(iso: string): string {
    75→  const d = new Date(iso);
    76→  const diff = dayDiff(iso);
    77→  if (diff <= 0) return `امروز ساعت ${clockFa(d)}`;
    78→  if (diff === 1) return `دیروز ساعت ${clockFa(d)}`;
    79→  return `${toFa(diff)} روز پیش`;
    80→}
    81→
    82→/** «امروز» / «دیروز» / «۲ روز پیش» */
    83→function relShort(iso: string): string {
    84→  const diff = dayDiff(iso);
    85→  if (diff <= 0) return "امروز";
    86→  if (diff === 1) return "دیروز";
    87→  return `${toFa(diff)} روز پیش`;
    88→}

    90→/** تیره‌کردن رنگ خط برای متن چیپ روی زمینه همان رنگ با شفافیت کم */
    91→function blendDark(hex: string, t: number): string {
    92→  const m = hex.replace("#", "");
    93→  const r = parseInt(m.slice(0, 2), 16);
    94→  const g = parseInt(m.slice(2, 4), 16);
    95→  const b = parseInt(m.slice(4, 6), 16);
    96→  const f = (v: number) => Math.round(v * (1 - t));
    97→  return `rgb(${f(r)}, ${f(g)}, ${f(b)})`;
    98→}
