    50→    50→function computeVisibleStats(hidden: Set<string>) {
    51→    51→  const lines = MAP_LINES.filter((ml) => !hidden.has(ml.lineId));
    52→    52→  const stationIds = new Set<string>();
    53→    53→  const count = new Map<string, number>();
    54→    54→  const color = new Map<string, string>();
    55→    55→  for (const ml of lines) {
    56→    56→    for (const id of ml.stations) {
    57→    57→      stationIds.add(id);
    58→    58→      count.set(id, (count.get(id) ?? 0) + 1);
    59→    59→      if (!color.has(id)) color.set(id, ml.color);
    60→    60→    }
    61→    61→  }
    62→    62→  const inter = new Set([...count.entries()].filter(([, n]) => n > 1).map(([id]) => id));
    63→    63→  return { lines, stationIds, inter, color };
    64→    64→}
    65→    65→
    66→    66→/* ---------- شبکه اشغال «بدنه خطوط» برای جلوگیری از افتادن متن روی خط ----------
    67→    67→   سلول‌هایی که مرکزشان کمتر از GRID_PAD از هر قطعه خط فاصله دارد علامت
    68→    68→   می‌خورند؛ برچسب‌ها پیش از جای‌گذاری با این شبکه چک می‌شوند. */
    69→    69→const GRID_CELL = 12;
    70→    70→const GRID_PAD = 8;
    71→    71→
    72→    72→function buildStrokeGrid(lines: { stations: string[] }[]) {
    73→    73→  const gw = Math.ceil(MAP_CANVAS.w / GRID_CELL);
    74→    74→  const gh = Math.ceil(MAP_CANVAS.h / GRID_CELL);
    75→    75→  const grid = new Uint8Array(gw * gh);
    76→    76→  const ptSegDist = (
    77→    77→    px: number, py: number,
    78→    78→    ax: number, ay: number, bx: number, by: number
    79→    79→  ): number => {
    80→    80→    const dx = bx - ax, dy = by - ay;
    81→    81→    const len2 = dx * dx + dy * dy || 1;
    82→    82→    const t = Math.max(0, Math.min(1, ((px - ax) * dx + (py - ay) * dy) / len2));
    83→    83→    return Math.hypot(px - (ax + dx * t), py - (ay + dy * t));
    84→    84→  };
    85→    85→  for (const ml of lines) {
    86→    86→    const pts: { x: number; y: number }[] = [];
    87→    87→    for (const id of ml.stations) {
    88→    88→      const s = MAP_STATION_BY_ID.get(id);
    89→    89→      if (s) pts.push(s);
    90→    90→    }
    91→    91→    for (let i = 0; i < pts.length - 1; i++) {
    92→    92→      const a = pts[i];
    93→    93→      const b = pts[i + 1];
    94→    94→      const c0x = Math.max(0, Math.floor((Math.min(a.x, b.x) - GRID_PAD) / GRID_CELL));
    95→    95→      const c1x = Math.min(gw - 1, Math.floor((Math.max(a.x, b.x) + GRID_PAD) / GRID_CELL));
    96→    96→      const c0y = Math.max(0, Math.floor((Math.min(a.y, b.y) - GRID_PAD) / GRID_CELL));
    97→    97→      const c1y = Math.min(gh - 1, Math.floor((Math.max(a.y, b.y) + GRID_PAD) / GRID_CELL));
    98→    98→      for (let cy = c0y; cy <= c1y; cy++) {
    99→    99→        for (let cx = c0x; cx <= c1x; cx++) {
   100→   100→          const px = cx * GRID_CELL + GRID_CELL / 2;
   101→   101→          const py = cy * GRID_CELL + GRID_CELL / 2;
   102→   102→          if (ptSegDist(px, py, a.x, a.y, b.x, b.y) <= GRID_PAD) grid[cy * gw + cx] = 1;
   103→   103→        }
   104→   104→      }
   105→   105→    }
   106→   106→  }
   107→   107→  return { grid, gw, gh };
   108→   108→}

   122→   122→let measureCtx: CanvasRenderingContext2D | null = null;

   124→   124→
   125→   125→function textWidthPx(text: string, weight: number, px: number, epoch: number): number {
   126→   126→  const key = `${epoch}|${weight}|${px}|${text}`;
   127→   127→  const hit = widthCache.get(key);
   128→   128→  if (hit !== undefined) return hit;
   129→   129→  let w: number;
   130→   130→  try {
   131→   131→    if (!measureCtx) {
   132→   132→      measureCtx = document.createElement("canvas").getContext("2d");
   133→   133→    }
   134→   134→    if (measureCtx) {
   135→   135→      measureCtx.font = `${weight} ${px}px Vazirmatn, sans-serif`;
   136→   136→      w = measureCtx.measureText(text).width;
   137→   137→    } else {
   138→   138→      w = text.length * px * 0.6;
   139→   139→    }
   140→   140→  } catch {
   141→   141→    w = text.length * px * 0.6;
   142→   142→  }
   143→   143→  widthCache.set(key, w);
   144→   144→  return w;
   145→   145→}

   185→   185→export default function MetroMap({
   186→   186→  onStationSelect,
   187→   187→  highlightStations,
   188→   188→  hiddenLineIds,
   189→   189→  onToggleLine,
   190→   190→  className = "",
   191→   191→}: MetroMapProps) {
   192→   192→  const [t, setT] = useState<Transform>({ scale: 1, tx: 0, ty: 0 });
   193→   193→  const [legendOpen, setLegendOpen] = useState(false);
   194→   194→  const [locate, setLocate] = useState<LocateState>({ kind: "idle" });
   195→   195→  const [box, setBox] = useState({ w: 0, h: 0 });
   196→   196→  const [fontsTick, setFontsTick] = useState(0);
   197→   197→
   198→   198→  /* ---------- خطوط نمایان (کلید چشم) + شبکه اشغال خطوط ---------- */
   199→   199→  const hiddenSet = useMemo(() => new Set(hiddenLineIds ?? []), [hiddenLineIds]);
   200→   200→  const visible = useMemo(() => computeVisibleStats(hiddenSet), [hiddenSet]);
   201→   201→  const strokeGrid = useMemo(() => buildStrokeGrid(visible.lines), [visible.lines]);
   202→   202→
   203→   203→  /* ---------- آینه همزمان ترنسفورم — جلوگیری از رقابت با React ---------- */
   204→   204→  const tRef = useRef<Transform>({ scale: 1, tx: 0, ty: 0 });
   205→   205→  const svgRef = useRef<SVGSVGElement>(null);
   206→   206→
   207→   207→  /* اشاره‌گرهای فعال — پایه ژست چندلمسی (t = آخرین زمان دیدن اشاره‌گر) */
   208→   208→  const pointersRef = useRef<Map<number, { x: number; y: number; t: number }>>(new Map());
   209→   209→  const panRef = useRef<{ startX: number; startY: number; tx: number; ty: number } | null>(null);
   210→   210→  const pinchRef = useRef<{
   211→   211→    dist0: number;
   212→   212→    vx0: number;
   213→   213→    vy0: number;
   214→   214→    s0: number;
   215→   215→    tx0: number;
   216→   216→    ty0: number;
   217→   217→  } | null>(null);
   218→   218→  const movedRef = useRef(false); // بعد از جابجایی، تپ ایستگاه نادیده گرفته می‌شود
   219→   219→
   220→   220→  /* ---------- اندازه واقعی بوم برای مقیاس ثابت متن ---------- */
   221→   221→  useEffect(() => {
   222→   222→    const el = svgRef.current;
   223→   223→    if (!el) return;
   224→   224→    const update = () => {
   225→   225→      const r = el.getBoundingClientRect();
   226→   226→      if (r.width < 2 || r.height < 2) return;
   227→   227→      setBox((b) =>
   228→   228→        Math.abs(b.w - r.width) > 0.5 || Math.abs(b.h - r.height) > 0.5
   229→   229→          ? { w: r.width, h: r.height }
   230→   230→          : b
   231→   231→      );
   232→   232→    };
   233→   233→    update();
   234→   234→    const ro = new ResizeObserver(update);
   235→   235→    ro.observe(el);
   236→   236→    return () => ro.disconnect();
   237→   237→  }, []);
   238→   238→
   239→   239→  /* ---------- بعد از لود کامل فونت، عرض متن‌ها دوباره اندازه‌گیری می‌شود ---------- */
   240→   240→  useEffect(() => {
   241→   241→    let alive = true;
   242→   242→    try {
   243→   243→      document.fonts?.ready
   244→   244→        .then(() => {
   245→   245→          if (!alive) return;
   246→   246→          widthCache.clear();
   247→   247→          setFontsTick((v) => v + 1);
   248→   248→        })
   249→   249→        .catch(() => {});
   250→   250→    } catch {
   251→   251→      /* مرورگرهای قدیمی */
   252→   252→    }
   253→   253→    return () => {
   254→   254→      alive = false;
   255→   255→    };
   256→   256→  }, []);
   257→   257→
   258→   258→  const clamp = (v: number, min: number, max: number) => Math.min(max, Math.max(min, v));
   259→   259→
   260→   260→  /** ثبت ترنسفورم جدید — همیشه از طریق این تابع */
   261→   261→  const commit = useCallback((next: Transform) => {
   262→   262→    const scale = clamp(next.scale, MIN_SCALE, MAX_SCALE);
   263→   263→    /* پنجره دید واقعی بر حسب واحدهای viewBox — با احتساب letterbox حالت meet.
   264→   264→       clamp قبلی فقط مرکز بوم را مبنا می‌گرفت و در بزرگ‌نمایی اجازه می‌داد
   265→   265→       کل نقشه از پنجره دید خارج شود (کاربر حس می‌کرد pan خراب است). */
   266→   266→    let wx0 = 0, wy0 = 0, winW = MAP_CANVAS.w, winH = MAP_CANVAS.h;
   267→   267→    const svg = svgRef.current;
   268→   268→    if (svg) {
   269→   269→      const rect = svg.getBoundingClientRect();
   270→   270→      if (rect.width > 2 && rect.height > 2) {
   271→   271→        const f = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   272→   272→        wx0 = -((rect.width - MAP_CANVAS.w * f) / 2) / f;
   273→   273→        wy0 = -((rect.height - MAP_CANVAS.h * f) / 2) / f;
   274→   274→        winW = rect.width / f;
   275→   275→        winH = rect.height / f;
   276→   276→      }
   277→   277→    }
   278→   278→    const OVER = 60; // اجازه سرریز اندک در لبه‌ها
   279→   279→    // لبه محتوا نباید از پنجره دید بیرون بزند — در هر بزرگ‌نمایی کل نقشه قابل رسیدن است
   280→   280→    const loX = wx0 + winW - scale * MAP_CANVAS.w - OVER;
   281→   281→    const hiX = wx0 + OVER;
   282→   282→    const loY = wy0 + winH - scale * MAP_CANVAS.h - OVER;
   283→   283→    const hiY = wy0 + OVER;
   284→   284→    const tx = loX <= hiX ? clamp(next.tx, loX, hiX) : (loX + hiX) / 2;
   285→   285→    const ty = loY <= hiY ? clamp(next.ty, loY, hiY) : (loY + hiY) / 2;
   286→   286→    const fixed = { scale, tx, ty };
   287→   287→    tRef.current = fixed;
   288→   288→    setT(fixed);
   289→   289→  }, []);
   290→   290→
   291→   291→  /** تبدیل مختصات صفحه به مختصات viewBox (با احتساب letterbox وضعیت meet) */
   292→   292→  const viewPoint = useCallback((clientX: number, clientY: number) => {
   293→   293→    const svg = svgRef.current;
   294→   294→    if (!svg) return { vx: 0, vy: 0 };
   295→   295→    const rect = svg.getBoundingClientRect();
   296→   296→    const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   297→   297→    const ox = (rect.width - MAP_CANVAS.w * fit) / 2;
   298→   298→    const oy = (rect.height - MAP_CANVAS.h * fit) / 2;
   299→   299→    return {
   300→   300→      vx: (clientX - rect.left - ox) / (fit * tRef.current.scale) - tRef.current.tx / tRef.current.scale,
   301→   301→      vy: (clientY - rect.top - oy) / (fit * tRef.current.scale) - tRef.current.ty / tRef.current.scale,
   302→   302→    };
   303→   303→  }, []);
   304→   304→
   305→   305→  /** بزرگ‌نمایی حول یک نقطه صفحه (پیش‌فرض: مرکز) — لنگر ثابت می‌ماند */
   306→   306→  const zoomAt = useCallback(
   307→   307→    (factor: number, clientX?: number, clientY?: number) => {
   308→   308→      const t0 = tRef.current;
   309→   309→      const svg = svgRef.current;
   310→   310→      if (!svg) {
   311→   311→        commit({ ...t0, scale: t0.scale * factor });
   312→   312→        return;
   313→   313→      }
   314→   314→      const rect = svg.getBoundingClientRect();
   315→   315→      const cx = clientX ?? rect.left + rect.width / 2;
   316→   316→      const cy = clientY ?? rect.top + rect.height / 2;
   317→   317→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   318→   318→      const ox = (rect.width - MAP_CANVAS.w * fit) / 2;
   319→   319→      const oy = (rect.height - MAP_CANVAS.h * fit) / 2;
   320→   320→      // نقطه لنگر در مختصات viewBox فعلی
   321→   321→      const ax = (cx - rect.left - ox) / fit;
   322→   322→      const ay = (cy - rect.top - oy) / fit;
   323→   323→      const scale = clamp(t0.scale * factor, MIN_SCALE, MAX_SCALE);
   324→   324→      commit({
   325→   325→        scale,
   326→   326→        tx: ax - ((ax - t0.tx) * scale) / t0.scale,
   327→   327→        ty: ay - ((ay - t0.ty) * scale) / t0.scale,
   328→   328→      });
   329→   329→    },
   330→   330→    [commit]
   331→   331→  );
   332→   332→
   333→   333→  /* ---------- چرخ ماوس (غیر-Passive تا preventDefault کار کند) ---------- */
   334→   334→  useEffect(() => {
   335→   335→    const el = svgRef.current;
   336→   336→    if (!el) return;
   337→   337→    const onWheel = (e: WheelEvent) => {
   338→   338→      e.preventDefault();
   339→   339→      zoomAt(e.deltaY < 0 ? 1.15 : 0.87, e.clientX, e.clientY);
   340→   340→    };
   341→   341→    el.addEventListener("wheel", onWheel, { passive: false });
   342→   342→    return () => el.removeEventListener("wheel", onWheel);
   343→   343→  }, [zoomAt]);
   344→   344→
   345→   345→  /* ---------- تور ایمنی سراسری برای اشاره‌گرهای گم‌شده ----------
   346→   346→     اگر pointerup/pointercancel انگشتی به SVG نرسد (مثلاً انگشت روی کارت
   347→   347→     راهنما، دکمه‌های شناور یا بیرون نقشه بالا آمده باشد، یا برچسبِ زیر
   348→   348→     انگشت در میانه ژست از DOM حذف شود)، اشاره‌گر «مرده» در حافظه می‌ماند
   349→   349→     و ژستِ بعدی به‌جای جابجایی، پینچِ خراب با لنگر مرده اجرا می‌شد —
   350→   350→     همین باعث «ناتوانی در pan بعد از zoom» بود. این شنوندهٔ سراسری
   351→   351→     (فاز capture) هر اشاره‌گر گم‌شده را بلافاصله پاک می‌کند. */
   352→   352→  useEffect(() => {
   353→   353→    const release = (e: PointerEvent) => {
   354→   354→      if (!pointersRef.current.has(e.pointerId)) return;
   355→   355→      pointersRef.current.delete(e.pointerId);
   356→   356→      if (pointersRef.current.size === 1) {
   357→   357→        // یک انگشت باقی مانده — ادامه با جابجایی از موقعیت فعلی او
   358→   358→        pinchRef.current = null;
   359→   359→        const [p] = [...pointersRef.current.values()];
   360→   360→        panRef.current = { startX: p.x, startY: p.y, tx: tRef.current.tx, ty: tRef.current.ty };
   361→   361→      } else if (pointersRef.current.size === 0) {
   362→   362→        panRef.current = null;
   363→   363→        pinchRef.current = null;
   364→   364→      }
   365→   365→    };
   366→   366→    const resetAll = () => {
