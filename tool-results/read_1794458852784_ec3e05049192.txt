    15→/* ============================================================
    16→   نقشه تعاملی مترو — موتور ژست چندلمسی (پینچ/جابجایی بدون خطا)
    17→   + موتور برچسب «فضای صفحه»: متن همیشه با اندازه ثابت و خوانا،
    18→     موقعیت‌یابی زنده روی نقشه و حذف هوشمند همپوشانی

    28→interface Transform {
    29→  scale: number;
    30→  tx: number;
    31→  ty: number;
    32→}
    33→
    34→const MIN_SCALE = 0.75;
    35→const MAX_SCALE = 4;
    36→const TAP_SLOP_PX = 7; // حرکت بیشتر از این = جابجایی، نه لمس ایستگاه
    37→
    38→const TEHRAN_LINES = LINES.filter((l) => l.cityId === "tehran");

    40→/* ایستگاه‌های تبادلی (اشتراک دو خط یا بیشتر) — ثابت و مستقل از رندر */
    41→const INTERCHANGE_IDS = new Set(
    42→  MAP_LINES.flatMap((ml) => ml.stations).filter(
    43→    (id) => MAP_LINES.filter((ml) => ml.stations.includes(id)).length > 1
    44→  )
    45→);

    47→/** نقاط هر خط به‌صورت رشته points برای polyline */
    48→function linePoints(stations: string[]): string {
    49→  return stations
    50→    .map((id) => {
    51→      const s = MAP_STATION_BY_ID.get(id);
    52→      return s ? `${s.x},${s.y}` : "";
    53→    })
    54→    .filter(Boolean)
    55→    .join(" ");
    56→}

    58→/* ---------------- اندازه‌گیری واقعی عرض متن (برای جلوگیری از همپوشانی) ---------------- */
    59→let measureCtx: CanvasRenderingContext2D | null = null;
    60→const widthCache = new Map<string, number>();
    61→
    62→function textWidthPx(text: string, weight: number, px: number, epoch: number): number {
    63→  const key = `${epoch}|${weight}|${px}|${text}`;
    64→  const hit = widthCache.get(key);
    65→  if (hit !== undefined) return hit;
    66→  let w: number;
    67→  try {
    68→    if (!measureCtx) {
    69→      measureCtx = document.createElement("canvas").getContext("2d");
    70→    }
    71→    if (measureCtx) {
    72→      measureCtx.font = `${weight} ${px}px Vazirmatn, sans-serif`;
    73→      w = measureCtx.measureText(text).width;
    74→    } else {
    75→      w = text.length * px * 0.6;
    76→    }
    77→  } catch {
    78→    w = text.length * px * 0.6;
    79→  }
    80→  widthCache.set(key, w);
    81→  return w;
    82→}

    91→  fontSize: number;
    92→  weight: number;
    93→  halo: number;
    94→}
    95→

    97→  key: string;
    98→  x: number;
    99→  y: number;
   100→  r: number;

   104→  color: string;

   108→interface Box {
   109→  x0: number;
   110→  y0: number;
   111→  x1: number;
   112→  y1: number;
   113→}
   114→
   115→type LocateState =
   116→  | { kind: "idle" }
   117→  | { kind: "busy" }
   118→  | { kind: "ok"; stationId: string; stationName: string; distanceM: number }
   119→  | { kind: "far" }
   120→  | { kind: "error"; message: string };
   121→
   122→export default function MetroMap({
   123→  onStationSelect,
   124→  highlightStations,
   125→  className = "",
   126→}: MetroMapProps) {
   127→  const [t, setT] = useState<Transform>({ scale: 1, tx: 0, ty: 0 });
   128→  const [legendOpen, setLegendOpen] = useState(false);
   129→  const [locate, setLocate] = useState<LocateState>({ kind: "idle" });
   130→  const [box, setBox] = useState({ w: 0, h: 0 });
   131→  const [fontsTick, setFontsTick] = useState(0);
   132→
   133→  /* ---------- آینه همزمان ترنسفورم — جلوگیری از رقابت با React ---------- */
   134→  const tRef = useRef<Transform>({ scale: 1, tx: 0, ty: 0 });
   135→  const svgRef = useRef<SVGSVGElement>(null);
   136→
   137→  /* اشاره‌گرهای فعال — پایه ژست چندلمسی */
   138→  const pointersRef = useRef<Map<number, { x: number; y: number }>>(new Map());
   139→  const panRef = useRef<{ startX: number; startY: number; tx: number; ty: number } | null>(null);
   140→  const pinchRef = useRef<{
   141→    dist0: number;
   142→    vx0: number;
   143→    vy0: number;
   144→    s0: number;
   145→    tx0: number;
   146→    ty0: number;
   147→  } | null>(null);
   148→  const movedRef = useRef(false); // بعد از جابجایی، تپ ایستگاه نادیده گرفته می‌شود
   149→
   150→  /* ---------- اندازه واقعی بوم برای مقیاس ثابت متن ---------- */
   151→  useEffect(() => {
   152→    const el = svgRef.current;
   153→    if (!el) return;
   154→    const update = () => {
   155→      const r = el.getBoundingClientRect();
   156→      if (r.width < 2 || r.height < 2) return;
   157→      setBox((b) =>
   158→        Math.abs(b.w - r.width) > 0.5 || Math.abs(b.h - r.height) > 0.5
   159→          ? { w: r.width, h: r.height }
   160→          : b
   161→      );
   162→    };
   163→    update();
   164→    const ro = new ResizeObserver(update);
   165→    ro.observe(el);
   166→    return () => ro.disconnect();
   167→  }, []);
   168→
   169→  /* ---------- بعد از لود کامل فونت، عرض متن‌ها دوباره اندازه‌گیری می‌شود ---------- */
   170→  useEffect(() => {
   171→    let alive = true;
   172→    try {
   173→      document.fonts?.ready
   174→        .then(() => {
   175→          if (!alive) return;
   176→          widthCache.clear();
   177→          setFontsTick((v) => v + 1);
   178→        })
   179→        .catch(() => {});
   180→    } catch {
   181→      /* مرورگرهای قدیمی */
   182→    }
   183→    return () => {
   184→      alive = false;
   185→    };
   186→  }, []);
   187→
   188→  const clamp = (v: number, min: number, max: number) => Math.min(max, Math.max(min, v));
   189→
   190→  /** ثبت ترنسفورم جدید — همیشه از طریق این تابع */
   191→  const commit = useCallback((next: Transform) => {
   192→    const scale = clamp(next.scale, MIN_SCALE, MAX_SCALE);
   193→    const half = MAP_CANVAS.w / 2;
   194→    const halfH = MAP_CANVAS.h / 2;
   195→    const tx = clamp(next.tx, half - scale * half - 60, -(half - scale * half) + 60);
   196→    const ty = clamp(next.ty, halfH - scale * halfH - 60, -(halfH - scale * halfH) + 60);
   197→    const fixed = { scale, tx, ty };
   198→    tRef.current = fixed;
   199→    setT(fixed);
   200→  }, []);
   201→
   202→  /** تبدیل مختصات صفحه به مختصات viewBox (با احتساب letterbox وضعیت meet) */
   203→  const viewPoint = useCallback((clientX: number, clientY: number) => {
   204→    const svg = svgRef.current;
   205→    if (!svg) return { vx: 0, vy: 0 };
   206→    const rect = svg.getBoundingClientRect();
   207→    const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   208→    const ox = (rect.width - MAP_CANVAS.w * fit) / 2;
   209→    const oy = (rect.height - MAP_CANVAS.h * fit) / 2;
   210→    return {
   211→      vx: (clientX - rect.left - ox) / (fit * tRef.current.scale) - tRef.current.tx / tRef.current.scale,
   212→      vy: (clientY - rect.top - oy) / (fit * tRef.current.scale) - tRef.current.ty / tRef.current.scale,
   213→    };
   214→  }, []);
   215→
   216→  /** بزرگ‌نمایی حول یک نقطه صفحه (پیش‌فرض: مرکز) — لنگر ثابت می‌ماند */
   217→  const zoomAt = useCallback(
   218→    (factor: number, clientX?: number, clientY?: number) => {
   219→      const t0 = tRef.current;
   220→      const svg = svgRef.current;
   221→      if (!svg) {
   222→        commit({ ...t0, scale: t0.scale * factor });
   223→        return;
   224→      }
   225→      const rect = svg.getBoundingClientRect();
   226→      const cx = clientX ?? rect.left + rect.width / 2;
   227→      const cy = clientY ?? rect.top + rect.height / 2;
   228→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   229→      const ox = (rect.width - MAP_CANVAS.w * fit) / 2;
   230→      const oy = (rect.height - MAP_CANVAS.h * fit) / 2;
   231→      // نقطه لنگر در مختصات viewBox فعلی
   232→      const ax = (cx - rect.left - ox) / fit;
   233→      const ay = (cy - rect.top - oy) / fit;
   234→      const scale = clamp(t0.scale * factor, MIN_SCALE, MAX_SCALE);
   235→      commit({
   236→        scale,
   237→        tx: ax - ((ax - t0.tx) * scale) / t0.scale,
   238→        ty: ay - ((ay - t0.ty) * scale) / t0.scale,
   239→      });
   240→    },
   241→    [commit]
   242→  );
   243→
   244→  /* ---------- چرخ ماوس (غیر-Passive تا preventDefault کار کند) ---------- */
   245→  useEffect(() => {
   246→    const el = svgRef.current;
   247→    if (!el) return;
   248→    const onWheel = (e: WheelEvent) => {
   249→      e.preventDefault();
   250→      zoomAt(e.deltaY < 0 ? 1.15 : 0.87, e.clientX, e.clientY);
   251→    };
   252→    el.addEventListener("wheel", onWheel, { passive: false });
   253→    return () => el.removeEventListener("wheel", onWheel);
   254→  }, [zoomAt]);
   255→
   256→  /* ---------- اشاره‌گرها ---------- */
   257→  const onPointerDown = (e: React.PointerEvent) => {
   258→    try {
   259→      (e.target as Element).setPointerCapture?.(e.pointerId);
   260→    } catch {
   261→      /* برخی محیط‌ها (رویدادهای ساختگی/WebView) خطای InvalidPointerId می‌دهند */
   262→    }
   263→    pointersRef.current.set(e.pointerId, { x: e.clientX, y: e.clientY });
   264→
   265→    if (pointersRef.current.size === 1) {
   266→      // شروع جابجایی تک‌انگشتی
   267→      movedRef.current = false;
   268→      panRef.current = { startX: e.clientX, startY: e.clientY, tx: tRef.current.tx, ty: tRef.current.ty };
   269→      pinchRef.current = null;
   270→    } else if (pointersRef.current.size === 2) {
   271→      // شروع پینچ — فاصله و نقطه میانی دو انگشت
   272→      const [p1, p2] = [...pointersRef.current.values()];
   273→      const t0 = tRef.current;
   274→      const svg = svgRef.current;
   275→      if (!svg) return;
   276→      const rect = svg.getBoundingClientRect();
   277→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   278→      const ox = (rect.width - MAP_CANVAS.w * fit) / 2;
   279→      const oy = (rect.height - MAP_CANVAS.h * fit) / 2;
   280→      const mx = (p1.x + p2.x) / 2;
   281→      const my = (p1.y + p2.y) / 2;
   282→      pinchRef.current = {
   283→        dist0: Math.hypot(p1.x - p2.x, p1.y - p2.y) || 1,
   284→        vx0: (mx - rect.left - ox) / fit,
   285→        vy0: (my - rect.top - oy) / fit,
   286→        s0: t0.scale,
   287→        tx0: t0.tx,
   288→        ty0: t0.ty,
   289→      };
   290→      panRef.current = null;
   291→    }
   292→  };
   293→
   294→  const onPointerMove = (e: React.PointerEvent) => {
   295→    if (!pointersRef.current.has(e.pointerId)) return;
   296→    pointersRef.current.set(e.pointerId, { x: e.clientX, y: e.clientY });
   297→
   298→    if (pinchRef.current && pointersRef.current.size >= 2) {
   299→      // پینچ: مقیاس + حفظ لنگر زیر انگشتان
   300→      const [p1, p2] = [...pointersRef.current.values()];
   301→      const p0 = pinchRef.current;
   302→      const dist = Math.hypot(p1.x - p2.x, p1.y - p2.y) || 1;
   303→      const svg = svgRef.current;
   304→      if (!svg) return;
   305→      const rect = svg.getBoundingClientRect();
   306→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   307→      const ox = (rect.width - MAP_CANVAS.w * fit) / 2;
   308→      const oy = (rect.height - MAP_CANVAS.h * fit) / 2;
   309→      const mx = (p1.x + p2.x) / 2;
   310→      const my = (p1.y + p2.y) / 2;
   311→      const vxm = (mx - rect.left - ox) / fit;
   312→      const vym = (my - rect.top - oy) / fit;
   313→      const scale = clamp((p0.s0 * dist) / p0.dist0, MIN_SCALE, MAX_SCALE);
   314→      // نقطه میانی اولیه باید زیر انگشتان بماند
   315→      const cx = (p0.vx0 - p0.tx0) / p0.s0;
   316→      const cy = (p0.vy0 - p0.ty0) / p0.s0;
   317→      commit({
   318→        scale,
   319→        tx: vxm - scale * cx,
   320→        ty: vym - scale * cy,
   321→      });
   322→      movedRef.current = true;
   323→      return;
   324→    }
   325→
   326→    if (panRef.current && pointersRef.current.size === 1) {
   327→      const p0 = panRef.current;
   328→      const svg = svgRef.current;
   329→      if (!svg) return;
   330→      const rect = svg.getBoundingClientRect();
   331→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   332→      const dx = (e.clientX - p0.startX) / fit;
   333→      const dy = (e.clientY - p0.startY) / fit;
   334→      if (Math.abs(e.clientX - p0.startX) > TAP_SLOP_PX || Math.abs(e.clientY - p0.startY) > TAP_SLOP_PX) {
   335→        movedRef.current = true;
   336→      }
   337→      commit({ scale: tRef.current.scale, tx: p0.tx + dx, ty: p0.ty + dy });
   338→    }
   339→  };
   340→
   341→  const endPointer = (e: React.PointerEvent) => {
   342→    try {
   343→      (e.target as Element).releasePointerCapture?.(e.pointerId);
   344→    } catch {
   345→      /* قبلاً آزاد شده */
   346→    }
   347→    pointersRef.current.delete(e.pointerId);
   348→    if (pointersRef.current.size === 1) {
   349→      // یک انگشت باقی مانده — پینچ تمام شده، ادامه با جابجایی
   350→      pinchRef.current = null;
   351→      const [p] = [...pointersRef.current.values()];
   352→      panRef.current = { startX: p.x, startY: p.y, tx: tRef.current.tx, ty: tRef.current.ty };
   353→    } else if (pointersRef.current.size === 0) {
   354→      panRef.current = null;
   355→      pinchRef.current = null;
   356→    }
   357→  };
   358→
   359→  /* ---------- مرکز کردن دید روی یک ایستگاه ---------- */
   360→  const zoomToStation = useCallback(
   361→    (sx: number, sy: number) => {
   362→      const svg = svgRef.current;
   363→      if (!svg) return;
   364→      const rect = svg.getBoundingClientRect();
   365→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   366→      const scale = 2.4;
   367→      commit({
   368→        scale,
   369→        tx: rect.width / (2 * fit) - scale * sx,
   370→        ty: rect.height / (2 * fit) - scale * sy,
   371→      });
   372→    },
   373→    [commit]
   374→  );
   375→
   376→  /* ---------- موقعیت‌یاب GPS ---------- */
   377→  const locateMe = useCallback(() => {
   378→    if (locate.kind === "busy") return;
   379→    if (!("geolocation" in navigator)) {
   380→      setLocate({ kind: "error", message: "مرورگر از موقعیت‌یابی پشتیبانی نمی‌کند" });
   381→      return;
   382→    }
   383→    setLocate({ kind: "busy" });
   384→    navigator.geolocation.getCurrentPosition(
   385→      (pos) => {
   386→        const { latitude, longitude } = pos.coords;
   387→        const near = nearestStationTo(latitude, longitude);
   388→        if (!near) {
   389→          setLocate({ kind: "far" });
   390→          return;
   391→        }
   392→        const st = MAP_STATION_BY_ID.get(near.id);
   393→        if (!st) {
   394→          setLocate({ kind: "far" });
   395→          return;
   396→        }
   397→        setLocate({ kind: "ok", stationId: near.id, stationName: near.name, distanceM: near.distanceM });
   398→        // بزرگ‌نمایی روی ایستگاه
   399→        zoomToStation(st.x, st.y);
   400→      },
   401→      (err) => {
   402→        setLocate({
   403→          kind: "error",
   404→          message:
   405→            err.code === err.PERMISSION_DENIED
   406→              ? "دسترسی موقعیت مکانی داده نشد"
   407→              : "موقعیت مکانی در دسترس نیست",
   408→        });
   409→      },
   410→      { enableHighAccuracy: true, timeout: 10000, maximumAge: 30000 }
   411→    );
   412→  }, [locate.kind, zoomToStation]);
   413→
   414→  const resetView = useCallback(() => {
   415→    commit({ scale: 1, tx: 0, ty: 0 });
   416→    setLocate({ kind: "idle" });
   417→  }, [commit]);
   418→
   419→  const highlighted = useMemo(() => {
   420→    const set = new Set(highlightStations ?? []);
   421→    if (locate.kind === "ok") set.add(locate.stationId);
   422→    return set;
   423→  }, [highlightStations, locate]);
   424→
   425→  /* ============================================================
   426→     موتور برچسب‌ها — مقیاس «فضای صفحه»:
   427→     برخلاف گذشته که متن‌ها داخل گروه بزرگ‌نمایی‌شونده بودند (و در
   428→     بزرگ‌نمایی ۱ روی هم ریخته دیده می‌شدند)، اینجا هر برچسب:
   429→     ۱. با اندازه ثابت روی صفحه رندر می‌شود (مستقل از زوم)
   430→     ۲. زنده از مختصات ایستگاه پیروی می‌کند
   431→     ۳. فقط اگر با برچسب مهم‌تری همپوشانی نداشته باشد نمایش
   432→        داده می‌شود — با نزدیک‌تر شدن (زوم بیشتر) برچسب‌های
   433→        بیشتری جا می‌گیرند و ظاهر می‌شوند
   434→     ============================================================ */
   435→  const fit = box.w > 0 && box.h > 0 ? Math.min(box.w / MAP_CANVAS.w, box.h / MAP_CANVAS.h) : 0;
   436→
   437→  const labelLayout = useMemo(() => {
   438→    const empty: { labels: LabelItem[]; badges: BadgeItem[] } = { labels: [], badges: [] };
   439→    if (fit <= 0) return empty;
   440→    const toVb = (px: number) => px / fit;
   441→
   442→    const boxes: Box[] = [];
   443→    const overlaps = (b: Box) =>
   444→      boxes.some((o) => b.x0 < o.x1 && b.x1 > o.x0 && b.y0 < o.y1 && b.y1 > o.y0);
   445→
   446→    /* ---------- ناحیه کارت‌های HTML روی نقشه (راهنما + کنترل‌ها) —
   447→       برچسب و نشان نباید زیر این کارت‌ها پنهان شوند ---------- */
   448→    const ox = (box.w - MAP_CANVAS.w * fit) / 2;
   449→    const oy = (box.h - MAP_CANVAS.h * fit) / 2;
   450→    const htmlZone = (left: number, top: number, w: number, h: number): Box | null => {
   451→      const x0 = Math.max(0, (left - ox) / fit);
   452→      const y0 = Math.max(0, (top - oy) / fit);
   453→      const x1 = Math.min(MAP_CANVAS.w, (left + w - ox) / fit);
   454→      const y1 = Math.min(MAP_CANVAS.h, (top + h - oy) / fit);
   455→      if (x1 <= x0 || y1 <= y0) return null;
   456→      return { x0, y0, x1, y1 };
   457→    };
   458→    const legendZone = htmlZone(10, 10, 158, legendOpen ? 185 : 40);
   459→    if (legendZone) boxes.push(legendZone);
   460→    // ستون دکمه‌های شناور پایین-چپ (۴ دکمه ۳۶px با فاصله)
   461→    const controlsZone = htmlZone(8, box.h - 196, 44, 188);
   462→    if (controlsZone) boxes.push(controlsZone);
   463→
   464→    /* ---------- نشان شماره خطوط (اندازه ثابت، پشته‌ای در ایستگاه مشترک) ---------- */
   465→    const badges: BadgeItem[] = [];
   466→    const badgeN = new Map<string, number>();
   467→    for (const ml of MAP_LINES) {
   468→      for (const id of [ml.stations[0], ml.stations[ml.stations.length - 1]]) {
   469→        const s = MAP_STATION_BY_ID.get(id);
   470→        if (!s) continue;
   471→        const n = badgeN.get(id) ?? 0;
   472→        badgeN.set(id, n + 1);
   473→        const r = toVb(8.5);
   474→        const side = s.x > MAP_CANVAS.w / 2 ? -1 : 1;
   475→        const bx = s.x * t.scale + t.tx + side * toVb(14);
   476→        const by = s.y * t.scale + t.ty - toVb(15) - n * toVb(19);
   477→        if (bx < -r || bx > MAP_CANVAS.w + r || by < -r || by > MAP_CANVAS.h + r) continue;
   478→        const bBox: Box = {
   479→          x0: bx - r - toVb(1),
   480→          y0: by - r - toVb(1),
   481→          x1: bx + r + toVb(1),
   482→          y1: by + r + toVb(1),
   483→        };
   484→        // نشان زیر کارت‌های HTML پنهان نمی‌شود
   485→        if (overlaps(bBox)) continue;
   486→        badges.push({
   487→          key: `${ml.lineId}-${id}-${n}`,
   488→          x: bx,
   489→          y: by,
   490→          r,
   491→          stroke: toVb(1.5),
   492→          font: toVb(10),
   493→          num: toFa(TEHRAN_LINES.find((l) => l.id === ml.lineId)?.number ?? 0),
   494→          color: ml.color,
   495→          ink: ml.lineId === "l4" ? "#1a1a1a" : "#ffffff",
   496→        });
   497→        boxes.push(bBox);
   498→      }
   499→    }
   500→
   501→    /* ---------- ناحیه حفاظتی نشان بزرگ ایستگاه‌های تبادلی ---------- */
   502→    const interExcl: { id: string; box: Box }[] = [];
   503→    for (const s of MAP_STATIONS) {
   504→      if (!INTERCHANGE_IDS.has(s.id)) continue;
   505→      const half = 6.4 * t.scale + 3 / fit;
   506→      const X = s.x * t.scale + t.tx;
   507→      const Y = s.y * t.scale + t.ty;
   508→      interExcl.push({
   509→        id: s.id,
   510→        box: { x0: X - half, y0: Y - half, x1: X + half, y1: Y + half },
   511→      });
   512→    }
   513→    const hitsInterchange = (b: Box, selfId: string) =>
   514→      interExcl.some(
   515→        (e) => e.id !== selfId && b.x0 < e.box.x1 && b.x1 > e.box.x0 && b.y0 < e.box.y1 && b.y1 > e.box.y0
   516→      );
   517→
   518→    /* ---------- برچسب‌ها به ترتیب اولویت (تبادلی ← پرچم‌دار ← عادی) ---------- */
   519→    const cands = MAP_STATIONS.map((s: MapStation) => {
   520→      const isInter = INTERCHANGE_IDS.has(s.id);
   521→      const prio = highlighted.has(s.id)
   522→        ? 1000
   523→        : isInter && s.label
   524→          ? 500
   525→          : isInter
   526→            ? 400
   527→            : s.label
   528→              ? 300
   529→              : 200;
   530→      return { s, prio, major: isInter || s.label };
   531→    }).sort((a, b) => b.prio - a.prio);
   532→
   533→    const labels: LabelItem[] = [];
   534→    for (const { s, prio, major } of cands) {
   535→      void prio;
   536→      const X = s.x * t.scale + t.tx;
   537→      const Y = s.y * t.scale + t.ty;
   538→      const margin = toVb(26);
   539→      if (X < -margin || X > MAP_CANVAS.w + margin || Y < -margin || Y > MAP_CANVAS.h + margin) continue;
   540→
   541→      const fontPx = major ? 11.5 : 10.5;
   542→      const weight = major ? 800 : 600;
   543→      const wVb = toVb(textWidthPx(s.name, weight, fontPx, fontsTick) + 7);
   544→      const hVb = toVb(fontPx * 1.1 + 5);
   545→      const gapVb = toVb(10);
   546→      // جهت متن فارسی rtl است: anchor «end» یعنی لبه چپ متن در x
   547→      const prefRight = s.x <= MAP_CANVAS.w / 2;
   548→      const sides: ("right" | "left")[] = prefRight ? ["right", "left"] : ["left", "right"];
   549→
   550→      for (const side of sides) {
   551→        const bx0 = side === "right" ? X + gapVb : X - gapVb - wVb;
   552→        const by0 = Y - hVb / 2;
   553→        const b: Box = { x0: bx0, y0: by0, x1: bx0 + wVb, y1: by0 + hVb };
   554→        if (b.x0 < 1 || b.x1 > MAP_CANVAS.w - 1 || b.y0 < 1 || b.y1 > MAP_CANVAS.h - 1) continue;
   555→        if (overlaps(b) || hitsInterchange(b, s.id)) continue;
   556→        boxes.push(b);
   557→        labels.push({
   558→          id: s.id,
   559→          name: s.name,
   560→          x: side === "right" ? X + gapVb : X - gapVb,
   561→          y: Y + toVb(fontPx * 0.35),
   562→          anchor: side === "right" ? "end" : "start",
   563→          fontSize: toVb(fontPx),
   564→          weight,
   565→          halo: toVb(major ? 3.4 : 2.8),
   566→        });
   567→        break;
   568→      }
   569→    }
   570→
   571→    return { labels, badges };
   572→  }, [t, fit, fontsTick, highlighted, legendOpen, box.w, box.h]);
   573→

   575→    <div className={`relative overflow-hidden rounded-2xl ${className}`}>
   576→      {/* لایه SVG نقشه */}
   577→      <svg
   578→        ref={svgRef}
   579→        viewBox={`0 0 ${MAP_CANVAS.w} ${MAP_CANVAS.h}`}
   580→        className="w-full h-full map-viewport select-none"
   581→        style={{ touchAction: "none" }}
   582→        onPointerDown={onPointerDown}
   583→        onPointerMove={onPointerMove}
   584→        onPointerUp={endPointer}
   585→        onPointerCancel={endPointer}
   586→        role="img"
   587→        aria-label="نقشه شماتیک شبکه متروی تهران — با دو انگشت بزرگ‌نمایی کنید"
   588→      >
   589→        <defs>
   590→          <pattern id="map-grid" width="28" height="28" patternUnits="userSpaceOnUse">
   591→            <path d="M 28 0 L 0 0 0 28" fill="none" stroke="var(--mq-map-grid)" strokeWidth="0.7" />
   592→          </pattern>
   593→        </defs>
   594→        <rect width={MAP_CANVAS.w} height={MAP_CANVAS.h} fill="var(--mq-map-bg)" />
   595→        <rect width={MAP_CANVAS.w} height={MAP_CANVAS.h} fill="url(#map-grid)" />
   596→
   597→        <g transform={`translate(${t.tx} ${t.ty}) scale(${t.scale})`}>
   598→          {/* خطوط — هر خط با زیرلایه روشن تا در تقاطع‌ها از هم جدا دیده شوند */}
   599→          {MAP_LINES.map((ml) => (
   600→            <g key={ml.lineId}>
   601→              <polyline
   602→                points={linePoints(ml.stations)}
   603→                fill="none"
   604→                stroke="var(--mq-map-bg)"
   605→                strokeWidth={10.5}
   606→                strokeLinecap="round"
   607→                strokeLinejoin="round"
   608→              />
   609→              <polyline
   610→                points={linePoints(ml.stations)}
   611→                fill="none"
   612→                stroke={ml.color}
   613→                strokeWidth={7}
   614→                strokeLinecap="round"
   615→                strokeLinejoin="round"
   616→              />
   617→            </g>
   618→          ))}
   619→
   620→          {/* ایستگاه‌ها — فقط نشان‌ها؛ برچسب‌ها در لایه جداگانه پایین رندر می‌شوند */}
   621→          {MAP_STATIONS.map((s) => {
   622→            const isInter = INTERCHANGE_IDS.has(s.id);
   623→            const isHi = highlighted.has(s.id);
   624→            const r = isInter ? 6.4 : 4.6;
   625→            return (
   626→              <g
   627→                key={s.id}
   628→                onClick={(e) => {
   629→                  e.stopPropagation();
   630→                  if (movedRef.current) return; // بعد از جابجایی نقشه، تپ نیست
   631→                  onStationSelect?.(s.id);
   632→                }}
   633→                className="cursor-pointer"
   634→              >
   635→                <circle cx={s.x} cy={s.y} r={15} fill="transparent" />
   636→                {isInter && <circle cx={s.x} cy={s.y} r={r + 3} fill="var(--mq-card)" opacity={0.95} />}
   637→                {isHi && <circle cx={s.x} cy={s.y} r={13} fill="#0d58a6" opacity={0.18} className="map-pulse" />}
   638→                <circle
   639→                  cx={s.x}
   640→                  cy={s.y}
   641→                  r={isHi ? 6.5 : r}
   642→                  fill={isHi ? "#0d58a6" : "var(--mq-card)"}
   643→                  stroke={isHi ? "#0d58a6" : "var(--mq-ink)"}
   644→                  strokeWidth={isHi ? 2.4 : 1.6}
   645→                />
   646→              </g>
   647→            );
   648→          })}
   649→        </g>
   650→
   651→        {/* ============================================================
   652→            لایه برچسب‌ها (خارج از گروه بزرگ‌نمایی — اندازه ثابت روی صفحه)
   653→            ============================================================ */}
   654→        {fit > 0 && (
   655→          <g style={{ pointerEvents: "none" }}>
   656→            {/* نشان شماره خطوط */}
   657→            {labelLayout.badges.map((b) => (
   658→              <g key={b.key}>
   659→                <circle
   660→                  cx={b.x}
   661→                  cy={b.y}
   662→                  r={b.r}
   663→                  fill={b.color}
   664→                  stroke="var(--mq-card)"
   665→                  strokeWidth={b.stroke}
   666→                />
   667→                <text
   668→                  x={b.x}
   669→                  y={b.y + b.r * 0.38}
   670→                  textAnchor="middle"
   671→                  fontSize={b.font}
   672→                  fontWeight={800}
   673→                  fill={b.ink}
   674→                  style={{ fontFamily: "Vazirmatn, sans-serif" }}
   675→                >
   676→                  {b.num}
   677→                </text>
   678→              </g>
   679→            ))}
   680→
   681→            {/* نام ایستگاه‌ها — با هاله روشن و کنترل همپوشانی */}
   682→            {labelLayout.labels.map((L) => (
   683→              <text
   684→                key={L.id}
   685→                x={L.x}
   686→                y={L.y}
   687→                textAnchor={L.anchor}
   688→                fontSize={L.fontSize}
   689→                fontWeight={L.weight}
   690→                fill="var(--mq-ink)"
   691→                stroke="var(--mq-card)"
   692→                strokeWidth={L.halo}
   693→                paintOrder="stroke"
   694→                strokeLinejoin="round"
   695→                style={{
   696→                  fontFamily: "Vazirmatn, sans-serif",
   697→                  direction: "rtl",
   698→                  pointerEvents: "auto",
   699→                  cursor: "pointer",
   700→                  userSelect: "none",
   701→                }}
   702→                onClick={(e) => {
   703→                  e.stopPropagation();
   704→                  if (movedRef.current) return; // بعد از جابجایی نقشه، تپ نیست
   705→                  onStationSelect?.(L.id);
   706→                }}
   707→              >
   708→                {L.name}
   709→              </text>
   710→            ))}
   711→          </g>
   712→        )}
   713→      </svg>
   714→
   715→      {/* ================= کارت راهنمای خطوط (جای دکمه در هر دو حالت ثابت است) ================= */}
   716→      <div className="absolute top-2.5 left-2.5 w-[9.5rem] rounded-xl bg-metro-card/95 backdrop-blur border border-metro-outline-var shadow-md overflow-hidden">
   717→        <button
   718→          onClick={() => setLegendOpen((o) => !o)}
   719→          aria-expanded={legendOpen}
   720→          aria-label={legendOpen ? "بستن راهنمای خطوط" : "نمایش راهنمای خطوط"}
   721→          className="w-full h-9 px-2.5 flex items-center gap-1.5 text-[10px] font-bold text-metro-ink active:bg-surface-container transition-colors"
   722→        >
   723→          <MI name="layers" size={15} className="text-metro-primary-c shrink-0" />
   724→          راهنمای خطوط
   725→          <MI
   726→            name="expand_more"
   727→            size={14}
   728→            className={`text-outline mr-auto transition-transform ${legendOpen ? "rotate-180" : ""}`}
   729→          />
   730→        </button>
   731→        {legendOpen && (
   732→          <div className="fade-in px-2.5 pb-2 grid grid-cols-1 gap-1 border-t border-metro-outline-var/50 pt-1.5">
   733→            {TEHRAN_LINES.map((l) => (
   734→              <div key={l.id} className="flex items-center gap-1.5">
   735→                <span className="w-2.5 h-2.5 rounded-full shrink-0" style={{ backgroundColor: l.color }} />
   736→                <span className="text-[9px] font-semibold text-metro-ink truncate">
   737→                  {l.name} — {l.terminals[0]} / {l.terminals[1]}
   738→                </span>
   739→              </div>
   740→            ))}
   741→          </div>
   742→        )}
   743→      </div>
   744→
   745→      {/* ================= نتیجه موقعیت‌یاب ================= */}
   746→      {locate.kind === "busy" && (
   747→        <div className="absolute top-2.5 right-2.5 bg-metro-card/95 backdrop-blur rounded-full shadow-md border border-metro-outline-var px-3 py-1.5 flex items-center gap-1.5">
   748→          <span className="w-3.5 h-3.5 border-2 border-metro-primary-c/30 border-t-metro-primary-c rounded-full animate-spin" />
   749→          <span className="text-[10px] font-bold text-metro-ink">در حال یافتن موقعیت...</span>
   750→        </div>
   751→      )}
   752→      {locate.kind === "ok" && (
   753→        <div className="fade-in absolute top-2.5 right-2.5 max-w-[70%] bg-metro-card/95 backdrop-blur rounded-xl shadow-md border border-metro-outline-var px-3 py-1.5 flex items-center gap-2">
   754→          <MI name="near_me" size={15} className="text-metro-primary-c shrink-0" />
   755→          <span className="text-[10px] font-bold text-metro-ink truncate">
   756→            نزدیک‌ترین ایستگاه: {locate.stationName} (حدود {toFa(Math.round(locate.distanceM / 50) * 50)} متر)
   757→          </span>
   758→          <button
   759→            onClick={() => setLocate({ kind: "idle" })}
   760→            aria-label="بستن"
   761→            className="text-outline shrink-0 active:scale-90 transition-transform"
   762→          >
   763→            <MI name="close" size={13} />
   764→          </button>
   765→        </div>
   766→      )}
   767→      {(locate.kind === "far" || locate.kind === "error") && (
   768→        <div className="fade-in absolute top-2.5 right-2.5 max-w-[70%] bg-metro-card/95 backdrop-blur rounded-xl shadow-md border border-error/40 px-3 py-1.5 flex items-center gap-2">
   769→          <MI name="location_off" size={15} className="text-error shrink-0" />
   770→          <span className="text-[10px] font-bold text-metro-ink">
   771→            {locate.kind === "far" ? "در محدوده ایستگاه‌های مترو نیستید" : locate.message}
   772→          </span>
   773→          <button
   774→            onClick={() => setLocate({ kind: "idle" })}
   775→            aria-label="بستن"
   776→            className="text-outline shrink-0 active:scale-90 transition-transform"
   777→          >
   778→            <MI name="close" size={13} />
   779→          </button>
   780→        </div>
   781→      )}
   782→
   783→      {/* ================= کنترل‌های شناور (بزرگ‌نمایی + موقعیت) ================= */}
   784→      <div className="absolute bottom-2.5 left-2.5 flex flex-col gap-1.5">
   785→        <button
   786→          onClick={locateMe}
   787→          aria-label="نزدیک‌ترین ایستگاه به من"
   788→          className="w-9 h-9 rounded-xl bg-metro-card shadow border border-metro-outline-var flex items-center justify-center text-metro-primary-c active:scale-95 transition-all"
   789→        >
   790→          <MI name="my_location" size={18} />
   791→        </button>
   792→        <button
   793→          onClick={() => zoomAt(1.25)}
   794→          aria-label="بزرگ‌نمایی"
   795→          className="w-9 h-9 rounded-xl bg-metro-card shadow border border-metro-outline-var flex items-center justify-center text-metro-primary-c active:scale-95 transition-all"
   796→        >
   797→          <MI name="add" size={18} />
   798→        </button>
   799→        <button
   800→          onClick={() => zoomAt(0.8)}
   801→          aria-label="کوچک‌نمایی"

   803→        >
   804→          <MI name="remove" size={18} />
   805→        </button>

   807→          onClick={resetView}
   808→          aria-label="بازگشت به نمای کامل"
   809→          className="w-9 h-9 rounded-xl bg-metro-card shadow border border-metro-outline-var flex items-center justify-center text-metro-primary-c active:scale-95 transition-all"
   810→        >
   811→          <MI name="center_focus_strong" size={18} />
   812→        </button>
   813→      </div>
   814→    </div>
   815→  );
