     4→import {
     5→  MAP_LINES,

    15→/* ============================================================
    16→   نقشه تعاملی مترو — موتور ژست چندلمسی (پینچ/جابجایی بدون خطا)
    17→   + موتور برچسب «فضای صفحه»: متن همیشه با اندازه ثابت و خوانا،
    18→     موقعیت‌یابی زنده روی نقشه و حذف هوشمند همپوشانی
    19→   + راهنمای خطوط + موقعیت‌یاب GPS
    20→   ============================================================ */

    22→interface MetroMapProps {
    23→  onStationSelect?: (stationId: string) => void;

    27→
    28→interface Transform {
    29→  scale: number;
    30→  tx: number;
    31→  ty: number;
    32→}
    33→
    34→const MIN_SCALE = 0.75;
    35→const MAX_SCALE = 4;
    36→const TAP_SLOP_PX = 7; // حرکت بیشتر از این = جابجایی، نه لمس ایستگاه
    37→
    38→const TEHRAN_LINES = LINES.filter((l) => l.cityId === "tehran");
    39→
    40→/* ایستگاه‌های تبادلی (اشتراک دو خط یا بیشتر) — ثابت و مستقل از رندر */
    41→const INTERCHANGE_IDS = new Set(
    42→  MAP_LINES.flatMap((ml) => ml.stations).filter(
    43→    (id) => MAP_LINES.filter((ml) => ml.stations.includes(id)).length > 1
    44→  )
    45→);

    47→/** نقاط هر خط به‌صورت رشته points برای polyline */
    48→function linePoints(stations: string[]): string {
    49→  return stations
    50→    .map((id) => {
    51→      const s = MAP_STATION_BY_ID.get(id);
    52→      return s ? `${s.x},${s.y}` : "";
    53→    })
    54→    .filter(Boolean)
    55→    .join(" ");
    56→}
    57→
    58→/* ---------------- اندازه‌گیری واقعی عرض متن (برای جلوگیری از همپوشانی) ---------------- */
    59→let measureCtx: CanvasRenderingContext2D | null = null;
    60→const widthCache = new Map<string, number>();
    61→
    62→function textWidthPx(text: string, weight: number, px: number, epoch: number): number {
    63→  const key = `${epoch}|${weight}|${px}|${text}`;
    64→  const hit = widthCache.get(key);
    65→  if (hit !== undefined) return hit;
    66→  let w: number;
    67→  try {
    68→    if (!measureCtx) {
    69→      measureCtx = document.createElement("canvas").getContext("2d");
    70→    }
    71→    if (measureCtx) {
    72→      measureCtx.font = `${weight} ${px}px Vazirmatn, sans-serif`;
    73→      w = measureCtx.measureText(text).width;
    74→    } else {
    75→      w = text.length * px * 0.6;
    76→    }
    77→  } catch {
    78→    w = text.length * px * 0.6;
    79→  }
    80→  widthCache.set(key, w);
    81→  return w;
    82→}

    84→/* ---------------- انواع لایه برچسب ---------------- */
    85→interface LabelItem {
    86→  id: string;
    87→  name: string;
    88→  x: number;
    89→  y: number;
    90→  anchor: "start" | "end";
    91→  fontSize: number;
    92→  weight: number;
    93→  halo: number;
    94→}
    95→
    96→interface BadgeItem {
    97→  key: string;
    98→  x: number;
    99→  y: number;
   100→  r: number;
   101→  stroke: number;
   102→  font: number;
   103→  num: string;
   104→  color: string;
   105→  ink: string;
   106→}
   107→
   108→interface Box {
   109→  x0: number;
   110→  y0: number;
   111→  x1: number;
   112→  y1: number;
   113→}
   114→
   115→type LocateState =
   116→  | { kind: "idle" }
   117→  | { kind: "busy" }
   118→  | { kind: "ok"; stationId: string; stationName: string; distanceM: number }
   119→  | { kind: "far" }
   120→  | { kind: "error"; message: string };
   121→
   122→export default function MetroMap({
   123→  onStationSelect,
   124→  highlightStations,
   125→  className = "",
   126→}: MetroMapProps) {
   127→  const [t, setT] = useState<Transform>({ scale: 1, tx: 0, ty: 0 });
   128→  const [legendOpen, setLegendOpen] = useState(false);
   129→  const [locate, setLocate] = useState<LocateState>({ kind: "idle" });
   130→  const [box, setBox] = useState({ w: 0, h: 0 });
   131→  const [fontsTick, setFontsTick] = useState(0);
   132→
   133→  /* ---------- آینه همزمان ترنسفورم — جلوگیری از رقابت با React ---------- */
   134→  const tRef = useRef<Transform>({ scale: 1, tx: 0, ty: 0 });
   135→  const svgRef = useRef<SVGSVGElement>(null);
   136→
   137→  /* اشاره‌گرهای فعال — پایه ژست چندلمسی */
   138→  const pointersRef = useRef<Map<number, { x: number; y: number }>>(new Map());
   139→  const panRef = useRef<{ startX: number; startY: number; tx: number; ty: number } | null>(null);
   140→  const pinchRef = useRef<{
   141→    dist0: number;
   142→    vx0: number;
   143→    vy0: number;
   144→    s0: number;
   145→    tx0: number;
   146→    ty0: number;
   147→  } | null>(null);
   148→  const movedRef = useRef(false); // بعد از جابجایی، تپ ایستگاه نادیده گرفته می‌شود
   149→
   150→  /* ---------- اندازه واقعی بوم برای مقیاس ثابت متن ---------- */
   151→  useEffect(() => {
   152→    const el = svgRef.current;
   153→    if (!el) return;
   154→    const update = () => {
   155→      const r = el.getBoundingClientRect();
   156→      if (r.width < 2 || r.height < 2) return;
   157→      setBox((b) =>
   158→        Math.abs(b.w - r.width) > 0.5 || Math.abs(b.h - r.height) > 0.5
   159→          ? { w: r.width, h: r.height }
   160→          : b
   161→      );
   162→    };
   163→    update();
   164→    const ro = new ResizeObserver(update);
   165→    ro.observe(el);
   166→    return () => ro.disconnect();
   167→  }, []);
   168→
   169→  /* ---------- بعد از لود کامل فونت، عرض متن‌ها دوباره اندازه‌گیری می‌شود ---------- */
   170→  useEffect(() => {
   171→    let alive = true;
   172→    try {
   173→      document.fonts?.ready
   174→        .then(() => {
   175→          if (!alive) return;
   176→          widthCache.clear();
   177→          setFontsTick((v) => v + 1);
   178→        })
   179→        .catch(() => {});
   180→    } catch {
   181→      /* مرورگرهای قدیمی */
   182→    }
   183→    return () => {
   184→      alive = false;
   185→    };
   186→  }, []);
   187→
   188→  const clamp = (v: number, min: number, max: number) => Math.min(max, Math.max(min, v));
   189→
   190→  /** ثبت ترنسفورم جدید — همیشه از طریق این تابع */
   191→  const commit = useCallback((next: Transform) => {
   192→    const scale = clamp(next.scale, MIN_SCALE, MAX_SCALE);
   193→    const half = MAP_CANVAS.w / 2;
   194→    const halfH = MAP_CANVAS.h / 2;
   195→    const tx = clamp(next.tx, half - scale * half - 60, -(half - scale * half) + 60);
   196→    const ty = clamp(next.ty, halfH - scale * halfH - 60, -(halfH - scale * halfH) + 60);
   197→    const fixed = { scale, tx, ty };
   198→    tRef.current = fixed;
   199→    setT(fixed);
   200→  }, []);
   201→
   202→  /** تبدیل مختصات صفحه به مختصات viewBox (با احتساب letterbox وضعیت meet) */
   203→  const viewPoint = useCallback((clientX: number, clientY: number) => {
   204→    const svg = svgRef.current;
   205→    if (!svg) return { vx: 0, vy: 0 };
   206→    const rect = svg.getBoundingClientRect();
   207→    const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   208→    const ox = (rect.width - MAP_CANVAS.w * fit) / 2;
   209→    const oy = (rect.height - MAP_CANVAS.h * fit) / 2;
   210→    return {
   211→      vx: (clientX - rect.left - ox) / (fit * tRef.current.scale) - tRef.current.tx / tRef.current.scale,
   212→      vy: (clientY - rect.top - oy) / (fit * tRef.current.scale) - tRef.current.ty / tRef.current.scale,
   213→    };
   214→  }, []);
   215→
   216→  /** بزرگ‌نمایی حول یک نقطه صفحه (پیش‌فرض: مرکز) — لنگر ثابت می‌ماند */
   217→  const zoomAt = useCallback(
   218→    (factor: number, clientX?: number, clientY?: number) => {
   219→      const t0 = tRef.current;
   220→      const svg = svgRef.current;
   221→      if (!svg) {
   222→        commit({ ...t0, scale: t0.scale * factor });
   223→        return;
   224→      }
   225→      const rect = svg.getBoundingClientRect();
   226→      const cx = clientX ?? rect.left + rect.width / 2;
   227→      const cy = clientY ?? rect.top + rect.height / 2;
   228→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   229→      const ox = (rect.width - MAP_CANVAS.w * fit) / 2;
   230→      const oy = (rect.height - MAP_CANVAS.h * fit) / 2;
   231→      // نقطه لنگر در مختصات viewBox فعلی
   232→      const ax = (cx - rect.left - ox) / fit;
   233→      const ay = (cy - rect.top - oy) / fit;
   234→      const scale = clamp(t0.scale * factor, MIN_SCALE, MAX_SCALE);
   235→      commit({
   236→        scale,
   237→        tx: ax - ((ax - t0.tx) * scale) / t0.scale,
   238→        ty: ay - ((ay - t0.ty) * scale) / t0.scale,
   239→      });
   240→    },
   241→    [commit]
   242→  );
   243→
   244→  /* ---------- چرخ ماوس (غیر-Passive تا preventDefault کار کند) ---------- */
   245→  useEffect(() => {
   246→    const el = svgRef.current;
   247→    if (!el) return;
   248→    const onWheel = (e: WheelEvent) => {
   249→      e.preventDefault();
   250→      zoomAt(e.deltaY < 0 ? 1.15 : 0.87, e.clientX, e.clientY);
   251→    };
   252→    el.addEventListener("wheel", onWheel, { passive: false });
   253→    return () => el.removeEventListener("wheel", onWheel);
   254→  }, [zoomAt]);
   255→
   256→  /* ---------- اشاره‌گرها ---------- */
   257→  const onPointerDown = (e: React.PointerEvent) => {
   258→    try {
   259→      (e.target as Element).setPointerCapture?.(e.pointerId);
   260→    } catch {
   261→      /* برخی محیط‌ها (رویدادهای ساختگی/WebView) خطای InvalidPointerId می‌دهند */
   262→    }
   263→    pointersRef.current.set(e.pointerId, { x: e.clientX, y: e.clientY });
   264→
   265→    if (pointersRef.current.size === 1) {
   266→      // شروع جابجایی تک‌انگشتی
   267→      movedRef.current = false;
   268→      panRef.current = { startX: e.clientX, startY: e.clientY, tx: tRef.current.tx, ty: tRef.current.ty };
   269→      pinchRef.current = null;
   270→    } else if (pointersRef.current.size === 2) {
   271→      // شروع پینچ — فاصله و نقطه میانی دو انگشت
   272→      const [p1, p2] = [...pointersRef.current.values()];
   273→      const t0 = tRef.current;
   274→      const svg = svgRef.current;
   275→      if (!svg) return;
   276→      const rect = svg.getBoundingClientRect();
   277→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   278→      const ox = (rect.width - MAP_CANVAS.w * fit) / 2;
   279→      const oy = (rect.height - MAP_CANVAS.h * fit) / 2;
   280→      const mx = (p1.x + p2.x) / 2;
   281→      const my = (p1.y + p2.y) / 2;
   282→      pinchRef.current = {
   283→        dist0: Math.hypot(p1.x - p2.x, p1.y - p2.y) || 1,
   284→        vx0: (mx - rect.left - ox) / fit,
   285→        vy0: (my - rect.top - oy) / fit,
   286→        s0: t0.scale,
   287→        tx0: t0.tx,
   288→        ty0: t0.ty,
   289→      };
   290→      panRef.current = null;
   291→    }
   292→  };
   293→
   294→  const onPointerMove = (e: React.PointerEvent) => {
   295→    if (!pointersRef.current.has(e.pointerId)) return;
   296→    pointersRef.current.set(e.pointerId, { x: e.clientX, y: e.clientY });
   297→
   298→    if (pinchRef.current && pointersRef.current.size >= 2) {
   299→      // پینچ: مقیاس + حفظ لنگر زیر انگشتان
   300→      const [p1, p2] = [...pointersRef.current.values()];
   301→      const p0 = pinchRef.current;
   302→      const dist = Math.hypot(p1.x - p2.x, p1.y - p2.y) || 1;
   303→      const svg = svgRef.current;
   304→      if (!svg) return;
   305→      const rect = svg.getBoundingClientRect();
   306→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   307→      const ox = (rect.width - MAP_CANVAS.w * fit) / 2;
   308→      const oy = (rect.height - MAP_CANVAS.h * fit) / 2;
   309→      const mx = (p1.x + p2.x) / 2;
   310→      const my = (p1.y + p2.y) / 2;
   311→      const vxm = (mx - rect.left - ox) / fit;
   312→      const vym = (my - rect.top - oy) / fit;
   313→      const scale = clamp((p0.s0 * dist) / p0.dist0, MIN_SCALE, MAX_SCALE);
   314→      // نقطه میانی اولیه باید زیر انگشتان بماند
   315→      const cx = (p0.vx0 - p0.tx0) / p0.s0;
   316→      const cy = (p0.vy0 - p0.ty0) / p0.s0;
   317→      commit({
   318→        scale,
   319→        tx: vxm - scale * cx,
   320→        ty: vym - scale * cy,
   321→      });
   322→      movedRef.current = true;
   323→      return;
   324→    }
   325→
   326→    if (panRef.current && pointersRef.current.size === 1) {
   327→      const p0 = panRef.current;
   328→      const svg = svgRef.current;
   329→      if (!svg) return;
   330→      const rect = svg.getBoundingClientRect();
   331→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   332→      const dx = (e.clientX - p0.startX) / fit;
   333→      const dy = (e.clientY - p0.startY) / fit;
   334→      if (Math.abs(e.clientX - p0.startX) > TAP_SLOP_PX || Math.abs(e.clientY - p0.startY) > TAP_SLOP_PX) {
   335→        movedRef.current = true;
   336→      }
   337→      commit({ scale: tRef.current.scale, tx: p0.tx + dx, ty: p0.ty + dy });
   338→    }
   339→  };
   340→
   341→  const endPointer = (e: React.PointerEvent) => {
   342→    try {
   343→      (e.target as Element).releasePointerCapture?.(e.pointerId);
   344→    } catch {
   345→      /* قبلاً آزاد شده */
   346→    }
   347→    pointersRef.current.delete(e.pointerId);
   348→    if (pointersRef.current.size === 1) {
   349→      // یک انگشت باقی مانده — پینچ تمام شده، ادامه با جابجایی
   350→      pinchRef.current = null;
   351→      const [p] = [...pointersRef.current.values()];
   352→      panRef.current = { startX: p.x, startY: p.y, tx: tRef.current.tx, ty: tRef.current.ty };
   353→    } else if (pointersRef.current.size === 0) {
   354→      panRef.current = null;
   355→      pinchRef.current = null;
   356→    }
   357→  };
   358→
   359→  /* ---------- مرکز کردن دید روی یک ایستگاه ---------- */
   360→  const zoomToStation = useCallback(
   361→    (sx: number, sy: number) => {
   362→      const svg = svgRef.current;
   363→      if (!svg) return;
   364→      const rect = svg.getBoundingClientRect();
   365→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   366→      const scale = 2.4;
   367→      commit({
   368→        scale,
   369→        tx: rect.width / (2 * fit) - scale * sx,
   370→        ty: rect.height / (2 * fit) - scale * sy,
   371→      });
   372→    },
   373→    [commit]
   374→  );
   375→
   376→  /* ---------- موقعیت‌یاب GPS ---------- */
   377→  const locateMe = useCallback(() => {
   378→    if (locate.kind === "busy") return;
   379→    /* احترام به کلید GPS در تنظیمات */
   380→    try {
   381→      if (localStorage.getItem("mj-gps-enabled") === "false") {
   382→        setLocate({ kind: "error", message: "دسترسی موقعیت مکانی در تنظیمات غیرفعال است" });
   383→        return;
   384→      }
   385→    } catch {
   386→      /* حافظه در دسترس نیست — ادامه */
   387→    }
   388→    if (!("geolocation" in navigator)) {
   389→      setLocate({ kind: "error", message: "مرورگر از موقعیت‌یابی پشتیبانی نمی‌کند" });
   390→      return;
   391→    }
   392→    setLocate({ kind: "busy" });
   393→    navigator.geolocation.getCurrentPosition(
   394→      (pos) => {
   395→        const { latitude, longitude } = pos.coords;
   396→        const near = nearestStationTo(latitude, longitude);
   397→        if (!near) {
   398→          setLocate({ kind: "far" });
   399→          return;
   400→        }
   401→        const st = MAP_STATION_BY_ID.get(near.id);
   402→        if (!st) {
   403→          setLocate({ kind: "far" });
   404→          return;
   405→        }
   406→        setLocate({ kind: "ok", stationId: near.id, stationName: near.name, distanceM: near.distanceM });
   407→        // بزرگ‌نمایی روی ایستگاه
   408→        zoomToStation(st.x, st.y);
   409→      },
   410→      (err) => {
   411→        setLocate({
   412→          kind: "error",
   413→          message:
   414→            err.code === err.PERMISSION_DENIED
   415→              ? "دسترسی موقعیت مکانی داده نشد"
   416→              : "موقعیت مکانی در دسترس نیست",
   417→        });
   418→      },
   419→      { enableHighAccuracy: true, timeout: 10000, maximumAge: 30000 }
   420→    );
   421→  }, [locate.kind, zoomToStation]);
   422→
   423→  const resetView = useCallback(() => {
   424→    commit({ scale: 1, tx: 0, ty: 0 });
   425→    setLocate({ kind: "idle" });
   426→  }, [commit]);
   427→
   428→  const highlighted = useMemo(() => {
   429→    const set = new Set(highlightStations ?? []);
   430→    if (locate.kind === "ok") set.add(locate.stationId);
   431→    return set;
   432→  }, [highlightStations, locate]);
   433→
   434→  /* ============================================================
   435→     موتور برچسب‌ها — مقیاس «فضای صفحه»:
   436→     برخلاف گذشته که متن‌ها داخل گروه بزرگ‌نمایی‌شونده بودند (و در
   437→     بزرگ‌نمایی ۱ روی هم ریخته دیده می‌شدند)، اینجا هر برچسب:
   438→     ۱. با اندازه ثابت روی صفحه رندر می‌شود (مستقل از زوم)
   439→     ۲. زنده از مختصات ایستگاه پیروی می‌کند
   440→     ۳. فقط اگر با برچسب مهم‌تری همپوشانی نداشته باشد نمایش
   441→        داده می‌شود — با نزدیک‌تر شدن (زوم بیشتر) برچسب‌های
   442→        بیشتری جا می‌گیرند و ظاهر می‌شوند
   443→     ============================================================ */
   444→  const fit = box.w > 0 && box.h > 0 ? Math.min(box.w / MAP_CANVAS.w, box.h / MAP_CANVAS.h) : 0;
   445→
   446→  const labelLayout = useMemo(() => {
   447→    const empty: { labels: LabelItem[]; badges: BadgeItem[] } = { labels: [], badges: [] };
   448→    if (fit <= 0) return empty;
   449→    const toVb = (px: number) => px / fit;
   450→
   451→    const boxes: Box[] = [];
   452→    const overlaps = (b: Box) =>
   453→      boxes.some((o) => b.x0 < o.x1 && b.x1 > o.x0 && b.y0 < o.y1 && b.y1 > o.y0);
   454→
   455→    /* ---------- ناحیه کارت‌های HTML روی نقشه (راهنما + کنترل‌ها) —
   456→       برچسب و نشان نباید زیر این کارت‌ها پنهان شوند ---------- */
   457→    const ox = (box.w - MAP_CANVAS.w * fit) / 2;
   458→    const oy = (box.h - MAP_CANVAS.h * fit) / 2;
   459→    const htmlZone = (left: number, top: number, w: number, h: number): Box | null => {
   460→      const x0 = Math.max(0, (left - ox) / fit);
   461→      const y0 = Math.max(0, (top - oy) / fit);
   462→      const x1 = Math.min(MAP_CANVAS.w, (left + w - ox) / fit);
   463→      const y1 = Math.min(MAP_CANVAS.h, (top + h - oy) / fit);
   464→      if (x1 <= x0 || y1 <= y0) return null;
   465→      return { x0, y0, x1, y1 };
   466→    };
   467→    const legendZone = htmlZone(10, 10, 158, legendOpen ? 185 : 40);
   468→    if (legendZone) boxes.push(legendZone);
   469→    // ستون دکمه‌های شناور پایین-چپ (۴ دکمه ۳۶px با فاصله)
   470→    const controlsZone = htmlZone(8, box.h - 196, 44, 188);
   471→    if (controlsZone) boxes.push(controlsZone);
   472→
   473→    /* ---------- نشان شماره خطوط (اندازه ثابت، پشته‌ای در ایستگاه مشترک) ---------- */
   474→    const badges: BadgeItem[] = [];
   475→    const badgeN = new Map<string, number>();
   476→    for (const ml of MAP_LINES) {
   477→      for (const id of [ml.stations[0], ml.stations[ml.stations.length - 1]]) {
   478→        const s = MAP_STATION_BY_ID.get(id);
   479→        if (!s) continue;
   480→        const n = badgeN.get(id) ?? 0;
   481→        badgeN.set(id, n + 1);
   482→        const r = toVb(8.5);
   483→        const side = s.x > MAP_CANVAS.w / 2 ? -1 : 1;
   484→        const bx = s.x * t.scale + t.tx + side * toVb(14);
   485→        const by = s.y * t.scale + t.ty - toVb(15) - n * toVb(19);
   486→        if (bx < -r || bx > MAP_CANVAS.w + r || by < -r || by > MAP_CANVAS.h + r) continue;
   487→        const bBox: Box = {
   488→          x0: bx - r - toVb(1),
   489→          y0: by - r - toVb(1),
   490→          x1: bx + r + toVb(1),
   491→          y1: by + r + toVb(1),
   492→        };
   493→        // نشان زیر کارت‌های HTML پنهان نمی‌شود
   494→        if (overlaps(bBox)) continue;
   495→        badges.push({
   496→          key: `${ml.lineId}-${id}-${n}`,
   497→          x: bx,
   498→          y: by,
   499→          r,
   500→          stroke: toVb(1.5),
   501→          font: toVb(10),
   502→          num: toFa(TEHRAN_LINES.find((l) => l.id === ml.lineId)?.number ?? 0),
   503→          color: ml.color,
   504→          ink: ml.lineId === "l4" ? "#1a1a1a" : "#ffffff",
   505→        });
   506→        boxes.push(bBox);
   507→      }
   508→    }
   509→
   510→    /* ---------- ناحیه حفاظتی نشان بزرگ ایستگاه‌های تبادلی ---------- */
   511→    const interExcl: { id: string; box: Box }[] = [];
   512→    for (const s of MAP_STATIONS) {
   513→      if (!INTERCHANGE_IDS.has(s.id)) continue;
   514→      const half = 6.4 * t.scale + 3 / fit;
   515→      const X = s.x * t.scale + t.tx;
   516→      const Y = s.y * t.scale + t.ty;
   517→      interExcl.push({
   518→        id: s.id,
   519→        box: { x0: X - half, y0: Y - half, x1: X + half, y1: Y + half },
   520→      });
   521→    }
   522→    const hitsInterchange = (b: Box, selfId: string) =>
   523→      interExcl.some(
   524→        (e) => e.id !== selfId && b.x0 < e.box.x1 && b.x1 > e.box.x0 && b.y0 < e.box.y1 && b.y1 > e.box.y0
   525→      );
   526→
   527→    /* ---------- برچسب‌ها به ترتیب اولویت (تبادلی ← پرچم‌دار ← عادی) ---------- */
   528→    const cands = MAP_STATIONS.map((s: MapStation) => {
   529→      const isInter = INTERCHANGE_IDS.has(s.id);
   530→      const prio = highlighted.has(s.id)
   531→        ? 1000
   532→        : isInter && s.label
   533→          ? 500
   534→          : isInter
   535→            ? 400
   536→            : s.label
   537→              ? 300
   538→              : 200;
   539→      return { s, prio, major: isInter || s.label };
   540→    }).sort((a, b) => b.prio - a.prio);
   541→
   542→    const labels: LabelItem[] = [];
   543→    for (const { s, prio, major } of cands) {
   544→      void prio;
   545→      const X = s.x * t.scale + t.tx;
   546→      const Y = s.y * t.scale + t.ty;
   547→      const margin = toVb(26);
   548→      if (X < -margin || X > MAP_CANVAS.w + margin || Y < -margin || Y > MAP_CANVAS.h + margin) continue;
   549→
   550→      const fontPx = major ? 11.5 : 10.5;
   551→      const weight = major ? 800 : 600;
   552→      const wVb = toVb(textWidthPx(s.name, weight, fontPx, fontsTick) + 7);
   553→      const hVb = toVb(fontPx * 1.1 + 5);
   554→      const gapVb = toVb(10);
   555→      // جهت متن فارسی rtl است: anchor «end» یعنی لبه چپ متن در x
   556→      const prefRight = s.x <= MAP_CANVAS.w / 2;
   557→      const sides: ("right" | "left")[] = prefRight ? ["right", "left"] : ["left", "right"];
   558→
   559→      for (const side of sides) {
   560→        const bx0 = side === "right" ? X + gapVb : X - gapVb - wVb;
   561→        const by0 = Y - hVb / 2;
   562→        const b: Box = { x0: bx0, y0: by0, x1: bx0 + wVb, y1: by0 + hVb };
   563→        if (b.x0 < 1 || b.x1 > MAP_CANVAS.w - 1 || b.y0 < 1 || b.y1 > MAP_CANVAS.h - 1) continue;
   564→        if (overlaps(b) || hitsInterchange(b, s.id)) continue;
   565→        boxes.push(b);
   566→        labels.push({
   567→          id: s.id,
   568→          name: s.name,
   569→          x: side === "right" ? X + gapVb : X - gapVb,
   570→          y: Y + toVb(fontPx * 0.35),
   571→          anchor: side === "right" ? "end" : "start",
   572→          fontSize: toVb(fontPx),
   573→          weight,
   574→          halo: toVb(major ? 3.4 : 2.8),
   575→        });
   576→        break;
   577→      }
   578→    }
   579→
   580→    return { labels, badges };
   581→  }, [t, fit, fontsTick, highlighted, legendOpen, box.w, box.h]);
   582→

   584→    <div className={`relative overflow-hidden rounded-2xl ${className}`}>
   585→      {/* لایه SVG نقشه */}
   586→      <svg
   587→        ref={svgRef}
   588→        viewBox={`0 0 ${MAP_CANVAS.w} ${MAP_CANVAS.h}`}
   589→        className="w-full h-full map-viewport select-none"
   590→        style={{ touchAction: "none" }}
   591→        onPointerDown={onPointerDown}
   592→        onPointerMove={onPointerMove}
   593→        onPointerUp={endPointer}
   594→        onPointerCancel={endPointer}

   596→        aria-label="نقشه شماتیک شبکه متروی تهران — با دو انگشت بزرگ‌نمایی کنید"
   597→      >
   598→        <defs>
   599→          <pattern id="map-grid" width="28" height="28" patternUnits="userSpaceOnUse">
   600→            <path d="M 28 0 L 0 0 0 28" fill="none" stroke="var(--mq-map-grid)" strokeWidth="0.7" />
   601→          </pattern>
   602→        </defs>
   603→        <rect width={MAP_CANVAS.w} height={MAP_CANVAS.h} fill="var(--mq-map-bg)" />
   604→        <rect width={MAP_CANVAS.w} height={MAP_CANVAS.h} fill="url(#map-grid)" />
   605→
   606→        <g transform={`translate(${t.tx} ${t.ty}) scale(${t.scale})`}>
   607→          {/* خطوط — هر خط با زیرلایه روشن تا در تقاطع‌ها از هم جدا دیده شوند */}
   608→          {MAP_LINES.map((ml) => (
   609→            <g key={ml.lineId}>
   610→              <polyline
   611→                points={linePoints(ml.stations)}
   612→                fill="none"
   613→                stroke="var(--mq-map-bg)"
   614→                strokeWidth={10.5}
   615→                strokeLinecap="round"
   616→                strokeLinejoin="round"
   617→              />
   618→              <polyline
   619→                points={linePoints(ml.stations)}
   620→                fill="none"
   621→                stroke={ml.color}
   622→                strokeWidth={7}
   623→                strokeLinecap="round"
   624→                strokeLinejoin="round"
   625→              />
   626→            </g>
   627→          ))}
   628→
   629→          {/* ایستگاه‌ها — فقط نشان‌ها؛ برچسب‌ها در لایه جداگانه پایین رندر می‌شوند */}
   630→          {MAP_STATIONS.map((s) => {
   631→            const isInter = INTERCHANGE_IDS.has(s.id);
   632→            const isHi = highlighted.has(s.id);
   633→            const r = isInter ? 6.4 : 4.6;
   634→            return (
   635→              <g
   636→                key={s.id}
   637→                onClick={(e) => {
   638→                  e.stopPropagation();
   639→                  if (movedRef.current) return; // بعد از جابجایی نقشه، تپ نیست
   640→                  onStationSelect?.(s.id);
   641→                }}
   642→                className="cursor-pointer"
   643→              >
   644→                <circle cx={s.x} cy={s.y} r={15} fill="transparent" />
   645→                {isInter && <circle cx={s.x} cy={s.y} r={r + 3} fill="var(--mq-card)" opacity={0.95} />}
   646→                {isHi && <circle cx={s.x} cy={s.y} r={13} fill="#0d58a6" opacity={0.18} className="map-pulse" />}
   647→                <circle
   648→                  cx={s.x}
   649→                  cy={s.y}
   650→                  r={isHi ? 6.5 : r}
   651→                  fill={isHi ? "#0d58a6" : "var(--mq-card)"}
   652→                  stroke={isHi ? "#0d58a6" : "var(--mq-ink)"}
   653→                  strokeWidth={isHi ? 2.4 : 1.6}
   654→                />

   658→        </g>
   659→
   660→        {/* ============================================================
   661→            لایه برچسب‌ها (خارج از گروه بزرگ‌نمایی — اندازه ثابت روی صفحه)
   662→            ============================================================ */}
   663→        {fit > 0 && (
   664→          <g style={{ pointerEvents: "none" }}>
   665→            {/* نشان شماره خطوط */}
   666→            {labelLayout.badges.map((b) => (
   667→              <g key={b.key}>
   668→                <circle
   669→                  cx={b.x}
   670→                  cy={b.y}
   671→                  r={b.r}
   672→                  fill={b.color}
   673→                  stroke="var(--mq-card)"
   674→                  strokeWidth={b.stroke}
   675→                />
   676→                <text
   677→                  x={b.x}
   678→                  y={b.y + b.r * 0.38}
   679→                  textAnchor="middle"
   680→                  fontSize={b.font}
   681→                  fontWeight={800}
   682→                  fill={b.ink}
   683→                  style={{ fontFamily: "Vazirmatn, sans-serif" }}
   684→                >
   685→                  {b.num}
   686→                </text>
   687→              </g>
   688→            ))}
   689→
   690→            {/* نام ایستگاه‌ها — با هاله روشن و کنترل همپوشانی */}
   691→            {labelLayout.labels.map((L) => (
   692→              <text
   693→                key={L.id}
   694→                x={L.x}
   695→                y={L.y}
   696→                textAnchor={L.anchor}
   697→                fontSize={L.fontSize}
   698→                fontWeight={L.weight}
   699→                fill="var(--mq-ink)"
   700→                stroke="var(--mq-card)"
   701→                strokeWidth={L.halo}
   702→                paintOrder="stroke"
   703→                strokeLinejoin="round"
   704→                style={{
   705→                  fontFamily: "Vazirmatn, sans-serif",
   706→                  direction: "rtl",
   707→                  pointerEvents: "auto",
   708→                  cursor: "pointer",
   709→                  userSelect: "none",
   710→                }}
   711→                onClick={(e) => {
   712→                  e.stopPropagation();
   713→                  if (movedRef.current) return; // بعد از جابجایی نقشه، تپ نیست
   714→                  onStationSelect?.(L.id);
   715→                }}
   716→              >
   717→                {L.name}
   718→              </text>
   719→            ))}
   720→          </g>
   721→        )}
   722→      </svg>
   723→
   724→      {/* ================= کارت راهنمای خطوط (جای دکمه در هر دو حالت ثابت است) ================= */}

   726→        <button
   727→          onClick={() => setLegendOpen((o) => !o)}
   728→          aria-expanded={legendOpen}
   729→          aria-label={legendOpen ? "بستن راهنمای خطوط" : "نمایش راهنمای خطوط"}
   730→          className="w-full h-9 px-2.5 flex items-center gap-1.5 text-[10px] font-bold text-metro-ink active:bg-surface-container transition-colors"
   731→        >
   732→          <MI name="layers" size={15} className="text-metro-primary-c shrink-0" />
   733→          راهنمای خطوط
   734→          <MI
   735→            name="expand_more"
   736→            size={14}
   737→            className={`text-outline mr-auto transition-transform ${legendOpen ? "rotate-180" : ""}`}
   738→          />
   739→        </button>
   740→        {legendOpen && (
   741→          <div className="fade-in px-2.5 pb-2 grid grid-cols-1 gap-1 border-t border-metro-outline-var/50 pt-1.5">
   742→            {TEHRAN_LINES.map((l) => (
   743→              <div key={l.id} className="flex items-center gap-1.5">
   744→                <span className="w-2.5 h-2.5 rounded-full shrink-0" style={{ backgroundColor: l.color }} />
   745→                <span className="text-[9px] font-semibold text-metro-ink truncate">
   746→                  {l.name} — {l.terminals[0]} / {l.terminals[1]}
   747→                </span>
   748→              </div>
   749→            ))}
   750→          </div>
   751→        )}
   752→      </div>
   753→
   754→      {/* ================= نتیجه موقعیت‌یاب ================= */}
   755→      {locate.kind === "busy" && (
   756→        <div className="absolute top-2.5 right-2.5 bg-metro-card/95 backdrop-blur rounded-full shadow-md border border-metro-outline-var px-3 py-1.5 flex items-center gap-1.5">
   757→          <span className="w-3.5 h-3.5 border-2 border-metro-primary-c/30 border-t-metro-primary-c rounded-full animate-spin" />
   758→          <span className="text-[10px] font-bold text-metro-ink">در حال یافتن موقعیت...</span>

   762→        <div className="fade-in absolute top-2.5 right-2.5 max-w-[70%] bg-metro-card/95 backdrop-blur rounded-xl shadow-md border border-metro-outline-var px-3 py-1.5 flex items-center gap-2">
   763→          <MI name="near_me" size={15} className="text-metro-primary-c shrink-0" />
   764→          <span className="text-[10px] font-bold text-metro-ink truncate">
   765→            نزدیک‌ترین ایستگاه: {locate.stationName} (حدود {toFa(Math.round(locate.distanceM / 50) * 50)} متر)
   766→          </span>
   767→          <button
   768→            onClick={() => setLocate({ kind: "idle" })}
   769→            aria-label="بستن"
   770→            className="text-outline shrink-0 active:scale-90 transition-transform"
   771→          >
   772→            <MI name="close" size={13} />
   773→          </button>
   774→        </div>
   775→      )}
   776→      {(locate.kind === "far" || locate.kind === "error") && (
   777→        <div className="fade-in absolute top-2.5 right-2.5 max-w-[70%] bg-metro-card/95 backdrop-blur rounded-xl shadow-md border border-error/40 px-3 py-1.5 flex items-center gap-2">
   778→          <MI name="location_off" size={15} className="text-error shrink-0" />
   779→          <span className="text-[10px] font-bold text-metro-ink">
   780→            {locate.kind === "far" ? "در محدوده ایستگاه‌های مترو نیستید" : locate.message}
   781→          </span>
   782→          <button
   783→            onClick={() => setLocate({ kind: "idle" })}
   784→            aria-label="بستن"
   785→            className="text-outline shrink-0 active:scale-90 transition-transform"
   786→          >
   787→            <MI name="close" size={13} />
   788→          </button>
   789→        </div>
   790→      )}
   791→
   792→      {/* ================= کنترل‌های شناور (بزرگ‌نمایی + موقعیت) ================= */}
   793→      <div className="absolute bottom-2.5 left-2.5 flex flex-col gap-1.5">
   794→        <button
   795→          onClick={locateMe}
   796→          aria-label="نزدیک‌ترین ایستگاه به من"
   797→          className="w-9 h-9 rounded-xl bg-metro-card shadow border border-metro-outline-var flex items-center justify-center text-metro-primary-c active:scale-95 transition-all"
   798→        >
   799→          <MI name="my_location" size={18} />
   800→        </button>
   801→        <button
   802→          onClick={() => zoomAt(1.25)}
   803→          aria-label="بزرگ‌نمایی"
   804→          className="w-9 h-9 rounded-xl bg-metro-card shadow border border-metro-outline-var flex items-center justify-center text-metro-primary-c active:scale-95 transition-all"

   806→          <MI name="add" size={18} />

   809→          onClick={() => zoomAt(0.8)}

   819→        >
   820→          <MI name="center_focus_strong" size={18} />
   821→        </button>
