   693→  const drawerOverlay = document.getElementById('drawerOverlay');
   694→  const drawerPanel = document.getElementById('drawerPanel');

   696→  function openDrawer() {
   697→    if (!menuDrawer) return;
   698→    menuDrawer.classList.remove('pointer-events-none');
   699→    drawerOverlay.classList.remove('opacity-0');
   700→    drawerOverlay.classList.add('opacity-100');
   701→    drawerPanel.classList.remove('translate-x-full');
   702→  }
   703→
   704→  function closeDrawer() {
   705→    if (!menuDrawer) return;
   706→    drawerOverlay.classList.remove('opacity-100');
   707→    drawerOverlay.classList.add('opacity-0');
   708→    drawerPanel.classList.add('translate-x-full');
   709→    setTimeout(() => {
   710→      menuDrawer.classList.add('pointer-events-none');
   711→    }, 300);
   712→  }
