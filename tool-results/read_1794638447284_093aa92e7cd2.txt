     2→   درمسیر — پنل مدیریت | مسیریاب API نسخه‌دار /api/v1
     3→   استاندارد: ورودی معتبر، خطای استاندارد، نقش‌ها، لاگ عملیات

     5→import { db, now, bumpApiStats, audit, changeLog } from "./db";
     6→import {
     7→  verifyPassword,
     8→  newToken,
     9→  loginAllowed,
    10→  registerFailedLogin,
    11→  clearLoginAttempts,
    12→  hashPassword,
    13→} from "./auth";
    14→
    15→type Role = "super" | "operator" | "viewer";
    16→interface AdminRow {
    17→  id: number;
    18→  username: string;
    19→  password_hash: string;
    20→  display_name: string;
    21→  role: Role;
    22→  is_active: number;
    23→  created_at: string;
    24→  last_login: string | null;
    25→}
    26→
    27→const SESSION_HOURS = 12;
    28→const RANK: Record<Role, number> = { viewer: 1, operator: 2, super: 3 };
    29→
    30→const ok = (data: any, status = 200) =>
    31→  new Response(JSON.stringify({ ok: true, data }), {
    32→    status,
    33→    headers: { "Content-Type": "application/json; charset=utf-8" },
    34→  });
    35→
    36→const fail = (code: string, message: string, status = 400) =>
    37→  new Response(JSON.stringify({ ok: false, error: { code, message } }), {
    38→    status,
    39→    headers: { "Content-Type": "application/json; charset=utf-8" },
    40→  });
    41→
    42→const str = (v: any, max = 300) =>
    43→  typeof v === "string" ? v.trim().slice(0, max) : "";
    44→
    45→const clientIp = (req: Request) =>
    46→  req.headers.get("X-Real-IP") || req.headers.get("X-Forwarded-For")?.split(",")[0]?.trim() || "local";
    47→
    48→function getSession(req: Request): AdminRow | null {
    49→  const auth = req.headers.get("Authorization") || "";
    50→  const token = auth.startsWith("Bearer ") ? auth.slice(7) : "";
    51→  if (!token) return null;
    52→  const row = db
    53→    .query(
    54→      `SELECT a.* FROM sessions s JOIN admins a ON a.id = s.admin_id
    55→       WHERE s.token = ? AND s.expires_at > ? AND a.is_active = 1`
    56→    )
    57→    .get(token, now()) as AdminRow | undefined;
    58→  return row ?? null;
    59→}
    60→
    61→function requireRole(admin: AdminRow, role: Role): Response | null {
    62→  if (RANK[admin.role] < RANK[role]) {
    63→    return fail("FORBIDDEN", "سطح دسترسی شما برای این عملیات کافی نیست", 403);
    64→  }
    65→  return null;
    66→}
    67→
    68→function touchSession(adminId: number, token: string, req: Request) {
    69→  const expires = new Date(Date.now() + SESSION_HOURS * 3600 * 1000).toISOString();
    70→  db.prepare(
    71→    "INSERT INTO sessions (token, admin_id, created_at, expires_at, ip, user_agent) VALUES (?,?,?,?,?,?) ON CONFLICT(token) DO UPDATE SET expires_at = excluded.expires_at"
    72→  ).run(token, adminId, now(), expires, clientIp(req), req.headers.get("User-Agent") ?? "");
    73→  db.prepare("DELETE FROM sessions WHERE expires_at < ?").run(now());
    74→}

    77→
    78→export async function handleApi(req: Request, url: URL): Promise<Response> {
    79→  bumpApiStats();
    80→  const path = url.pathname.replace(/^\/api\/v1\/?/, "").replace(/\/$/, "");
    81→  const method = req.method.toUpperCase();
    82→  const ip = clientIp(req);
    83→
    84→  let body: any = {};
    85→  if (method === "POST" || method === "PUT" || method === "PATCH") {
    86→    try {
    87→      body = await req.json();
    88→    } catch {
    89→      body = {};
    90→    }
    91→  }
    92→
    93→  try {
    94→    return await route(path, method, url, req, body, ip);
    95→  } catch (e: any) {
    96→    console.error("[darmasir-admin][api]", e?.message ?? e);
    97→    return fail("INTERNAL", "خطای داخلی سرور پنل", 500);
    98→  }
    99→}
   100→
   101→async function route(
   102→  path: string,
   103→  method: string,
   104→  url: URL,
   105→  req: Request,
   106→  body: any,
   107→  ip: string
   108→): Promise<Response> {
   109→  const admin = getSession(req);
   110→  const seg = path.split("/");
   111→
   112→  /* ---------- auth ---------- */
   113→  if (path === "auth/login" && method === "POST") {
   114→    if (!loginAllowed(ip))
   115→      return fail("RATE_LIMIT", "تلاش‌های ناموفق زیاد است؛ چند دقیقه بعد دوباره امتحان کنید", 429);
   116→    const username = str(body.username, 60);
   117→    const password = typeof body.password === "string" ? body.password : "";
   118→    if (!username || !password)
   119→      return fail("BAD_INPUT", "نام کاربری و گذرواژه الزامی است");
   120→    const row = db.query("SELECT * FROM admins WHERE username = ?").get(username) as AdminRow | undefined;
   121→    if (!row || !verifyPassword(password, row.password_hash) || !row.is_active) {
   122→      registerFailedLogin(ip);
   123→      audit(null, username || "ناشناس", "login_failed", `نام کاربری: ${username}`, ip);
   124→      return fail("INVALID_LOGIN", "نام کاربری یا گذرواژه نادرست است", 401);
   125→    }
   126→    clearLoginAttempts(ip);
   127→    const token = newToken();
   128→    touchSession(row.id, token, req);
   129→    db.prepare("UPDATE admins SET last_login = ? WHERE id = ?").run(now(), row.id);
   130→    audit(row.id, row.display_name, "login", "ورود موفق", ip);
   131→    return ok({
   132→      token,
   133→      admin: { id: row.id, username: row.username, display_name: row.display_name, role: row.role },
   134→    });
   135→  }
   136→
   137→  if (!admin) return fail("UNAUTHORIZED", "ابتدا وارد پنل شوید", 401);
   138→
   139→  if (path === "auth/me" && method === "GET") {
   140→    return ok({ id: admin.id, username: admin.username, display_name: admin.display_name, role: admin.role });
   141→  }
   142→
   143→  if (path === "auth/change-password" && method === "POST") {
   144→    const current = typeof body.current_password === "string" ? body.current_password : "";
   145→    const next = typeof body.new_password === "string" ? body.new_password : "";
   146→    if (!verifyPassword(current, admin.password_hash))
   147→      return fail("INVALID_LOGIN", "گذرواژه فعلی نادرست است", 401);
   148→    if (next.length < 8) return fail("BAD_INPUT", "گذرواژه جدید باید حداقل ۸ کاراکتر باشد");
   149→    db.prepare("UPDATE admins SET password_hash = ? WHERE id = ?").run(hashPassword(next), admin.id);
   150→    audit(admin.id, admin.display_name, "password_change", "تغییر گذرواژه شخصی", ip);
   151→    return ok({ changed: true });
   152→  }
   153→
   154→  if (path === "auth/logout" && method === "POST") {
   155→    const auth = req.headers.get("Authorization") || "";
   156→    const token = auth.startsWith("Bearer ") ? auth.slice(7) : "";
   157→    db.prepare("DELETE FROM sessions WHERE token = ?").run(token);
   158→    audit(admin.id, admin.display_name, "logout", "خروج از پنل", ip);
   159→    return ok({ logged_out: true });
   160→  }
   161→
   162→  /* ---------- dashboard ---------- */
   163→  if (path === "dashboard" && method === "GET") {
   164→    const q = (sql: string) => (db.query(sql).get() as any).c ?? 0;
   165→    const days: { day: string; count: number }[] = [];
   166→    for (let i = 6; i >= 0; i--) {
   167→      const d = new Date(Date.now() - i * 86400000).toISOString().slice(0, 10);
   168→      days.push({ day: d, count: 0 });
   169→    }
   170→    (db.query("SELECT day, count FROM api_stats WHERE day >= ?").all(
   171→      new Date(Date.now() - 6 * 86400000).toISOString().slice(0, 10)
   172→    ) as any[]).forEach((r) => {
   173→      const f = days.find((x) => x.day === r.day);
   174→      if (f) f.count = r.count;
   175→    });
   176→    return ok({
   177→      users_total: q("SELECT COUNT(*) AS c FROM app_users"),
   178→      users_active: q("SELECT COUNT(*) AS c FROM app_users WHERE status = 'active'"),
   179→      trips_total: q("SELECT COALESCE(SUM(trips_count),0) AS c FROM app_users"),
   180→      lines_active: q("SELECT COUNT(*) AS c FROM lines WHERE is_active = 1"),
   181→      stations_total: q("SELECT COUNT(*) AS c FROM stations"),
   182→      ads_active: (
   183→        db
   184→          .query("SELECT COUNT(*) AS c FROM ads WHERE is_active = 1 AND (ends_at IS NULL OR ends_at > ?)")
   185→          .get(now()) as any
   186→      ).c,
   187→      api_last7: days,
   188→      audit_recent: db
   189→        .query("SELECT action, admin_name, details, created_at FROM audit_log ORDER BY id DESC LIMIT 8")
   190→        .all(),
   191→      services: [
   192→        { name: "دیتابیس پنل", status: "ok", note: "SQLite محلی فعال" },
   193→        {
   194→          name: "سرویس صوتی",
   195→          status: (db.query("SELECT value FROM settings WHERE key = 'voice_enabled'").get() as any)?.value === "on" ? "ok" : "off",
   196→          note: (db.query("SELECT value FROM settings WHERE key = 'voice_service'").get() as any)?.value ?? "-",
   197→        },
   198→        { name: "دروازه API پنل", status: "ok", note: "پورت ۳۰۳۵" },
   199→      ],
   200→    });
   201→  }
   202→
   203→  /* ---------- users (کاربران اپ) ---------- */
   204→  if (seg[0] === "users") {
   205→    if (method === "GET") {
   206→      if (seg[1] && !isNaN(Number(seg[1]))) {
   207→        const u = db
   208→          .query("SELECT id, phone, display_name, status, trips_count, created_at, last_seen FROM app_users WHERE id = ?")
   209→          .get(Number(seg[1]));
   210→        if (!u) return fail("NOT_FOUND", "کاربر یافت نشد", 404);
   211→        return ok(u);
   212→      }
   213→      const q = str(url.searchParams.get("q") ?? "", 60);
   214→      const status = str(url.searchParams.get("status") ?? "", 20);
   215→      let sql = "SELECT id, phone, display_name, status, trips_count, created_at, last_seen FROM app_users WHERE 1=1";
   216→      const args: any[] = [];
   217→      if (q) {
   218→        sql += " AND (display_name LIKE ? OR phone LIKE ?)";
   219→        args.push(`%${q}%`, `%${q}%`);
   220→      }
   221→      if (status) {
   222→        sql += " AND status = ?";
   223→        args.push(status);
   224→      }
   225→      sql += " ORDER BY id DESC LIMIT 200";
   226→      return ok(db.prepare(sql).all(...args));
   227→    }
   228→    const guard = requireRole(admin, "operator");
   229→    if (guard) return guard;
   230→    if (path === "users/bulk-status" && method === "POST") {
   231→      const status = ["active", "limited", "disabled"].includes(body.status) ? body.status : null;
   232→      const ids: number[] = Array.isArray(body.ids) ? body.ids.map(Number).filter(Boolean) : [];
   233→      if (!status || !ids.length) return fail("BAD_INPUT", "وضعیت و فهرست کاربران الزامی است");
   234→      const stmt = db.prepare("UPDATE app_users SET status = ? WHERE id = ?");
   235→      let n = 0;
   236→      for (const id of ids) { stmt.run(status, id); n++; }
   237→      changeLog("app_user", null, "bulk_status", admin.display_name, `${n} کاربر → ${status}`);
   238→      audit(admin.id, admin.display_name, "user_bulk_status", `${n} کاربر → ${status}`, ip);
   239→      return ok({ updated: n });
   240→    }
   241→    if (seg[1] === "trips" && seg[2] && method === "POST") {
   242→      const id = Number(seg[2]);
   243→      const u = db.prepare("SELECT * FROM app_users WHERE id = ?").get(id) as any;
   244→      if (!u) return fail("NOT_FOUND", "کاربر یافت نشد", 404);
   245→      const delta = Math.round(Number(body.delta) || 0);
   246→      if (!delta) return fail("BAD_INPUT", "مقدار تغییر سفر الزامی است");
   247→      const next = Math.max(0, (u.trips_count || 0) + delta);
   248→      db.prepare("UPDATE app_users SET trips_count = ? WHERE id = ?").run(next, id);
   249→      changeLog("app_user", id, "update", admin.display_name, `سفرها: ${u.trips_count} → ${next}`);
   250→      audit(admin.id, admin.display_name, "user_update", `سفرهای کاربر #${id} → ${next}`, ip);
   251→      return ok({ trips_count: next });
   252→    }
   253→    if (method === "POST") {
   254→      const name = str(body.display_name, 80);
   255→      const phone = str(body.phone, 20) || null;
   256→      const status = ["active", "limited", "disabled"].includes(body.status) ? body.status : "active";
   257→      if (!name) return fail("BAD_INPUT", "نام کاربر الزامی است");
   258→      try {
   259→        const r = db
   260→          .prepare("INSERT INTO app_users (phone, display_name, status, created_at, last_seen) VALUES (?,?,?,?,?)")
   261→          .run(phone, name, status, now(), now());
   262→        changeLog("app_user", Number(r.lastInsertRowid), "create", admin.display_name, name);
   263→        audit(admin.id, admin.display_name, "user_create", name, ip);
   264→        return ok({ id: Number(r.lastInsertRowid) }, 201);
   265→      } catch {
   266→        return fail("DUPLICATE", "این شماره قبلاً ثبت شده است", 409);
   267→      }
   268→    }
   269→    const id = Number(seg[1]);
   270→    if (!id) return fail("NOT_FOUND", "کاربر یافت نشد", 404);
   271→    if (method === "PUT") {
   272→      const patch: string[] = [];
   273→      const args: any[] = [];
   274→      if (body.display_name !== undefined) {
   275→        patch.push("display_name = ?");
   276→        args.push(str(body.display_name, 80));
   277→      }
   278→      if (body.phone !== undefined) {
   279→        patch.push("phone = ?");
   280→        args.push(str(body.phone, 20) || null);
   281→      }
   282→      if (body.status !== undefined) {
   283→        if (!["active", "limited", "disabled"].includes(body.status))
   284→          return fail("BAD_INPUT", "وضعیت نامعتبر است");
   285→        patch.push("status = ?");
   286→        args.push(body.status);
   287→      }
   288→      if (!patch.length) return fail("BAD_INPUT", "چیزی برای بروزرسانی نیست");
   289→      args.push(id);
   290→      db.prepare(`UPDATE app_users SET ${patch.join(", ")} WHERE id = ?`).run(...args);
   291→      changeLog("app_user", id, "update", admin.display_name, JSON.stringify(body).slice(0, 200));
   292→      audit(admin.id, admin.display_name, "user_update", `#${id}`, ip);
   293→      return ok({ updated: true });
   294→    }
   295→    if (method === "DELETE") {
   296→      db.prepare("DELETE FROM app_users WHERE id = ?").run(id);
   297→      changeLog("app_user", id, "delete", admin.display_name);
   298→      audit(admin.id, admin.display_name, "user_delete", `#${id}`, ip);
   299→      return ok({ deleted: true });
   300→    }
   301→  }
   302→
   303→  /* ---------- lines ---------- */
   304→  if (seg[0] === "lines") {
   305→    if (method === "GET") {
   306→      const mode = str(url.searchParams.get("mode") ?? "", 10);
   307→      const baseSql =
   308→        "SELECT l.*, (SELECT COUNT(*) FROM stations s WHERE s.line_id = l.id AND s.is_active = 1) AS station_count FROM lines l";
   309→      const rows = mode
   310→        ? db.prepare(`${baseSql} WHERE l.mode = ? ORDER BY l.sort`).all(mode)
   311→        : db.prepare(`${baseSql} ORDER BY l.sort`).all();
   312→      return ok(rows);
   313→    }
   314→    const guard = requireRole(admin, "operator");
   315→    if (guard) return guard;
   316→    if (path === "lines/reorder" && method === "POST") {
   317→      const id = Number(body.id);
   318→      const dir = body.dir === "up" ? "up" : body.dir === "down" ? "down" : null;
   319→      if (!id || !dir) return fail("BAD_INPUT", "شناسه خط و جهت جابه‌جایی الزامی است");
   320→      const cur = db.prepare("SELECT * FROM lines WHERE id = ?").get(id) as any;
   321→      if (!cur) return fail("NOT_FOUND", "خط یافت نشد", 404);
   322→      const neighbor = (
   323→        dir === "up"
   324→          ? db.query("SELECT * FROM lines WHERE mode = ? AND sort < ? ORDER BY sort DESC LIMIT 1").get(cur.mode, cur.sort)
   325→          : db.query("SELECT * FROM lines WHERE mode = ? AND sort > ? ORDER BY sort LIMIT 1").get(cur.mode, cur.sort)
   326→      ) as any | undefined;
   327→      if (!neighbor) return ok({ moved: false });
   328→      db.prepare("UPDATE lines SET sort = ? WHERE id = ?").run(neighbor.sort, cur.id);
   329→      db.prepare("UPDATE lines SET sort = ? WHERE id = ?").run(cur.sort, neighbor.id);
   330→      changeLog("line", id, "update", admin.display_name, `جابه‌جایی ترتیب ${cur.name}`);
   331→      audit(admin.id, admin.display_name, "line_update", `ترتیب ${cur.name}`, ip);
   332→      return ok({ moved: true });
   333→    }
   334→    if (path === "lines/duplicate" && method === "POST") {
   335→      const id = Number(body.id);
   336→      const cur = db.prepare("SELECT * FROM lines WHERE id = ?").get(id) as any;
   337→      if (!cur) return fail("NOT_FOUND", "خط یافت نشد", 404);
   338→      const maxSort = (db.query("SELECT COALESCE(MAX(sort),0) AS m FROM lines").get() as any).m;
   339→      const r = db
   340→        .prepare(
   341→          "INSERT INTO lines (mode, number, name, color, terminals, direction, is_active, sort) VALUES (?,?,?,?,?,?,0,?)"
   342→        )
   343→        .run(cur.mode, cur.number, `${cur.name} (کپی)`, cur.color, cur.terminals, cur.direction, maxSort + 1);
   344→      changeLog("line", Number(r.lastInsertRowid), "create", admin.display_name, `${cur.name} (کپی)`);
   345→      audit(admin.id, admin.display_name, "line_create", `تکثیر ${cur.name}`, ip);
   346→      return ok({ id: Number(r.lastInsertRowid) }, 201);
   347→    }
   348→    if (method === "POST") {
   349→      const name = str(body.name, 80);
   350→      if (!name) return fail("BAD_INPUT", "نام خط الزامی است");
   351→      const r = db
   352→        .prepare(
   353→          "INSERT INTO lines (mode, number, name, color, terminals, direction, is_active, sort) VALUES (?,?,?,?,?,?,?,?)"
   354→        )
   355→        .run(
   356→          ["metro", "bus", "brt"].includes(body.mode) ? body.mode : "metro",
   357→          Number(body.number) || null,
   358→          name,
   359→          /^#[0-9a-fA-F]{6}$/.test(body.color ?? "") ? body.color : "#009640",
   360→          str(body.terminals, 120) || null,
   361→          str(body.direction, 60) || null,
   362→          body.is_active === 0 ? 0 : 1,
   363→          Number(body.sort) || 0
   364→        );
   365→      changeLog("line", Number(r.lastInsertRowid), "create", admin.display_name, name);
   366→      audit(admin.id, admin.display_name, "line_create", name, ip);
   367→      return ok({ id: Number(r.lastInsertRowid) }, 201);
   368→    }
   369→    const id = Number(seg[1]);
   370→    if (!id) return fail("NOT_FOUND", "خط یافت نشد", 404);
   371→    if (method === "PUT") {
   372→      const cur = db.prepare("SELECT * FROM lines WHERE id = ?").get(id) as any;
   373→      if (!cur) return fail("NOT_FOUND", "خط یافت نشد", 404);
   374→      const next = { ...cur, ...body, name: body.name !== undefined ? str(body.name, 80) || cur.name : cur.name };
   375→      db.prepare(
   376→        "UPDATE lines SET mode=?, number=?, name=?, color=?, terminals=?, direction=?, is_active=?, sort=? WHERE id=?"
   377→      ).run(
   378→        ["metro", "bus", "brt"].includes(next.mode) ? next.mode : "metro",
   379→        Number(next.number) || null,
   380→        next.name,
   381→        /^#[0-9a-fA-F]{6}$/.test(next.color ?? "") ? next.color : "#009640",
   382→        next.terminals,
   383→        next.direction,
   384→        next.is_active ? 1 : 0,
   385→        Number(next.sort) || 0,
   386→        id
   387→      );
   388→      changeLog("line", id, "update", admin.display_name, next.name);
   389→      audit(admin.id, admin.display_name, "line_update", next.name, ip);
   390→      return ok({ updated: true });
   391→    }
   392→    if (method === "DELETE") {
   393→      db.prepare("DELETE FROM lines WHERE id = ?").run(id);
   394→      changeLog("line", id, "delete", admin.display_name);
   395→      audit(admin.id, admin.display_name, "line_delete", `#${id}`, ip);
   396→      return ok({ deleted: true });
   397→    }
   398→  }
   399→
   400→  /* ---------- stations ---------- */
   401→  if (seg[0] === "stations") {
   402→    if (method === "GET") {
   403→      const lineId = Number(url.searchParams.get("line_id"));
   404→      const rows = lineId
   405→        ? db.prepare("SELECT * FROM stations WHERE line_id = ? ORDER BY order_index").all(lineId)
   406→        : db.prepare("SELECT * FROM stations ORDER BY line_id, order_index LIMIT 500").all();
   407→      return ok(rows);
   408→    }
   409→    const guard = requireRole(admin, "operator");
   410→    if (guard) return guard;
   411→    if (path === "stations/reorder" && method === "POST") {
   412→      const id = Number(body.id);
   413→      const dir = body.dir === "up" ? "up" : body.dir === "down" ? "down" : null;
   414→      if (!id || !dir) return fail("BAD_INPUT", "شناسه ایستگاه و جهت جابه‌جایی الزامی است");
   415→      const cur = db.prepare("SELECT * FROM stations WHERE id = ?").get(id) as any;
   416→      if (!cur) return fail("NOT_FOUND", "ایستگاه یافت نشد", 404);
   417→      const neighbor = (

   823→      ads_totals: { impressions: totals.imp, clicks: totals.clk },
   824→      changes,
   825→      active_sessions: sessions,
   826→    });
   827→  }
   828→
   829→  /* ---------- security (فقط مدیر کل) ---------- */
   830→  if (seg[0] === "security" && method === "GET") {
   831→    const guard = requireRole(admin, "super");
   832→    if (guard) return guard;
   833→    return ok({
   834→      admins: db
   835→        .query("SELECT id, username, display_name, role, is_active, created_at, last_login FROM admins ORDER BY id")
   836→        .all(),
   837→      sessions: db
   838→        .query(
   839→          `SELECT s.token, s.created_at, s.expires_at, s.ip, s.user_agent, a.username, a.display_name
   840→           FROM sessions s JOIN admins a ON a.id = s.admin_id
   841→           WHERE s.expires_at > ? ORDER BY s.created_at DESC`
   842→        )
   843→        .all(now()),
   844→      audit: db.query("SELECT * FROM audit_log ORDER BY id DESC LIMIT 100").all(),
   845→    });
   846→  }
   847→
   848→  /* ---------- ابطال نشست (فقط مدیر کل) ---------- */
   849→  if (seg[0] === "sessions" && seg[1] && method === "DELETE") {
   850→    const guard = requireRole(admin, "super");
   851→    if (guard) return guard;
   852→    const token = seg[1];
   853→    const row = db.query("SELECT admin_id FROM sessions WHERE token = ?").get(token) as any;
   854→    if (!row) return fail("NOT_FOUND", "نشست یافت نشد", 404);
   855→    if (row.admin_id === admin.id) return fail("FORBIDDEN", "برای خروج از نشست خود از دکمه خروج استفاده کنید", 403);
   856→    db.prepare("DELETE FROM sessions WHERE token = ?").run(token);
   857→    audit(admin.id, admin.display_name, "session_revoke", "ابطال نشست مدیر", ip);
   858→    return ok({ revoked: true });
   859→  }
   860→
   861→  if (seg[0] === "admins") {
   862→    const guard = requireRole(admin, "super");
   863→    if (guard) return guard;
   864→    if (method === "POST") {
   865→      const username = str(body.username, 60);
   866→      const displayName = str(body.display_name, 80);
   867→      const password = typeof body.password === "string" ? body.password : "";
   868→      const role = ["super", "operator", "viewer"].includes(body.role) ? body.role : "viewer";
   869→      if (!username || !displayName || password.length < 8)
   870→        return fail("BAD_INPUT", "نام کاربری، نام نمایشی و گذرواژه (حداقل ۸ کاراکتر) الزامی است");
   871→      try {
   872→        const r = db
   873→          .prepare("INSERT INTO admins (username, password_hash, display_name, role, created_at) VALUES (?,?,?,?,?)")
   874→          .run(username, hashPassword(password), displayName, role, now());
   875→        audit(admin.id, admin.display_name, "admin_create", username, ip);
   876→        return ok({ id: Number(r.lastInsertRowid) }, 201);
   877→      } catch {
   878→        return fail("DUPLICATE", "این نام کاربری قبلاً وجود دارد", 409);
   879→      }
   880→    }
   881→    const id = Number(seg[1]);
   882→    if (!id) return fail("NOT_FOUND", "مدیر یافت نشد", 404);
   883→    if (method === "PUT") {
   884→      const cur = db.prepare("SELECT * FROM admins WHERE id = ?").get(id) as AdminRow | undefined;
   885→      if (!cur) return fail("NOT_FOUND", "مدیر یافت نشد", 404);
   886→      if (typeof body.password === "string" && body.password.length >= 8) {
   887→        db.prepare("UPDATE admins SET password_hash = ? WHERE id = ?").run(hashPassword(body.password), id);
   888→      }
   889→      if (body.role !== undefined && ["super", "operator", "viewer"].includes(body.role))
   890→        db.prepare("UPDATE admins SET role = ? WHERE id = ?").run(body.role, id);
   891→      if (body.is_active !== undefined)
   892→        db.prepare("UPDATE admins SET is_active = ? WHERE id = ?").run(body.is_active ? 1 : 0, id);
   893→      if (body.display_name !== undefined)
   894→        db.prepare("UPDATE admins SET display_name = ? WHERE id = ?").run(str(body.display_name, 80) || cur.display_name, id);
   895→      audit(admin.id, admin.display_name, "admin_update", cur.username, ip);
   896→      return ok({ updated: true });
   897→    }
   898→    if (method === "DELETE") {
   899→      if (id === admin.id) return fail("FORBIDDEN", "حذف حساب خودتان ممکن نیست", 403);
   900→      const remaining = (db.query("SELECT COUNT(*) AS c FROM admins WHERE is_active = 1").get() as any).c;
   901→      if (remaining <= 1) return fail("FORBIDDEN", "آخرین مدیر فعال را نمی‌توان حذف کرد", 403);
   902→      db.prepare("DELETE FROM admins WHERE id = ?").run(id);
   903→      audit(admin.id, admin.display_name, "admin_delete", `#${id}`, ip);
   904→      return ok({ deleted: true });
   905→    }
   906→  }
   907→
   908→  return fail("NOT_FOUND", "مسیر API یافت نشد", 404);
