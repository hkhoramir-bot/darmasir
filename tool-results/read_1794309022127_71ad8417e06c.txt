   391→/* derive stations: merge same-id entries across lines (interchanges are separate ids) */
   392→function hash(s: string): number {
   393→  let h = 0;
   394→  for (let i = 0; i < s.length; i++) h = (h * 31 + s.charCodeAt(i)) >>> 0;
   395→  return h;
   396→}
   397→
   398→const AMENITY_POOL: AmenityKey[] = [
   399→  "escalator", "elevator", "wc", "atm", "bus", "taxi", "parking", "baby", "bike", "wifi",
   400→];
   401→
   402→export const STATIONS: StationDef[] = RAW.map((r) => {
   403→  const h = hash(r.id);
   404→  const pool = AMENITY_POOL.filter((_, i) => (h >> i) % 3 !== 0);
   405→  const amenities = r.amenities ?? (pool.length >= 3 ? pool.slice(0, 3 + (h % 3)) : ["escalator" as AmenityKey]);
   406→  return {
   407→    id: r.id,
   408→    name: r.name,
   409→    cityId: LINES.find((l) => l.id === r.line)!.cityId,
   410→    lines: [{ lineId: r.line, order: r.order }],
   411→    amenities,

   474→export function nextArrivals(lineId: string, stationId: string, now = Date.now()): number[] {
   475→  const line = getLine(lineId);
   476→  const headwaySec = Math.round(line.headway * 60);
   477→  const phase = hash(stationId + lineId + new Date(now).getHours()) % headwaySec;
   478→  const elapsed = Math.floor(now / 1000) % headwaySec;
   479→  let next = phase - elapsed;
   480→  while (next <= 0) next += headwaySec;
   481→  const late = line.status === "delay" ? 8 * 60 : 0;
   482→  return [next + late, next + headwaySec + late];
   483→}

   485→export type CrowdLevel = "quiet" | "moderate" | "busy";

   487→export function crowdingNow(lineId: string, now = new Date()): CrowdLevel {
   488→  const h = now.getHours();
   489→  const peak = (h >= 7 && h <= 9) || (h >= 16.5 && h <= 19);
   490→  const mid = (h >= 10 && h <= 16) || (h >= 19.5 && h <= 22);
   491→  const line = getLine(lineId);
   492→  if (line.status === "crowded") return peak ? "busy" : mid ? "busy" : "moderate";
   493→  if (peak) return hash(lineId + h) % 4 === 0 ? "quiet" : "busy";
   494→  if (mid) return hash(lineId + h) % 3 === 0 ? "busy" : "moderate";
   495→  return hash(lineId + h) % 3 === 0 ? "moderate" : "quiet";
   496→}

   680→export interface RoutePlan {
   681→  legs: RouteLeg[];

   683→  totalStations: number;

   691→const TRANSFER_MIN = 4.5;
   692→const BOARDING_MIN = 1.5;
   693→
   694→export function fareFor(stops: number): number {
   695→  if (stops <= 6) return 2800;
   696→  if (stops <= 12) return 3200;
   697→  if (stops <= 20) return 3800;
   698→  return 4500;
   699→}
   700→
   701→interface State {
   702→  station: string;
   703→  line: string | null;
   704→  time: number;
   705→  transfers: number;
   706→  path: { station: string; line: string | null }[];
   707→}
   708→
   709→/** Dijkstra over (station, line) states. */
   710→export function computeRoute(
   711→  fromId: string,
   712→  toId: string,
   713→  pref: "fastest" | "least_transfers" | "quietest" = "fastest",
   714→  now = new Date()
   715→): RoutePlan | null {
   716→  const from = getStation(fromId);
   717→  const to = getStation(toId);
   718→  if (!from || !to || fromId === toId) return null;
   719→
   720→  const seg = (lineId: string) => {
   721→    const base = getLine(lineId).segmentMinutes;
   722→    const crowd = crowdingNow(lineId, now);
   723→    const factor = pref === "quietest" ? (crowd === "busy" ? 1.35 : crowd === "moderate" ? 1.1 : 0.95) : 1;
   724→    return base * factor;
   725→  };
   726→
   727→  const start: State = { station: fromId, line: null, time: 0, transfers: 0, path: [{ station: fromId, line: null }] };
   728→  const best = new Map<string, number>();
   729→  const key = (s: State) => `${s.station}|${s.line ?? "*"}`;
   730→  const queue: State[] = [start];
   731→  best.set(key(start), 0);
   732→
   733→  let done: State | null = null;
   734→  const visited = new Set<string>();
   735→
   736→  while (queue.length) {
   737→    // pick min
   738→    let idx = 0;
   739→    for (let i = 1; i < queue.length; i++) {
   740→      if (queue[i].time < queue[idx].time) idx = i;
   741→    }
   742→    const cur = queue.splice(idx, 1)[0];
   743→    if (visited.has(key(cur))) continue;
   744→    visited.add(key(cur));
   745→
   746→    if (cur.station === toId) {
   747→      done = cur;
   748→      break;
   749→    }
   750→
   751→    const st = getStation(cur.station)!;
   752→    for (const ls of st.lines) {
   753→      // board / continue on line ls.lineId
   754→      const isTransfer = cur.line !== null && cur.line !== ls.lineId;
   755→      const penalty = cur.line === null ? BOARDING_MIN : isTransfer ? TRANSFER_MIN : 0;
   756→      const transferCost =
   757→        pref === "least_transfers" ? (isTransfer ? 40 : 0) : 0;
   758→
   759→      // neighbors along the line
   760→      const list = stationsOfLine(ls.lineId);
   761→      const myIdx = list.findIndex((x) => x.id === cur.station);
   762→      for (const delta of [-1, 1]) {
   763→        const nIdx = myIdx + delta;
   764→        if (nIdx < 0 || nIdx >= list.length) continue;
   765→        const nb = list[nIdx];
   766→        const t = cur.time + seg(ls.lineId) + penalty + transferCost;
   767→        const ns: State = {
   768→          station: nb.id,
   769→          line: ls.lineId,
   770→          time: t,
   771→          transfers: cur.transfers + (isTransfer ? 1 : 0),
   772→          path: [...cur.path, { station: nb.id, line: ls.lineId }],
   773→        };
   774→        const k = key(ns);
   775→        if (!best.has(k) || t < best.get(k)! - 1e-9) {
   776→          best.set(k, t);
   777→          queue.push(ns);
   778→        }
   779→      }
   780→    }
   781→  }
   782→
   783→  if (!done) return null;
   784→
   785→  // compress path into legs
   786→  const legs: RouteLeg[] = [];
   787→  const path = done.path.slice(1);
   788→  let curLine: string | null = null;
   789→  let legStart = path[0].station;
   790→  let prev = fromId;
   791→
   792→  const flush = (endStation: string) => {
   793→    if (!curLine) return;
   794→    const line = getLine(curLine);
   795→    const ids = stationsOfLine(curLine).map((s) => s.id);
   796→    const i1 = ids.indexOf(legStart);
   797→    const i2 = ids.indexOf(endStation);
   798→    const slice = i1 <= i2 ? ids.slice(i1, i2 + 1) : ids.slice(i2, i1 + 1).reverse();
   799→    const names = slice.map((id) => getStation(id)!.name);
   800→    legs.push({
   801→      lineId: curLine,
   802→      lineName: line.name,
   803→      color: line.color,
   804→      textOnColor: line.textOnColor,
   805→      fromId: slice[0],
   806→      fromName: names[0],
   807→      toId: slice[slice.length - 1],
   808→      toName: names[names.length - 1],
   809→      stationIds: slice,
   810→      stationNames: names,
   811→      stops: slice.length - 1,
   812→      minutes: Math.max(1, Math.round((slice.length - 1) * line.segmentMinutes)),
   813→    });
   814→  };
   815→
   816→  for (const step of path) {
   817→    if (curLine === null) {
   818→      curLine = step.line;
   819→      legStart = prev;
   820→    } else if (step.line !== curLine) {
   821→      flush(prev);
   822→      curLine = step.line;
   823→      legStart = prev;
   824→    }
   825→    prev = step.station;
   826→  }
   827→  flush(prev);
   828→
   829→  const totalStops = legs.reduce((a, l) => a + l.stops, 0);
   830→  // زمان نمایشی واقعی = جمع زمان لگ‌ها + هزینه تعویض خط + سوار شدن (بدون وزن بهینه‌سازی)
   831→  const totalMinutes = Math.round(
   832→    legs.reduce((a, l) => a + l.minutes, 0) + (legs.length - 1) * TRANSFER_MIN + BOARDING_MIN
   833→  );
   834→  const transferStations = legs.slice(0, -1).map((l) => l.toName);
   835→  const busyLeg = legs.find((l) => crowdingNow(l.lineId, now) === "busy");
   836→
   837→  return {
   838→    legs,
   839→    totalMinutes,
   840→    totalStations: totalStops + 1,
   841→    transfers: legs.length - 1,
   842→    transferStations,
   843→    fare: fareFor(totalStops),
   844→    pref,
   845→    crowdNote: busyLeg
   846→      ? `ازدحام بالا روی ${busyLeg.lineName} در ساعات اوج`

   857→export function searchStations(opts: {
   858→  q?: string;
   859→  lineId?: string;
   860→  facility?: AmenityKey;
   861→  cityId?: CityId;
   862→}): StationDef[] {
   863→  const q = opts.q?.trim();
   864→  let res = STATIONS;
   865→  if (opts.cityId) res = res.filter((s) => s.cityId === opts.cityId);
   866→  if (opts.lineId) res = res.filter((s) => s.lines.some((l) => l.lineId === opts.lineId));
   867→  if (opts.facility) res = res.filter((s) => s.amenities.includes(opts.facility!));
   868→  if (q) {
   869→    const nq = toEnDigits(q).toLowerCase();
   870→    res = res.filter(
   871→      (s) => s.name.toLowerCase().includes(nq) || s.name.replace(/ /g, "").includes(nq.replace(/ /g, ""))
   872→    );
   873→  }
   874→  return res;
   875→}
