    24→  headway: number; // minutes
    25→  firstTrain: string; // 05:30
    26→  lastTrain: string; // 23:00

    28→  badges?: string[]; // e.g. فرودگاه

   380→/* derive stations: merge same-id entries across lines (interchanges are separate ids) */
   381→function hash(s: string): number {
   382→  let h = 0;
   383→  for (let i = 0; i < s.length; i++) h = (h * 31 + s.charCodeAt(i)) >>> 0;
   384→  return h;
   385→}
   386→
   387→const AMENITY_POOL: AmenityKey[] = [
   388→  "escalator", "elevator", "wc", "atm", "bus", "taxi", "parking", "baby", "bike", "wifi",
   389→];
   390→
   391→export const STATIONS: StationDef[] = RAW.map((r) => {
   392→  const h = hash(r.id);
   393→  const pool = AMENITY_POOL.filter((_, i) => (h >> i) % 3 !== 0);
   394→  const amenities = r.amenities ?? (pool.length >= 3 ? pool.slice(0, 3 + (h % 3)) : ["escalator" as AmenityKey]);
   395→  return {
   396→    id: r.id,
   397→    name: r.name,
   398→    cityId: LINES.find((l) => l.id === r.line)!.cityId,
   399→    lines: [{ lineId: r.line, order: r.order }],
   400→    amenities,
   401→    featured: r.featured,
   402→  };
   403→}).reduce<StationDef[]>((acc, s) => {
   404→  const found = acc.find((x) => x.id === s.id);
   405→  if (found) found.lines.push(...s.lines);
   406→  else acc.push(s);
   407→  return acc;
   408→}, []);

   412→export function getCity(id: CityId): City {

   415→export function getLine(id: string): LineDef {
   416→  return LINES.find((l) => l.id === id) ?? LINES[0];
   417→}
   418→export function getStation(id: string): StationDef | undefined {
   419→  return STATIONS.find((s) => s.id === id);
   420→}
   421→export function stationsOfLine(lineId: string): StationDef[] {
   422→  const line = getLine(lineId);
   423→  return STATIONS.filter((s) => s.lines.some((l) => l.lineId === lineId)).sort(
   424→    (a, b) =>
   425→      a.lines.find((l) => l.lineId === lineId)!.order -
   426→      b.lines.find((l) => l.lineId === lineId)!.order
   427→  );
   428→}

   455→export function nextArrivals(lineId: string, stationId: string, now = Date.now()): number[] {
   456→  const line = getLine(lineId);
   457→  const headwaySec = Math.round(line.headway * 60);
   458→  const phase = hash(stationId + lineId + new Date(now).getHours()) % headwaySec;
   459→  const elapsed = Math.floor(now / 1000) % headwaySec;
   460→  let next = phase - elapsed;
   461→  while (next <= 0) next += headwaySec;
   462→  const late = line.status === "delay" ? 8 * 60 : 0;
   463→  return [next + late, next + headwaySec + late];
   464→}

   466→export type CrowdLevel = "quiet" | "moderate" | "busy";

   468→export function crowdingNow(lineId: string, now = new Date()): CrowdLevel {
   469→  const h = now.getHours();
   470→  const peak = (h >= 7 && h <= 9) || (h >= 16.5 && h <= 19);
   471→  const mid = (h >= 10 && h <= 16) || (h >= 19.5 && h <= 22);
   472→  const line = getLine(lineId);
   473→  if (line.status === "crowded") return peak ? "busy" : mid ? "busy" : "moderate";
   474→  if (peak) return hash(lineId + h) % 4 === 0 ? "quiet" : "busy";
   475→  if (mid) return hash(lineId + h) % 3 === 0 ? "busy" : "moderate";
   476→  return hash(lineId + h) % 3 === 0 ? "moderate" : "quiet";
   477→}

   673→const BOARDING_MIN = 1.5;

   675→export function fareFor(stops: number): number {
   676→  if (stops <= 6) return 2800;
   677→  if (stops <= 12) return 3200;
   678→  if (stops <= 20) return 3800;
   679→  return 4500;

   682→interface State {
   683→  station: string;
   684→  line: string | null;
   685→  time: number;
   686→  transfers: number;
   687→  path: { station: string; line: string | null }[];
   688→}
   689→
   690→/** Dijkstra over (station, line) states. */
   691→export function computeRoute(
   692→  fromId: string,
   693→  toId: string,
   694→  pref: "fastest" | "least_transfers" | "quietest" = "fastest",
   695→  now = new Date()
   696→): RoutePlan | null {
   697→  const from = getStation(fromId);
   698→  const to = getStation(toId);
   699→  if (!from || !to || fromId === toId) return null;
   700→
   701→  const seg = (lineId: string) => {
   702→    const base = getLine(lineId).segmentMinutes;
   703→    const crowd = crowdingNow(lineId, now);
   704→    const factor = pref === "quietest" ? (crowd === "busy" ? 1.35 : crowd === "moderate" ? 1.1 : 0.95) : 1;
   705→    return base * factor;
   706→  };
   707→
   708→  const start: State = { station: fromId, line: null, time: 0, transfers: 0, path: [{ station: fromId, line: null }] };
   709→  const best = new Map<string, number>();
   710→  const key = (s: State) => `${s.station}|${s.line ?? "*"}`;
   711→  const queue: State[] = [start];
   712→  best.set(key(start), 0);
   713→
   714→  let done: State | null = null;
   715→  const visited = new Set<string>();
   716→
   717→  while (queue.length) {
   718→    // pick min
   719→    let idx = 0;
   720→    for (let i = 1; i < queue.length; i++) {
   721→      if (queue[i].time < queue[idx].time) idx = i;
   722→    }
   723→    const cur = queue.splice(idx, 1)[0];
   724→    if (visited.has(key(cur))) continue;
   725→    visited.add(key(cur));
   726→
   727→    if (cur.station === toId) {
   728→      done = cur;
   729→      break;
   730→    }
   731→
   732→    const st = getStation(cur.station)!;
   733→    for (const ls of st.lines) {
   734→      // board / continue on line ls.lineId
   735→      const isTransfer = cur.line !== null && cur.line !== ls.lineId;
   736→      const penalty = cur.line === null ? BOARDING_MIN : isTransfer ? TRANSFER_MIN : 0;
   737→      const transferCost =
   738→        pref === "least_transfers" ? (isTransfer ? 40 : 0) : 0;
   739→
   740→      // neighbors along the line
   741→      const list = stationsOfLine(ls.lineId);
   742→      const myIdx = list.findIndex((x) => x.id === cur.station);
   743→      for (const delta of [-1, 1]) {
   744→        const nIdx = myIdx + delta;
   745→        if (nIdx < 0 || nIdx >= list.length) continue;
   746→        const nb = list[nIdx];
   747→        const t = cur.time + seg(ls.lineId) + penalty + transferCost;
   748→        const ns: State = {
   749→          station: nb.id,
   750→          line: ls.lineId,
   751→          time: t,
   752→          transfers: cur.transfers + (isTransfer ? 1 : 0),
   753→          path: [...cur.path, { station: nb.id, line: ls.lineId }],
   754→        };
   755→        const k = key(ns);
   756→        if (!best.has(k) || t < best.get(k)! - 1e-9) {
   757→          best.set(k, t);
   758→          queue.push(ns);
   759→        }
   760→      }
   761→    }
   762→  }
   763→
   764→  if (!done) return null;
   765→
   766→  // compress path into legs
   767→  const legs: RouteLeg[] = [];
   768→  const path = done.path.slice(1);
   769→  let curLine: string | null = null;
   770→  let legStart = path[0].station;
   771→  let prev = fromId;
   772→
   773→  const flush = (endStation: string) => {
   774→    if (!curLine) return;
   775→    const line = getLine(curLine);
   776→    const ids = stationsOfLine(curLine).map((s) => s.id);
   777→    const i1 = ids.indexOf(legStart);
   778→    const i2 = ids.indexOf(endStation);
   779→    const slice = i1 <= i2 ? ids.slice(i1, i2 + 1) : ids.slice(i2, i1 + 1).reverse();
   780→    const names = slice.map((id) => getStation(id)!.name);
   781→    legs.push({
   782→      lineId: curLine,
   783→      lineName: line.name,
   784→      color: line.color,
   785→      textOnColor: line.textOnColor,
   786→      fromId: slice[0],
   787→      fromName: names[0],
   788→      toId: slice[slice.length - 1],
   789→      toName: names[names.length - 1],
   790→      stationIds: slice,
   791→      stationNames: names,
   792→      stops: slice.length - 1,
   793→      minutes: Math.round(slice.length * line.segmentMinutes),
   794→    });
   795→  };
   796→
   797→  for (const step of path) {
   798→    if (curLine === null) {
   799→      curLine = step.line;
   800→      legStart = prev;
   801→    } else if (step.line !== curLine) {
   802→      flush(prev);
   803→      curLine = step.line;
   804→      legStart = prev;
   805→    }
   806→    prev = step.station;
   807→  }
   808→  flush(prev);
   809→
   810→  const totalStops = legs.reduce((a, l) => a + l.stops, 0);
   811→  const totalMinutes = Math.round(done.time);
   812→  const transferStations = legs.slice(0, -1).map((l, i) => l.toId);
   813→  const busyLeg = legs.find((l) => crowdingNow(l.lineId, now) === "busy");
   814→
   815→  return {
   816→    legs,
   817→    totalMinutes,
   818→    totalStations: totalStops + 1,
   819→    transfers: legs.length - 1,
   820→    transferStations,
   821→    fare: fareFor(totalStops),
   822→    pref,
   823→    crowdNote: busyLeg
   824→      ? `ازدحام بالا روی ${busyLeg.lineName} در ساعات اوج`
   825→      : "سرویس‌دهی همه خطوط مسیر عادی و روان است",
   826→  };
   827→}
