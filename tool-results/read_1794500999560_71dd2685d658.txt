   544→   544→function hash(s: string): number {
   545→   545→  let h = 0;
   546→   546→  for (let i = 0; i < s.length; i++) h = (h * 31 + s.charCodeAt(i)) >>> 0;
   547→   547→  return h;
   548→   548→}

   550→   550→const AMENITY_POOL: AmenityKey[] = [
   551→   551→  "escalator", "elevator", "wc", "atm", "bus", "taxi", "parking", "baby", "bike", "wifi",
   552→   552→];
   553→   553→
   554→   554→export const STATIONS: StationDef[] = RAW.map((r) => {
   555→   555→  const h = hash(r.id);
   556→   556→  const pool = AMENITY_POOL.filter((_, i) => (h >> i) % 3 !== 0);
   557→   557→  const amenities = r.amenities ?? (pool.length >= 3 ? pool.slice(0, 3 + (h % 3)) : ["escalator" as AmenityKey]);
   558→   558→  return {
   559→   559→    id: r.id,
   560→   560→    name: r.name,
   561→   561→    cityId: LINES.find((l) => l.id === r.line)!.cityId,
   562→   562→    lines: [{ lineId: r.line, order: r.order }],
   563→   563→    amenities,
   564→   564→    featured: r.featured,
   565→   565→  };
   566→   566→}).reduce<StationDef[]>((acc, s) => {
   567→   567→  const found = acc.find((x) => x.id === s.id);
   568→   568→  if (found) found.lines.push(...s.lines);
   569→   569→  else acc.push(s);
   570→   570→  return acc;

   605→   605→  return String(n).replace(/\d/g, (d) => FA_DIGITS[Number(d)]);
   606→   606→}
   607→   607→export function toEnDigits(s: string): string {
   608→   608→  return s
   609→   609→    .replace(/[۰-۹]/g, (d) => String(FA_DIGITS.indexOf(d)))
   610→   610→    .replace(/[٠-٩]/g, (d) => String("٠١٢٣٤٥٦٧٨٩".indexOf(d)));
   611→   611→}
   612→   612→export function faMoney(n: number): string {
   613→   613→  return toFa(n.toLocaleString("en-US"));
   614→   614→}

   616→   616→/* ------------------------------ موتور جدول حرکت رسمی (۱۴۰۴) ------------------------------ */
   617→   617→/**
   618→   618→ * مترو تهران فید لحظه‌ای عمومی ندارد؛ زمان ورود قطارها بر اساس
   619→   619→ * جدول حرکت رسمی شرکت مترو تهران (سامانه metro.tehran.ir) محاسبه می‌شود:
   620→   620→ * ساعات سرویس: شنبه تا چهارشنبه ۵:۳۰ تا ۲۲:۳۰ — پنجشنبه ۶:۰۰ — جمعه ۶:۳۰
   621→   621→ * و سرفاصله اوج/غیراوج رسمی هر خط.
   622→   622→ */
   623→   623→
   624→   624→/** بازه‌های ساعات اوج (دقیقه از نیمه‌شب) — صبح و عصر */
   625→   625→const PEAK_WINDOWS: [number, number][] = [
   626→   626→  [420, 570], // ۷:۰۰ تا ۹:۳۰
   627→   627→  [990, 1140], // ۱۶:۳۰ تا ۱۹:۰۰
   628→   628→];
   629→   629→
   630→   630→/** فاصله حرکت قطارها در یک لحظه از روز — اوج / غیراوج */
   631→   631→export function headwayAt(line: LineDef, mins: number): number {
   632→   632→  const peak = PEAK_WINDOWS.some(([a, b]) => mins >= a && mins < b);
   633→   633→  return peak ? (line.headwayPeak ?? line.headway) : (line.headwayOff ?? line.headway);
   634→   634→}
   635→   635→
   636→   636→/**
   637→   637→ * ساعت شروع سرویس بر اساس روز هفته (الگوی رسمی ۱۴۰۴):
   638→   638→ * شنبه تا چهارشنبه: ۵:۳۰ | پنجشنبه: ۶:۰۰ | جمعه: ۶:۳۰
   639→   639→ */
   640→   640→export function serviceStartMin(line: LineDef, d = new Date()): number {
   641→   641→  const [fh, fm] = line.firstTrain.split(":").map(Number);
   642→   642→  let start = fh * 60 + fm;
   643→   643→  const day = d.getDay(); // ۰ یکشنبه … ۳ چهارشنبه، ۴ پنجشنبه، ۵ جمعه، ۶ شنبه
   644→   644→  if (day === 4) start += 30; // پنجشنبه ۶:۰۰
   645→   645→  else if (day === 5) start += 60; // جمعه ۶:۳۰
   646→   646→  return start;
   647→   647→}
   648→   648→
   649→   649→/** ساعت حرکت آخرین قطار از سرخط */
   650→   650→export function serviceEndMin(line: LineDef): number {
   651→   651→  const [lh, lm] = line.lastTrain.split(":").map(Number);
   652→   652→  return lh * 60 + lm;
   653→   653→}
   654→   654→
   655→   655→/** آیا الان داخل ساعات سرویس‌دهی خط هستیم؟ */
   656→   656→export function isWithinService(line: { firstTrain: string; lastTrain: string }, d = new Date()): boolean {
   657→   657→  const mins = d.getHours() * 60 + d.getMinutes();
   658→   658→  return mins >= serviceStartMin(line as LineDef, d) && mins <= serviceEndMin(line as LineDef);
   659→   659→}
   660→   660→
   661→   661→/** زمان‌های حرکت قطارها از سرخط با فاصله اوج/غیراوج */
   662→   662→function departureGrid(line: LineDef, fromMin: number, toMin: number): number[] {
   663→   663→  const out: number[] = [];
   664→   664→  let t = fromMin;
   665→   665→  while (t <= toMin + 1e-9) {
   666→   666→    out.push(t);
   667→   667→    t += headwayAt(line, t);
   668→   668→  }
   669→   669→  return out;
   670→   670→}
   671→   671→
   672→   672→const fmtClock = (m: number) => {
   673→   673→  const mm = Math.round(m) % 1440;
   674→   674→  return `${String(Math.floor(mm / 60)).padStart(2, "0")}:${String(mm % 60).padStart(2, "0")}`;
   675→   675→};

   677→   677→export interface ArrivalDirDetail {
   678→   678→  /** دقایق تا قطار بعدی و پس از آن */
   679→   679→  minutes: number[];
   680→   680→  /** ساعت رسیدن دو قطار بعدی (HH:MM) */
   681→   681→  at: string[];
   682→   682→  /** خارج از ساعات سرویس */
   683→   683→  outOfService: boolean;
   684→   684→}
   685→   685→
   686→   686→/** دو قطار بعدی یک ایستگاه در یک جهت مشخص — محاسبه قطعی از جدول حرکت */
   687→   687→function scheduleArrivalsDir(
   688→   688→  lineId: string,
   689→   689→  stationId: string,
   690→   690→  dir: 0 | 1,
   691→   691→  now: number
   692→   692→): ArrivalDirDetail {
   693→   693→  const line = getLine(lineId);
   694→   694→  const list = stationsOfLine(lineId);
   695→   695→  const idx = list.findIndex((s) => s.id === stationId);
   696→   696→  if (idx < 0) return { minutes: [], at: [], outOfService: true };
   697→   697→
   698→   698→  const d = new Date(now);
   699→   699→  const nowMin = d.getHours() * 60 + d.getMinutes() + d.getSeconds() / 60;
   700→   700→  const start = serviceStartMin(line, d);
   701→   701→  const end = serviceEndMin(line);
   702→   702→  // تا ۴۵ دقیقه پس از پایان، رسیدن آخرین قطارها به ایستگاه‌های انتهایی ادامه دارد
   703→   703→  if (nowMin < start || nowMin > end + 45) {
   704→   704→    return { minutes: [], at: [], outOfService: true };
   705→   705→  }
   706→   706→
   707→   707→  const hops = dir === 0 ? idx : list.length - 1 - idx;
   708→   708→  const travel = hops * line.segmentMinutes;
   709→   709→  // ۴۰ ثانیه توقف در ایستگاه برای سوار و پیاده شدن
   710→   710→  const deps = departureGrid(line, start, end).filter((dep) => dep + travel >= nowMin + 0.6);
   711→   711→  const next2 = deps.slice(0, 2);
   712→   712→  return {
   713→   713→    minutes: next2.map((dep) => Math.max(1, Math.round(dep + travel - nowMin))),
   714→   714→    at: next2.map((dep) => fmtClock(dep + travel)),
   715→   715→    outOfService: false,
   716→   716→  };
   717→   717→}
   718→   718→
   719→   719→/** ورودهای بعدی به سمت terminals[1] — سازگاری با نسخه قبل */
   720→   720→export function nextArrivals(lineId: string, stationId: string, now = Date.now()): number[] {
   721→   721→  return scheduleArrivalsDir(lineId, stationId, 0, now).minutes;
   722→   722→}

   724→   724→export interface DirectionalArrival {
   725→   725→  /** نام ایستگاه مقصد قطار در این جهت (مثل تجریش) */
   726→   726→  to: string;

   729→   729→  /** ساعت رسیدن دو قطار بعدی (HH:MM) */
   730→   730→  at: string[];
   731→   731→  /** خارج از ساعات سرویس */
   732→   732→  outOfService: boolean;
   733→   733→  crowd: CrowdLevel;
   734→   734→  crowdLabel: string;
   735→   735→}

   737→   737→/**
   738→   738→ * ورودهای بعدی به هر دو جهت — با نام درستِ مقصد هر جهت و بر اساس جدول حرکت رسمی.
   739→   739→ * ایستگاه سرخط فقط یک جهت دارد؛ ایستگاه میانی هر دو جهت.
   740→   740→ */
   741→   741→export function nextArrivalsBoth(
   742→   742→  lineId: string,
   743→   743→  stationId: string,
   744→   744→  now = Date.now()
   745→   745→): DirectionalArrival[] {
   746→   746→  const line = getLine(lineId);
   747→   747→  const list = stationsOfLine(lineId);
   748→   748→  const idx = list.findIndex((s) => s.id === stationId);
   749→   749→  if (idx < 0) return [];
   750→   750→
   751→   751→  const out: DirectionalArrival[] = [];
   752→   752→  const mk = (to: string, dir: 0 | 1): DirectionalArrival => {
   753→   753→    const r = scheduleArrivalsDir(lineId, stationId, dir, now);
   754→   754→    const crowd = crowdingNow(lineId, new Date(now));
   755→   755→    return {
   756→   756→      to,
   757→   757→      minutes: r.minutes,
   758→   758→      at: r.at,
   759→   759→      outOfService: r.outOfService,
   760→   760→      crowd,
   761→   761→      crowdLabel: CROWD_LABEL[crowd],
   762→   762→    };
   763→   763→  };
   764→   764→
   765→   765→  // به سمت terminals[1] — مگر این ایستگاه خودش همان سرخط باشد
   766→   766→  if (idx < list.length - 1) out.push(mk(line.terminals[1], 0));
   767→   767→  // به سمت terminals[0] — مگر این ایستگاه خودش همان سرخط باشد
   768→   768→  if (idx > 0) out.push(mk(line.terminals[0], 1));
   769→   769→  return out;
   770→   770→}

   772→   772→export type CrowdLevel = "quiet" | "moderate" | "busy";
   773→   773→
   774→   774→export function crowdingNow(lineId: string, now = new Date()): CrowdLevel {
   775→   775→  const h = now.getHours();
   776→   776→  const peak = (h >= 7 && h <= 9) || (h >= 16.5 && h <= 19);
   777→   777→  const mid = (h >= 10 && h <= 16) || (h >= 19.5 && h <= 22);
   778→   778→  const line = getLine(lineId);
   779→   779→  if (line.status === "crowded") return peak ? "busy" : mid ? "busy" : "moderate";
   780→   780→  if (peak) return hash(lineId + h) % 4 === 0 ? "quiet" : "busy";
   781→   781→  if (mid) return hash(lineId + h) % 3 === 0 ? "busy" : "moderate";
   782→   782→  return hash(lineId + h) % 3 === 0 ? "moderate" : "quiet";
   783→   783→}

  1032→  1032→/** Dijkstra over (station, line) states. */
  1033→  1033→export function computeRoute(
  1034→  1034→  fromId: string,
  1035→  1035→  toId: string,
  1036→  1036→  pref: "fastest" | "least_transfers" | "quietest" = "fastest",
  1037→  1037→  now = new Date()
  1038→  1038→): RoutePlan | null {
  1039→  1039→  const from = getStation(fromId);
  1040→  1040→  const to = getStation(toId);
  1041→  1041→  if (!from || !to || fromId === toId) return null;
  1042→  1042→
  1043→  1043→  const seg = (lineId: string) => {
  1044→  1044→    const base = getLine(lineId).segmentMinutes;
  1045→  1045→    const crowd = crowdingNow(lineId, now);
  1046→  1046→    const factor = pref === "quietest" ? (crowd === "busy" ? 1.35 : crowd === "moderate" ? 1.1 : 0.95) : 1;
  1047→  1047→    return base * factor;
  1048→  1048→  };
  1049→  1049→
  1050→  1050→  const start: State = { station: fromId, line: null, time: 0, transfers: 0, path: [{ station: fromId, line: null }] };
  1051→  1051→  const best = new Map<string, number>();
  1052→  1052→  const key = (s: State) => `${s.station}|${s.line ?? "*"}`;
  1053→  1053→  const queue: State[] = [start];
  1054→  1054→  best.set(key(start), 0);
  1055→  1055→
  1056→  1056→  let done: State | null = null;
  1057→  1057→  const visited = new Set<string>();
  1058→  1058→
  1059→  1059→  while (queue.length) {
  1060→  1060→    // pick min
  1061→  1061→    let idx = 0;
  1062→  1062→    for (let i = 1; i < queue.length; i++) {
  1063→  1063→      if (queue[i].time < queue[idx].time) idx = i;
  1064→  1064→    }
  1065→  1065→    const cur = queue.splice(idx, 1)[0];
  1066→  1066→    if (visited.has(key(cur))) continue;
  1067→  1067→    visited.add(key(cur));
  1068→  1068→
  1069→  1069→    if (cur.station === toId) {
  1070→  1070→      done = cur;
  1071→  1071→      break;
  1072→  1072→    }
  1073→  1073→
  1074→  1074→    const st = getStation(cur.station)!;
  1075→  1075→    for (const ls of st.lines) {
  1076→  1076→      // board / continue on line ls.lineId
  1077→  1077→      const isTransfer = cur.line !== null && cur.line !== ls.lineId;
  1078→  1078→      const penalty = cur.line === null ? BOARDING_MIN : isTransfer ? TRANSFER_MIN : 0;
  1079→  1079→      const transferCost =
  1080→  1080→        pref === "least_transfers" ? (isTransfer ? 40 : 0) : 0;
  1081→  1081→
  1082→  1082→      // neighbors along the line
  1083→  1083→      const list = stationsOfLine(ls.lineId);
  1084→  1084→      const myIdx = list.findIndex((x) => x.id === cur.station);
  1085→  1085→      for (const delta of [-1, 1]) {
  1086→  1086→        const nIdx = myIdx + delta;
  1087→  1087→        if (nIdx < 0 || nIdx >= list.length) continue;
  1088→  1088→        const nb = list[nIdx];
  1089→  1089→        const t = cur.time + seg(ls.lineId) + penalty + transferCost;
  1090→  1090→        const ns: State = {
  1091→  1091→          station: nb.id,
  1092→  1092→          line: ls.lineId,
  1093→  1093→          time: t,
  1094→  1094→          transfers: cur.transfers + (isTransfer ? 1 : 0),
  1095→  1095→          path: [...cur.path, { station: nb.id, line: ls.lineId }],
  1096→  1096→        };
  1097→  1097→        const k = key(ns);
  1098→  1098→        if (!best.has(k) || t < best.get(k)! - 1e-9) {
  1099→  1099→          best.set(k, t);
  1100→  1100→          queue.push(ns);
  1101→  1101→        }
  1102→  1102→      }
  1103→  1103→    }
  1104→  1104→  }
  1105→  1105→
  1106→  1106→  if (!done) return null;
  1107→  1107→
  1108→  1108→  // compress path into legs
  1109→  1109→  const legs: RouteLeg[] = [];
  1110→  1110→  const path = done.path.slice(1);
  1111→  1111→  let curLine: string | null = null;
  1112→  1112→  let legStart = path[0].station;
  1113→  1113→  let prev = fromId;
  1114→  1114→
  1115→  1115→  const flush = (endStation: string) => {
  1116→  1116→    if (!curLine) return;
  1117→  1117→    const line = getLine(curLine);
  1118→  1118→    const ids = stationsOfLine(curLine).map((s) => s.id);
  1119→  1119→    const i1 = ids.indexOf(legStart);
  1120→  1120→    const i2 = ids.indexOf(endStation);
  1121→  1121→    const slice = i1 <= i2 ? ids.slice(i1, i2 + 1) : ids.slice(i2, i1 + 1).reverse();
  1122→  1122→    const names = slice.map((id) => getStation(id)!.name);
  1123→  1123→    legs.push({
  1124→  1124→      lineId: curLine,
  1125→  1125→      lineName: line.name,
  1126→  1126→      color: line.color,
  1127→  1127→      textOnColor: line.textOnColor,
  1128→  1128→      fromId: slice[0],
  1129→  1129→      fromName: names[0],
  1130→  1130→      toId: slice[slice.length - 1],
  1131→  1131→      toName: names[names.length - 1],
  1132→  1132→      stationIds: slice,
  1133→  1133→      stationNames: names,
  1134→  1134→      stops: slice.length - 1,
  1135→  1135→      minutes: Math.max(1, Math.round((slice.length - 1) * line.segmentMinutes)),
  1136→  1136→    });
  1137→  1137→  };
  1138→  1138→
  1139→  1139→  for (const step of path) {
  1140→  1140→    if (curLine === null) {
  1141→  1141→      curLine = step.line;
  1142→  1142→      legStart = prev;
  1143→  1143→    } else if (step.line !== curLine) {
  1144→  1144→      flush(prev);
  1145→  1145→      curLine = step.line;
  1146→  1146→      legStart = prev;
  1147→  1147→    }
  1148→  1148→    prev = step.station;
  1149→  1149→  }
  1150→  1150→  flush(prev);
  1151→  1151→
  1152→  1152→  const totalStops = legs.reduce((a, l) => a + l.stops, 0);
  1153→  1153→  // زمان نمایشی واقعی = جمع زمان لگ‌ها + هزینه تعویض خط + سوار شدن (بدون وزن بهینه‌سازی)
  1154→  1154→  const totalMinutes = Math.round(
  1155→  1155→    legs.reduce((a, l) => a + l.minutes, 0) + (legs.length - 1) * TRANSFER_MIN + BOARDING_MIN
  1156→  1156→  );
  1157→  1157→  const transferStations = legs.slice(0, -1).map((l) => l.toName);
  1158→  1158→  const busyLeg = legs.find((l) => crowdingNow(l.lineId, now) === "busy");
  1159→  1159→
  1160→  1160→  return {
  1161→  1161→    legs,
  1162→  1162→    totalMinutes,
  1163→  1163→    totalStations: totalStops + 1,
  1164→  1164→    transfers: legs.length - 1,
  1165→  1165→    transferStations,
  1166→  1166→    fare: fareFor(totalStops),
  1167→  1167→    pref,
  1168→  1168→    crowdNote: busyLeg
  1169→  1169→      ? `ازدحام بالا روی ${busyLeg.lineName} در ساعات اوج`
  1170→  1170→      : "سرویس‌دهی همه خطوط مسیر عادی و روان است",
  1171→  1171→  };
  1172→  1172→}
