    71→  return { cls: lineChipCls(lineId), label: `خط ${toFa(getLine(lineId)?.number ?? 0)}` };

   107→function StationField({
   108→  field,
   109→  sel,
   110→  text,
   111→  open,
   112→  onOpen,
   113→  onText,
   114→  onPick,
   115→  onClear,
   116→}: {
   117→  field: Field;
   118→  sel: StationDef | null;
   119→  text: string;
   120→  open: boolean;
   121→  onOpen: (f: Field) => void;
   122→  onText: (f: Field, v: string) => void;
   123→  onPick: (f: Field, s: StationDef) => void;
   124→  onClear: (f: Field) => void;
   125→}) {
   126→  const city = useShell((s) => s.city);
   127→  const isFrom = field === "from";
   128→
   129→  const suggestions = useMemo(() => {
   130→    const q = text.trim();
   131→    if (!q) {
   132→      const featured = searchStations({ cityId: city }).filter((s) => s.featured);
   133→      return (featured.length ? featured : searchStations({ cityId: city })).slice(0, 5);
   134→    }
   135→    const res = searchStations({ q, cityId: city });
   136→    return (res.length ? res : searchStations({ q })).slice(0, 5);
   137→  }, [text, city]);
   138→
   139→  const dotStyle = sel ? { backgroundColor: getLine(sel.lines[0]?.lineId ?? "")?.color } : undefined;
   140→  const chip = sel ? chipFor(sel) : null;
   141→
   142→  /* حالت نمایش (ایستگاه انتخاب‌شده) — دقیقاً مطابق کارت خلاصه طرح _10 */
   143→  if (!open && sel && chip) {
   144→    return (
   145→      <button
   146→        type="button"
   147→        onClick={() => onOpen(field)}
   148→        className="w-full flex items-center gap-2.5 bg-surface-container-low px-3 py-2 rounded-xl border border-outline-variant/20 text-right active:scale-[0.99] transition-all"
   149→        aria-label={isFrom ? `تغییر مبدأ: ${sel.name}` : `تغییر مقصد: ${sel.name}`}
   150→      >
   151→        <span className="w-2.5 h-2.5 rounded-full flex-shrink-0" style={dotStyle} />
   152→        <div className="flex-1 flex items-center justify-between min-w-0 gap-2">
   153→          <span className="text-body-md font-body-md font-medium text-on-surface truncate">{sel.name}</span>
   154→          <span className={`text-label-sm font-label-sm px-2 py-0.5 rounded font-bold flex-shrink-0 ${chip.cls}`}>
   155→            {chip.label}
   156→          </span>
   157→        </div>
   158→      </button>
   159→    );
   160→  }
   161→
   162→  /* حالت ورودی + فهرست پیشنهاد جستجو */
   163→  return (
   164→    <div className="relative flex items-center gap-2.5 bg-surface-container-low px-3 py-2 rounded-xl border border-outline-variant/20">
   165→      <span className="w-2.5 h-2.5 rounded-full bg-outline-variant flex-shrink-0" />
   166→      <input
   167→        type="text"
   168→        value={text}
   169→        onChange={(e) => onText(field, e.target.value)}
   170→        onFocus={() => onOpen(field)}
   171→        autoFocus={open}
   172→        placeholder={isFrom ? "ایستگاه مبدأ..." : "ایستگاه مقصد..."}
   173→        aria-label={isFrom ? "ایستگاه مبدأ" : "ایستگاه مقصد"}
   174→        className="flex-1 min-w-0 bg-transparent text-body-md font-body-md font-medium text-on-surface placeholder:text-outline focus:outline-none border-none p-0"
   175→      />
   176→      {text && (
   177→        <button
   178→          type="button"
   179→          aria-label="پاک کردن"
   180→          onClick={() => onClear(field)}
   181→          className="text-outline hover:text-on-surface p-1 rounded-full hover:bg-surface-container-high transition-colors shrink-0 active:scale-95"
   182→        >
   183→          <MI name="close" size={16} />
   184→        </button>
   185→      )}
   186→      {open && (
   187→        <div className="absolute top-full left-0 right-0 mt-1.5 bg-surface-container-lowest rounded-xl shadow-lg border border-outline-variant/40 py-1 z-[60] text-xs">
   188→          {suggestions.length === 0 && <div className="px-3 py-2 text-outline">ایستگاهی یافت نشد</div>}
   189→          {suggestions.map((s) => {
   190→            const c = chipFor(s);
   191→            return (
   192→              <button
   193→                key={s.id}
   194→                type="button"
   195→                onClick={() => onPick(field, s)}
   196→                className="w-full text-right px-3 py-2 hover:bg-surface-container text-on-surface flex items-center justify-between gap-2 active:bg-surface-container transition-colors"
   197→              >
   198→                <span className="truncate">{s.name}</span>
   199→                <span className={`text-[10px] font-bold px-1.5 py-0.5 rounded shrink-0 ${c.cls}`}>{c.label}</span>
   200→              </button>
   201→            );
   202→          })}
   203→        </div>
   204→      )}
   205→    </div>
   206→  );
   207→}

   326→/* --------------- نشان زنده قطار بعدی + شلوغی سکو هر بخش --------------- */
   327→function LegArrivalPill({ leg }: { leg: RouteLeg }) {
   328→  const [tick, setTick] = useState(0);
   329→  useEffect(() => {
   330→    const t = setInterval(() => setTick((x) => x + 1), 30000);
   331→    return () => clearInterval(t);
   332→  }, []);
   333→
   334→  const info = useMemo(() => {
   335→    const line = getLine(leg.lineId);
   336→    const list = stationsOfLine(leg.lineId);
   337→    const iFrom = list.findIndex((s) => s.id === leg.fromId);
   338→    const iTo = list.findIndex((s) => s.id === leg.toId);
   339→    if (!line || iFrom < 0 || iTo < 0) return null;
   340→    const terminal = line.terminals[iTo >= iFrom ? 1 : 0];
   341→    const dirs = nextArrivalsBoth(leg.lineId, leg.fromId);
   342→    const arr = dirs.find((d) => d.to === terminal) ?? dirs[0];
   343→    if (!arr) return null;
   344→    if (arr.outOfService || arr.minutes.length === 0) {
   345→      const d = new Date();
   346→      const fallback = arr.outOfService ? null : headwayAt(line, d.getHours() * 60 + d.getMinutes());
   347→      return { out: arr.outOfService, minutes: fallback, crowd: arr.crowd, crowdLabel: arr.crowdLabel };
   348→    }
   349→    return { out: false, minutes: arr.minutes[0], crowd: arr.crowd, crowdLabel: arr.crowdLabel };
   350→  }, [leg, tick]);
   351→
   352→  if (!info) return null;
   353→
   354→  return (
   355→    <div className="mt-2.5 bg-surface-container-low rounded-xl p-2.5 px-3 flex items-center justify-between gap-2">
   356→      <div className="flex items-center gap-1.5 text-on-surface-variant min-w-0">
   357→        <span className="inline-flex shrink-0" style={{ color: leg.color }}>
   358→          <MI name="schedule" size={16} />
   359→        </span>
   360→        <span className="text-body-sm font-body-sm shrink-0">قطار بعدی:</span>
   361→        <span className="text-body-sm font-body-sm font-bold text-on-surface truncate">
   362→          {info.out || info.minutes === null ? "خارج از ساعات سرویس" : `${toFa(info.minutes)} دقیقه دیگر`}
   363→        </span>
   364→      </div>
   365→      <div className="flex items-center gap-1 shrink-0">
   366→        <span className="text-label-sm font-label-sm text-outline">شلوغی سکو:</span>
   367→        <span className={`text-label-sm font-label-sm font-bold px-1.5 py-0.5 rounded ${CROWD_CHIP[info.crowd]}`}>
   368→          {info.crowdLabel}
   369→        </span>
   370→      </div>
   371→    </div>
   372→  );
   373→}

   412→/* --------------- گام‌های میانی: آکاردئون ایستگاه‌های عبوری --------------- */
   413→function LegAccordionRow({ leg }: { leg: RouteLeg }) {
   414→  const [open, setOpen] = useState(false);
   415→  const mid = leg.stationIds.slice(1, -1);
   416→  if (mid.length === 0) return null;
   417→  const perStop = leg.stops > 0 ? leg.minutes / leg.stops : 0;
   418→  const midMinutes = Math.round(perStop * (mid.length + 1));
   419→
   420→  const hintFor = (sid: string) => {
   421→    const others = interchangeLinesOf(sid)
   422→      .filter((id) => id !== leg.lineId)
   423→      .map((id) => getLine(id)?.number ?? 0)
   424→      .filter((n) => n > 0)
   425→      .sort((a, b) => a - b);
   426→    return others.length ? `تقاطع خط ${others.map((n) => toFa(n)).join(" و ")}` : null;
   427→  };
   428→
   429→  return (
   430→    <div className="relative flex gap-3 pb-6">
   431→      <div className="flex flex-col items-center flex-shrink-0 w-8">
   432→        <div className="w-1.5 h-full rounded-full" style={{ backgroundColor: leg.color }} />
   433→      </div>
   434→      <div className="flex-1 min-w-0">
   435→        <button
   436→          type="button"
   437→          onClick={() => setOpen((v) => !v)}
   438→          aria-expanded={open}
   439→          className="w-full bg-surface-container/60 hover:bg-surface-container rounded-xl px-3 py-2 flex items-center justify-between gap-2 transition-colors"
   440→        >
   441→          <div className="flex items-center gap-2 min-w-0">
   442→            <span className="w-2 h-2 rounded-full shrink-0" style={{ backgroundColor: leg.color }} />
   443→            <span className="text-body-sm font-body-sm text-on-surface-variant font-medium truncate">
   444→              {toFa(mid.length)} ایستگاه عبوری ({toFa(midMinutes)} دقیقه)
   445→            </span>
   446→          </div>
   447→          <MI
   448→            name="expand_more"

   653→/* ------------------------------ بدنه صفحه ------------------------------ */
   654→function PlanInner() {
   655→  const router = useRouter();
   656→  const searchParams = useSearchParams();
   657→  const { toast } = useToast();
   658→  const announcer = useAnnouncer();
   659→  const city = useShell((s) => s.city);
   660→
   661→  const [starting, setStarting] = useState(false);
   662→  const [openDrop, setOpenDrop] = useState<Field | null>(null);

   664→  const [toText, setToText] = useState("");
   665→  const [fromSel, setFromSel] = useState<StationDef | null>(null);
   666→  const [toSel, setToSel] = useState<StationDef | null>(null);
   667→  const [pref, setPref] = useState<Pref>("fastest");
   668→
   669→  /* اعلام صوتی ایستگاه‌ها — ذخیره در حافظه محلی */
   670→  const [voiceOn, setVoiceOn] = useState(true);
   671→  useEffect(() => {
   672→    try {
   673→      setVoiceOn(localStorage.getItem("metrojo-plan-voice") !== "0");
   674→    } catch {}
   675→  }, []);
   676→  const toggleVoice = () => {
   677→    setVoiceOn((v) => {
   678→      const nv = !v;
   679→      try {
   680→        localStorage.setItem("metrojo-plan-voice", nv ? "1" : "0");
   681→      } catch {}
   682→      return nv;
   683→    });
   684→  };
   685→
   686→  /* بیدارباش مقصد — فقط حالت کلاینت + حافظه محلی */
   687→  const [alarm, setAlarm] = useState<{ on: boolean; minutes: number }>({ on: false, minutes: 2 });
   688→  useEffect(() => {
   689→    try {
   690→      const raw = localStorage.getItem("metrojo-plan-alarm");
   691→      if (raw) {
   692→        const p = JSON.parse(raw) as { on?: boolean; minutes?: number };
   693→        if (typeof p.on === "boolean") setAlarm({ on: p.on, minutes: p.minutes ?? 2 });
   694→      }
   695→    } catch {}
   696→  }, []);
   697→  const setAlarmState = (next: { on: boolean; minutes: number }) => {
   698→    setAlarm(next);
   699→    try {
   700→      localStorage.setItem("metrojo-plan-alarm", JSON.stringify(next));
   701→    } catch {}
   702→  };
   703→
   704→  const qFrom = searchParams.get("from");
   705→  const qTo = searchParams.get("to");
   706→  const qPref = searchParams.get("pref");
   707→
   708→  const fromStation = qFrom ? getStation(qFrom) : undefined;
   709→  const toStation = qTo ? getStation(qTo) : undefined;
   710→  const hasParams = !!(fromStation && toStation && fromStation.id !== toStation.id);
   711→
   712→  /* همگام‌سازی انتخاب‌ها و اولویت از پارامترهای آدرس */
   713→  useEffect(() => {
   714→    if (fromStation) {
   715→      setFromSel(fromStation);
   716→      setFromText(fromStation.name);
   717→    }
   718→    if (toStation) {
   719→      setToSel(toStation);
   720→      setToText(toStation.name);
   721→    }
   722→    setPref(qPref === "least_transfers" || qPref === "quietest" ? qPref : "fastest");
   723→  }, [qFrom, qTo, qPref, fromStation, toStation]);
   724→
   725→  const plan = useMemo(
   726→    () =>
   727→      fromStation && toStation && fromStation.id !== toStation.id
   728→        ? computeRoute(fromStation.id, toStation.id, pref)
   729→        : null,
   730→    [fromStation, toStation, pref]
   731→  );
   732→
   733→  /* یادداشت خروجی‌های مقصد */
   734→  const exitsText = useMemo(() => {
   735→    if (!toStation) return "";
   736→    return exitsFor(toStation.id)
   737→      .slice(0, 2)
   738→      .map((e) => `${e.label}: ${e.landmarks[0] ?? e.direction}`)
   739→      .join(" • ");
   740→  }, [toStation]);
   741→
   742→  /* متن وضعیت خط اول مسیر */
   743→  const firstLine = plan ? getLine(plan.legs[0]?.lineId ?? "") : undefined;
   744→  const lineOk = !firstLine?.status || firstLine.status === "normal";
   745→  const statusText = (() => {
   746→    if (!firstLine) return "وضعیت خطوط";
   747→    if (firstLine.status === "closed") return `وضعیت خطوط: خط ${toFa(firstLine.number)} مسدود است`;
   748→    if (firstLine.status === "delay") return `وضعیت خطوط: ${firstLine.statusNote ?? "تأخیر در سرویس‌دهی"}`;
   749→    if (firstLine.status === "crowded" || crowdingNow(firstLine.id) === "busy")
   750→      return `وضعیت خطوط: ازدحام در خط ${toFa(firstLine.number)}`;
   751→    return `وضعیت خطوط: خط ${toFa(firstLine.number)} فعال و روان`;
   752→  })();
   753→
   754→  /* رفع نام آزاد به ایستگاه (مطابق منطق قبلی صفحه) */
   755→  const resolve = (text: string): StationDef | undefined => {
   756→    const q = text.trim().replace(/[«»"]/g, "");
   757→    if (!q) return undefined;
   758→    const candidates = [...searchStations({ q, cityId: city }), ...searchStations({ q })];
   759→    const exact = candidates.find((s) => s.name === q);
   760→    if (exact) return exact;
   761→    const incl = candidates.find((s) => s.name.includes(q) || q.includes(s.name));
   762→    if (incl) return incl;
   763→    const words = q.split(/[\s()]+/).filter((w) => w.length > 1);
   764→    return candidates.find((s) => {
   765→      const sn = s.name.replace(/[()]/g, " ");
   766→      return words.length > 0 && words.every((w) => sn.includes(w));
   767→    });
   768→  };
   769→
   770→  const onText = (field: Field, v: string) => {
   771→    if (field === "from") {
   772→      setFromText(v);
   773→      setFromSel(null);
   774→    } else {
   775→      setToText(v);
   776→      setToSel(null);
   777→    }
   778→  };
   779→
   780→  const onClearField = (field: Field) => {
   781→    if (field === "from") {
   782→      setFromText("");
   783→      setFromSel(null);
   784→    } else {
   785→      setToText("");
   786→      setToSel(null);
   787→    }
   788→    setOpenDrop(field);
   789→    /* در حالت نتیجه، حذف پارامتر → بازگشت به حالت ورودی با حفظ ایستگاه دیگر */
   790→    if (qFrom && qTo) {
   791→      const params = new URLSearchParams();
   792→      if (field === "from" && qTo) params.set("to", qTo);
   793→      if (field === "to" && qFrom) params.set("from", qFrom);
   794→      params.set("pref", pref);
   795→      router.replace(`/plan?${params.toString()}`, { scroll: false });
   796→    }
   797→  };
   798→
   799→  const onPick = (field: Field, s: StationDef) => {
   800→    if (field === "from") {
   801→      setFromSel(s);
   802→      setFromText(s.name);
   803→    } else {
   804→      setToSel(s);
   805→      setToText(s.name);
   806→    }
   807→    setOpenDrop(null);
   808→    /* در حالت نتیجه، انتخاب بلافاصله مسیر را دوباره محاسبه می‌کند */
   809→    if (qFrom && qTo) {
   810→      const f = field === "from" ? s : fromStation;
   811→      const t = field === "to" ? s : toStation;
   812→      if (f && t && f.id !== t.id) {
   813→        router.replace(`/plan?from=${f.id}&to=${t.id}&pref=${pref}`, { scroll: false });
   814→      }
   815→    }
   816→  };
   817→
   818→  const onSwap = () => {
   819→    if (qFrom && qTo && fromStation && toStation) {
   820→      router.replace(`/plan?from=${toStation.id}&to=${fromStation.id}&pref=${pref}`, { scroll: false });
   821→      return;
   822→    }
   823→    const f = fromSel ?? resolve(fromText);
   824→    const t = toSel ?? resolve(toText);
   825→    setFromSel(t ?? null);
   826→    setFromText(t?.name ?? "");
   827→    setToSel(f ?? null);
   828→    setToText(f?.name ?? "");
   829→  };
   830→
   831→  const onPref = (p: Pref) => {
   832→    setPref(p);
   833→    if (qFrom && qTo) {
   834→      router.replace(`/plan?from=${qFrom}&to=${qTo}&pref=${p}`, { scroll: false });
   835→    }
   836→  };
   837→
   838→  const submit = () => {
   839→    const f = fromSel ?? resolve(fromText);
   840→    const t = toSel ?? resolve(toText);
   841→    if (!f || !t) {
   842→      toast({
   843→        title: "مبدأ و مقصد را انتخاب کنید",
   844→        description: "نام ایستگاه را از فهرست پیشنهاد برگزینید",
   845→        variant: "destructive",
   846→      });
   847→      return;
   848→    }
   849→    if (f.id === t.id) {
   850→      toast({ title: "مبدأ و مقصد یکسان هستند", variant: "destructive" });
   851→      return;
   852→    }
   853→    router.replace(`/plan?from=${f.id}&to=${t.id}&pref=${pref}`);
   854→  };
   855→
   856→  const startTrip = async () => {
   857→    if (!plan || !fromStation || !toStation || starting) return;
   858→    setStarting(true);
   859→    try {
   860→      const res = await fetch("/api/history", {
   861→        method: "POST",
   862→        headers: { "Content-Type": "application/json" },
   863→        body: JSON.stringify({
   864→          fromStationId: fromStation.id,
   865→          toStationId: toStation.id,
   866→          pref: plan.pref,
   867→        }),
   868→      });
   869→      if (res.ok) {
   870→        toast({
   871→          title: "سفر در تاریخچه ثبت شد",
   872→          description: `${fromStation.name} → ${toStation.name}`,
   873→        });
   874→        announcer
   875→          .announce(`ناوبری آغاز شد. مدت سفر حدود ${toFa(Math.round(plan.totalMinutes))} دقیقه`)
   876→          .catch(() => {});
   877→      }
   878→    } catch {
   879→      toast({
   880→        title: "ثبت سفر ناموفق بود",
   881→        description: "به ناوبری روی نقشه ادامه می‌دهیم",
   882→        variant: "destructive",
   883→      });
   884→    } finally {
   885→      setStarting(false);
   886→      router.push(`/map?from=${fromStation.id}&to=${toStation.id}`);
   887→    }
   888→  };
   889→
   890→  const doShare = async () => {
   891→    const text =
   892→      fromStation && toStation
   893→        ? `مسیر ${fromStation.name} → ${toStation.name} — متروجو`
   894→        : "متروجو — مسیریابی هوشمند سفر";
   895→    const url = typeof window !== "undefined" ? window.location.href : "";
   896→    try {
   897→      if (navigator.share) {
   898→        await navigator.share({ title: "متروجو", text, url });
   899→        return;
   900→      }
   901→      await navigator.clipboard.writeText(`${text} ${url}`);
   902→      toast({ title: "لینک مسیر کپی شد", description: "برای اشتراک‌گذاری آماده است" });
   903→    } catch {
   904→      /* کاربر اشتراک‌گذاری را لغو کرد */
   905→    }
   906→  };
   907→
   908→  const focusPrefs = () => {
   909→    document.getElementById("plan-prefs")?.scrollIntoView({ behavior: "smooth", block: "center" });
   910→  };
   911→
   912→  const sameStation =
