    63→    63→
    64→    64→function textWidthPx(text: string, weight: number, px: number, epoch: number): number {
    65→    65→  const key = `${epoch}|${weight}|${px}|${text}`;
    66→    66→  const hit = widthCache.get(key);
    67→    67→  if (hit !== undefined) return hit;
    68→    68→  let w: number;
    69→    69→  try {
    70→    70→    if (!measureCtx) {
    71→    71→      measureCtx = document.createElement("canvas").getContext("2d");
    72→    72→    }
    73→    73→    if (measureCtx) {
    74→    74→      measureCtx.font = `${weight} ${px}px Vazirmatn, sans-serif`;
    75→    75→      w = measureCtx.measureText(text).width;
    76→    76→    } else {
    77→    77→      w = text.length * px * 0.6;
    78→    78→    }
    79→    79→  } catch {
    80→    80→    w = text.length * px * 0.6;
    81→    81→  }
    82→    82→  widthCache.set(key, w);
    83→    83→  return w;
    84→    84→}

   124→   124→export default function MetroMap({
   125→   125→  onStationSelect,
   126→   126→  highlightStations,
   127→   127→  className = "",
   128→   128→}: MetroMapProps) {
   129→   129→  const [t, setT] = useState<Transform>({ scale: 1, tx: 0, ty: 0 });
   130→   130→  const [legendOpen, setLegendOpen] = useState(false);
   131→   131→  const [locate, setLocate] = useState<LocateState>({ kind: "idle" });
   132→   132→  const [box, setBox] = useState({ w: 0, h: 0 });
   133→   133→  const [fontsTick, setFontsTick] = useState(0);
   134→   134→
   135→   135→  /* ---------- آینه همزمان ترنسفورم — جلوگیری از رقابت با React ---------- */
   136→   136→  const tRef = useRef<Transform>({ scale: 1, tx: 0, ty: 0 });
   137→   137→  const svgRef = useRef<SVGSVGElement>(null);
   138→   138→
   139→   139→  /* اشاره‌گرهای فعال — پایه ژست چندلمسی (t = آخرین زمان دیدن اشاره‌گر) */
   140→   140→  const pointersRef = useRef<Map<number, { x: number; y: number; t: number }>>(new Map());
   141→   141→  const panRef = useRef<{ startX: number; startY: number; tx: number; ty: number } | null>(null);
   142→   142→  const pinchRef = useRef<{
   143→   143→    dist0: number;
   144→   144→    vx0: number;
   145→   145→    vy0: number;
   146→   146→    s0: number;
   147→   147→    tx0: number;
   148→   148→    ty0: number;
   149→   149→  } | null>(null);
   150→   150→  const movedRef = useRef(false); // بعد از جابجایی، تپ ایستگاه نادیده گرفته می‌شود
   151→   151→
   152→   152→  /* ---------- اندازه واقعی بوم برای مقیاس ثابت متن ---------- */
   153→   153→  useEffect(() => {
   154→   154→    const el = svgRef.current;
   155→   155→    if (!el) return;
   156→   156→    const update = () => {
   157→   157→      const r = el.getBoundingClientRect();
   158→   158→      if (r.width < 2 || r.height < 2) return;
   159→   159→      setBox((b) =>
   160→   160→        Math.abs(b.w - r.width) > 0.5 || Math.abs(b.h - r.height) > 0.5
   161→   161→          ? { w: r.width, h: r.height }
   162→   162→          : b
   163→   163→      );
   164→   164→    };
   165→   165→    update();
   166→   166→    const ro = new ResizeObserver(update);
   167→   167→    ro.observe(el);
   168→   168→    return () => ro.disconnect();
   169→   169→  }, []);
   170→   170→
   171→   171→  /* ---------- بعد از لود کامل فونت، عرض متن‌ها دوباره اندازه‌گیری می‌شود ---------- */
   172→   172→  useEffect(() => {
   173→   173→    let alive = true;
   174→   174→    try {
   175→   175→      document.fonts?.ready
   176→   176→        .then(() => {
   177→   177→          if (!alive) return;
   178→   178→          widthCache.clear();
   179→   179→          setFontsTick((v) => v + 1);
   180→   180→        })
   181→   181→        .catch(() => {});
   182→   182→    } catch {
   183→   183→      /* مرورگرهای قدیمی */
   184→   184→    }
   185→   185→    return () => {
   186→   186→      alive = false;
   187→   187→    };
   188→   188→  }, []);
   189→   189→
   190→   190→  const clamp = (v: number, min: number, max: number) => Math.min(max, Math.max(min, v));
   191→   191→
   192→   192→  /** ثبت ترنسفورم جدید — همیشه از طریق این تابع */
   193→   193→  const commit = useCallback((next: Transform) => {
   194→   194→    const scale = clamp(next.scale, MIN_SCALE, MAX_SCALE);
   195→   195→    /* پنجره دید واقعی بر حسب واحدهای viewBox — با احتساب letterbox حالت meet.
   196→   196→       clamp قبلی فقط مرکز بوم را مبنا می‌گرفت و در بزرگ‌نمایی اجازه می‌داد
   197→   197→       کل نقشه از پنجره دید خارج شود (کاربر حس می‌کرد pan خراب است). */
   198→   198→    let wx0 = 0, wy0 = 0, winW = MAP_CANVAS.w, winH = MAP_CANVAS.h;
   199→   199→    const svg = svgRef.current;
   200→   200→    if (svg) {
   201→   201→      const rect = svg.getBoundingClientRect();
   202→   202→      if (rect.width > 2 && rect.height > 2) {
   203→   203→        const f = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   204→   204→        wx0 = -((rect.width - MAP_CANVAS.w * f) / 2) / f;
   205→   205→        wy0 = -((rect.height - MAP_CANVAS.h * f) / 2) / f;
   206→   206→        winW = rect.width / f;
   207→   207→        winH = rect.height / f;
   208→   208→      }
   209→   209→    }
   210→   210→    const OVER = 60; // اجازه سرریز اندک در لبه‌ها
   211→   211→    // لبه محتوا نباید از پنجره دید بیرون بزند — در هر بزرگ‌نمایی کل نقشه قابل رسیدن است
   212→   212→    const loX = wx0 + winW - scale * MAP_CANVAS.w - OVER;
   213→   213→    const hiX = wx0 + OVER;
   214→   214→    const loY = wy0 + winH - scale * MAP_CANVAS.h - OVER;
   215→   215→    const hiY = wy0 + OVER;
   216→   216→    const tx = loX <= hiX ? clamp(next.tx, loX, hiX) : (loX + hiX) / 2;
   217→   217→    const ty = loY <= hiY ? clamp(next.ty, loY, hiY) : (loY + hiY) / 2;
   218→   218→    const fixed = { scale, tx, ty };
   219→   219→    tRef.current = fixed;
   220→   220→    setT(fixed);
   221→   221→  }, []);
   222→   222→
   223→   223→  /** تبدیل مختصات صفحه به مختصات viewBox (با احتساب letterbox وضعیت meet) */
   224→   224→  const viewPoint = useCallback((clientX: number, clientY: number) => {
   225→   225→    const svg = svgRef.current;
   226→   226→    if (!svg) return { vx: 0, vy: 0 };
   227→   227→    const rect = svg.getBoundingClientRect();
   228→   228→    const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   229→   229→    const ox = (rect.width - MAP_CANVAS.w * fit) / 2;
   230→   230→    const oy = (rect.height - MAP_CANVAS.h * fit) / 2;
   231→   231→    return {
   232→   232→      vx: (clientX - rect.left - ox) / (fit * tRef.current.scale) - tRef.current.tx / tRef.current.scale,
   233→   233→      vy: (clientY - rect.top - oy) / (fit * tRef.current.scale) - tRef.current.ty / tRef.current.scale,
   234→   234→    };
   235→   235→  }, []);
   236→   236→
   237→   237→  /** بزرگ‌نمایی حول یک نقطه صفحه (پیش‌فرض: مرکز) — لنگر ثابت می‌ماند */
   238→   238→  const zoomAt = useCallback(
   239→   239→    (factor: number, clientX?: number, clientY?: number) => {
   240→   240→      const t0 = tRef.current;
   241→   241→      const svg = svgRef.current;
   242→   242→      if (!svg) {
   243→   243→        commit({ ...t0, scale: t0.scale * factor });
   244→   244→        return;
   245→   245→      }
   246→   246→      const rect = svg.getBoundingClientRect();
   247→   247→      const cx = clientX ?? rect.left + rect.width / 2;
   248→   248→      const cy = clientY ?? rect.top + rect.height / 2;
   249→   249→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   250→   250→      const ox = (rect.width - MAP_CANVAS.w * fit) / 2;
   251→   251→      const oy = (rect.height - MAP_CANVAS.h * fit) / 2;
   252→   252→      // نقطه لنگر در مختصات viewBox فعلی
   253→   253→      const ax = (cx - rect.left - ox) / fit;
   254→   254→      const ay = (cy - rect.top - oy) / fit;
   255→   255→      const scale = clamp(t0.scale * factor, MIN_SCALE, MAX_SCALE);
   256→   256→      commit({
   257→   257→        scale,
   258→   258→        tx: ax - ((ax - t0.tx) * scale) / t0.scale,
   259→   259→        ty: ay - ((ay - t0.ty) * scale) / t0.scale,
   260→   260→      });
   261→   261→    },
   262→   262→    [commit]
   263→   263→  );
   264→   264→
   265→   265→  /* ---------- چرخ ماوس (غیر-Passive تا preventDefault کار کند) ---------- */
   266→   266→  useEffect(() => {
   267→   267→    const el = svgRef.current;
   268→   268→    if (!el) return;
   269→   269→    const onWheel = (e: WheelEvent) => {
   270→   270→      e.preventDefault();
   271→   271→      zoomAt(e.deltaY < 0 ? 1.15 : 0.87, e.clientX, e.clientY);
   272→   272→    };
   273→   273→    el.addEventListener("wheel", onWheel, { passive: false });
   274→   274→    return () => el.removeEventListener("wheel", onWheel);
   275→   275→  }, [zoomAt]);
   276→   276→
   277→   277→  /* ---------- تور ایمنی سراسری برای اشاره‌گرهای گم‌شده ----------
   278→   278→     اگر pointerup/pointercancel انگشتی به SVG نرسد (مثلاً انگشت روی کارت
   279→   279→     راهنما، دکمه‌های شناور یا بیرون نقشه بالا آمده باشد، یا برچسبِ زیر
   280→   280→     انگشت در میانه ژست از DOM حذف شود)، اشاره‌گر «مرده» در حافظه می‌ماند
   281→   281→     و ژستِ بعدی به‌جای جابجایی، پینچِ خراب با لنگر مرده اجرا می‌شد —
   282→   282→     همین باعث «ناتوانی در pan بعد از zoom» بود. این شنوندهٔ سراسری
   283→   283→     (فاز capture) هر اشاره‌گر گم‌شده را بلافاصله پاک می‌کند. */
   284→   284→  useEffect(() => {
   285→   285→    const release = (e: PointerEvent) => {
   286→   286→      if (!pointersRef.current.has(e.pointerId)) return;
   287→   287→      pointersRef.current.delete(e.pointerId);
   288→   288→      if (pointersRef.current.size === 1) {
   289→   289→        // یک انگشت باقی مانده — ادامه با جابجایی از موقعیت فعلی او
   290→   290→        pinchRef.current = null;
   291→   291→        const [p] = [...pointersRef.current.values()];
   292→   292→        panRef.current = { startX: p.x, startY: p.y, tx: tRef.current.tx, ty: tRef.current.ty };
   293→   293→      } else if (pointersRef.current.size === 0) {
   294→   294→        panRef.current = null;
   295→   295→        pinchRef.current = null;
   296→   296→      }
   297→   297→    };
   298→   298→    const resetAll = () => {
   299→   299→      pointersRef.current.clear();
   300→   300→      panRef.current = null;
   301→   301→      pinchRef.current = null;
   302→   302→    };
   303→   303→    const onVisibility = () => {
   304→   304→      if (document.hidden) resetAll();
   305→   305→    };
   306→   306→    window.addEventListener("pointerup", release, true);
   307→   307→    window.addEventListener("pointercancel", release, true);
   308→   308→    window.addEventListener("blur", resetAll);
   309→   309→    document.addEventListener("visibilitychange", onVisibility);
   310→   310→    return () => {
   311→   311→      window.removeEventListener("pointerup", release, true);
   312→   312→      window.removeEventListener("pointercancel", release, true);
   313→   313→      window.removeEventListener("blur", resetAll);
   314→   314→      document.removeEventListener("visibilitychange", onVisibility);
   315→   315→    };
   316→   316→  }, []);
   317→   317→
   318→   318→  /* ---------- اشاره‌گرها ---------- */
   319→   319→  const onPointerDown = (e: React.PointerEvent) => {
   320→   320→    /* پاک‌سازی اشاره‌گرهای مرده — pointerup آن‌ها هرگز به نقشه نرسیده
   321→   321→       (انگشت روی کارت HTML یا بیرون نقشه بالا آمده بود). همین اشاره‌گرهای
   322→   322→       مرده ژستِ بعدی را به پینچ خراب تبدیل می‌کردند. */
   323→   323→    const now = performance.now();
   324→   324→    for (const [id, p] of pointersRef.current) {
   325→   325→      if (now - p.t > STALE_POINTER_MS) pointersRef.current.delete(id);
   326→   326→    }
   327→   327→    pointersRef.current.set(e.pointerId, { x: e.clientX, y: e.clientY, t: now });
   328→   328→
   329→   329→    if (pointersRef.current.size === 1) {
   330→   330→      // شروع جابجایی تک‌انگشتی — کپچر عمداً فعال نمی‌شود تا تپ روی
   331→   331→      // برچسب/ایستگاه با click واقعی روی خود عنصر کار کند؛
   332→   332→      // کپچر روی SVG به‌محض عبور از آستانه درگ فعال می‌شود.
   333→   333→      movedRef.current = false;
   334→   334→      panRef.current = { startX: e.clientX, startY: e.clientY, tx: tRef.current.tx, ty: tRef.current.ty };
   335→   335→      pinchRef.current = null;
   336→   336→    } else if (pointersRef.current.size === 2) {
   337→   337→      // شروع پینچ — کپچر هر دو انگشت روی خود SVG (عنصر پایدار که هرگز
   338→   338→      // unmount نمی‌شود؛ برخلاف برچسب‌ها که در میانه ژست حذف می‌شوند)
   339→   339→      try {
   340→   340→        svgRef.current?.setPointerCapture?.(e.pointerId);
   341→   341→        for (const id of pointersRef.current.keys()) {
   342→   342→          if (id !== e.pointerId) svgRef.current?.setPointerCapture?.(id);
   343→   343→        }
   344→   344→      } catch {
   345→   345→        /* برخی محیط‌ها (رویدادهای ساختگی/WebView) خطای InvalidPointerId می‌دهند */
   346→   346→      }
   347→   347→      const [p1, p2] = [...pointersRef.current.values()];
   348→   348→      const t0 = tRef.current;
   349→   349→      const svg = svgRef.current;
   350→   350→      if (!svg) return;
   351→   351→      const rect = svg.getBoundingClientRect();
   352→   352→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   353→   353→      const ox = (rect.width - MAP_CANVAS.w * fit) / 2;
   354→   354→      const oy = (rect.height - MAP_CANVAS.h * fit) / 2;
   355→   355→      const mx = (p1.x + p2.x) / 2;
   356→   356→      const my = (p1.y + p2.y) / 2;
   357→   357→      pinchRef.current = {
   358→   358→        dist0: Math.hypot(p1.x - p2.x, p1.y - p2.y) || 1,
   359→   359→        vx0: (mx - rect.left - ox) / fit,
   360→   360→        vy0: (my - rect.top - oy) / fit,
   361→   361→        s0: t0.scale,
   362→   362→        tx0: t0.tx,
   363→   363→        ty0: t0.ty,
   364→   364→      };
   365→   365→      panRef.current = null;
   366→   366→    }
   367→   367→  };
   368→   368→
   369→   369→  const onPointerMove = (e: React.PointerEvent) => {
   370→   370→    if (!pointersRef.current.has(e.pointerId)) return;
   371→   371→    pointersRef.current.set(e.pointerId, { x: e.clientX, y: e.clientY, t: performance.now() });
   372→   372→
   373→   373→    if (pinchRef.current && pointersRef.current.size >= 2) {

   377→   377→      const dist = Math.hypot(p1.x - p2.x, p1.y - p2.y) || 1;
   378→   378→      const svg = svgRef.current;
   379→   379→      if (!svg) return;
   380→   380→      const rect = svg.getBoundingClientRect();
   381→   381→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   382→   382→      const ox = (rect.width - MAP_CANVAS.w * fit) / 2;
   383→   383→      const oy = (rect.height - MAP_CANVAS.h * fit) / 2;
   384→   384→      const mx = (p1.x + p2.x) / 2;
   385→   385→      const my = (p1.y + p2.y) / 2;
   386→   386→      const vxm = (mx - rect.left - ox) / fit;
   387→   387→      const vym = (my - rect.top - oy) / fit;
   388→   388→      const scale = clamp((p0.s0 * dist) / p0.dist0, MIN_SCALE, MAX_SCALE);
   389→   389→      // نقطه میانی اولیه باید زیر انگشتان بماند
   390→   390→      const cx = (p0.vx0 - p0.tx0) / p0.s0;
   391→   391→      const cy = (p0.vy0 - p0.ty0) / p0.s0;
   392→   392→      commit({
   393→   393→        scale,
   394→   394→        tx: vxm - scale * cx,
   395→   395→        ty: vym - scale * cy,
   396→   396→      });
   397→   397→      movedRef.current = true;
   398→   398→      return;
   399→   399→    }
   400→   400→
   401→   401→    if (panRef.current && pointersRef.current.size === 1) {
   402→   402→      const p0 = panRef.current;
   403→   403→      const svg = svgRef.current;
   404→   404→      if (!svg) return;
   405→   405→      const rect = svg.getBoundingClientRect();

   407→   407→      const dx = (e.clientX - p0.startX) / fit;
   408→   408→      const dy = (e.clientY - p0.startY) / fit;

   417→   417→          }
   418→   418→        }
   419→   419→        movedRef.current = true;
   420→   420→      }
   421→   421→      commit({ scale: tRef.current.scale, tx: p0.tx + dx, ty: p0.ty + dy });
   422→   422→    }

   457→   457→    [commit]

   669→   669→  }, [t, fit, fontsTick, highlighted, legendOpen, box.w, box.h]);
