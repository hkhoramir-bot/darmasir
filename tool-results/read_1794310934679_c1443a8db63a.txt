   519→   519→function hash(s: string): number {
   520→   520→  let h = 0;
   521→   521→  for (let i = 0; i < s.length; i++) h = (h * 31 + s.charCodeAt(i)) >>> 0;
   522→   522→  return h;
   523→   523→}

   525→   525→const AMENITY_POOL: AmenityKey[] = [
   526→   526→  "escalator", "elevator", "wc", "atm", "bus", "taxi", "parking", "baby", "bike", "wifi",
   527→   527→];
   528→   528→
   529→   529→export const STATIONS: StationDef[] = RAW.map((r) => {
   530→   530→  const h = hash(r.id);
   531→   531→  const pool = AMENITY_POOL.filter((_, i) => (h >> i) % 3 !== 0);
   532→   532→  const amenities = r.amenities ?? (pool.length >= 3 ? pool.slice(0, 3 + (h % 3)) : ["escalator" as AmenityKey]);
   533→   533→  return {
   534→   534→    id: r.id,
   535→   535→    name: r.name,
   536→   536→    cityId: LINES.find((l) => l.id === r.line)!.cityId,
   537→   537→    lines: [{ lineId: r.line, order: r.order }],
   538→   538→    amenities,
   539→   539→    featured: r.featured,
   540→   540→  };
   541→   541→}).reduce<StationDef[]>((acc, s) => {
   542→   542→  const found = acc.find((x) => x.id === s.id);
   543→   543→  if (found) found.lines.push(...s.lines);
   544→   544→  else acc.push(s);
   545→   545→  return acc;
   546→   546→}, []);

   594→   594→export function isWithinService(line: { firstTrain: string; lastTrain: string }, d = new Date()): boolean {
   595→   595→  const [fh, fm] = line.firstTrain.split(":").map(Number);

   601→   601→export function nextArrivals(lineId: string, stationId: string, now = Date.now()): number[] {
   602→   602→  const line = getLine(lineId);
   603→   603→  const headwaySec = Math.round(line.headway * 60);
   604→   604→  const phase = hash(stationId + lineId + new Date(now).getHours()) % headwaySec;
   605→   605→  const elapsed = Math.floor(now / 1000) % headwaySec;
   606→   606→  let next = phase - elapsed;
   607→   607→  while (next <= 0) next += headwaySec;
   608→   608→  const late = line.status === "delay" ? 8 * 60 : 0;
   609→   609→  return [next + late, next + headwaySec + late];
   610→   610→}

   614→   614→export function crowdingNow(lineId: string, now = new Date()): CrowdLevel {
   615→   615→  const h = now.getHours();
   616→   616→  const peak = (h >= 7 && h <= 9) || (h >= 16.5 && h <= 19);
   617→   617→  const mid = (h >= 10 && h <= 16) || (h >= 19.5 && h <= 22);
   618→   618→  const line = getLine(lineId);
   619→   619→  if (line.status === "crowded") return peak ? "busy" : mid ? "busy" : "moderate";
   620→   620→  if (peak) return hash(lineId + h) % 4 === 0 ? "quiet" : "busy";
   621→   621→  if (mid) return hash(lineId + h) % 3 === 0 ? "busy" : "moderate";
   622→   622→  return hash(lineId + h) % 3 === 0 ? "moderate" : "quiet";
   623→   623→}

   866→   866→  station: string;

   873→   873→/** Dijkstra over (station, line) states. */
   874→   874→export function computeRoute(
   875→   875→  fromId: string,
   876→   876→  toId: string,
   877→   877→  pref: "fastest" | "least_transfers" | "quietest" = "fastest",
   878→   878→  now = new Date()
   879→   879→): RoutePlan | null {
   880→   880→  const from = getStation(fromId);
   881→   881→  const to = getStation(toId);
   882→   882→  if (!from || !to || fromId === toId) return null;
   883→   883→
   884→   884→  const seg = (lineId: string) => {
   885→   885→    const base = getLine(lineId).segmentMinutes;
   886→   886→    const crowd = crowdingNow(lineId, now);
   887→   887→    const factor = pref === "quietest" ? (crowd === "busy" ? 1.35 : crowd === "moderate" ? 1.1 : 0.95) : 1;
   888→   888→    return base * factor;
   889→   889→  };
   890→   890→
   891→   891→  const start: State = { station: fromId, line: null, time: 0, transfers: 0, path: [{ station: fromId, line: null }] };
   892→   892→  const best = new Map<string, number>();
   893→   893→  const key = (s: State) => `${s.station}|${s.line ?? "*"}`;
   894→   894→  const queue: State[] = [start];
   895→   895→  best.set(key(start), 0);
   896→   896→
   897→   897→  let done: State | null = null;
   898→   898→  const visited = new Set<string>();
   899→   899→
   900→   900→  while (queue.length) {
   901→   901→    // pick min
   902→   902→    let idx = 0;
   903→   903→    for (let i = 1; i < queue.length; i++) {
   904→   904→      if (queue[i].time < queue[idx].time) idx = i;
   905→   905→    }
   906→   906→    const cur = queue.splice(idx, 1)[0];
   907→   907→    if (visited.has(key(cur))) continue;
   908→   908→    visited.add(key(cur));
   909→   909→
   910→   910→    if (cur.station === toId) {
   911→   911→      done = cur;
   912→   912→      break;
   913→   913→    }
   914→   914→
   915→   915→    const st = getStation(cur.station)!;
   916→   916→    for (const ls of st.lines) {
   917→   917→      // board / continue on line ls.lineId
   918→   918→      const isTransfer = cur.line !== null && cur.line !== ls.lineId;
   919→   919→      const penalty = cur.line === null ? BOARDING_MIN : isTransfer ? TRANSFER_MIN : 0;
   920→   920→      const transferCost =
   921→   921→        pref === "least_transfers" ? (isTransfer ? 40 : 0) : 0;
   922→   922→
   923→   923→      // neighbors along the line
   924→   924→      const list = stationsOfLine(ls.lineId);
   925→   925→      const myIdx = list.findIndex((x) => x.id === cur.station);
   926→   926→      for (const delta of [-1, 1]) {
   927→   927→        const nIdx = myIdx + delta;
   928→   928→        if (nIdx < 0 || nIdx >= list.length) continue;
   929→   929→        const nb = list[nIdx];
   930→   930→        const t = cur.time + seg(ls.lineId) + penalty + transferCost;
   931→   931→        const ns: State = {
   932→   932→          station: nb.id,
   933→   933→          line: ls.lineId,
   934→   934→          time: t,
   935→   935→          transfers: cur.transfers + (isTransfer ? 1 : 0),
   936→   936→          path: [...cur.path, { station: nb.id, line: ls.lineId }],
   937→   937→        };
   938→   938→        const k = key(ns);
   939→   939→        if (!best.has(k) || t < best.get(k)! - 1e-9) {
   940→   940→          best.set(k, t);
   941→   941→          queue.push(ns);
   942→   942→        }
   943→   943→      }
   944→   944→    }
   945→   945→  }
   946→   946→
   947→   947→  if (!done) return null;
   948→   948→
   949→   949→  // compress path into legs
   950→   950→  const legs: RouteLeg[] = [];
   951→   951→  const path = done.path.slice(1);
   952→   952→  let curLine: string | null = null;
   953→   953→  let legStart = path[0].station;
   954→   954→  let prev = fromId;
   955→   955→
   956→   956→  const flush = (endStation: string) => {
   957→   957→    if (!curLine) return;
   958→   958→    const line = getLine(curLine);
   959→   959→    const ids = stationsOfLine(curLine).map((s) => s.id);
   960→   960→    const i1 = ids.indexOf(legStart);
   961→   961→    const i2 = ids.indexOf(endStation);
   962→   962→    const slice = i1 <= i2 ? ids.slice(i1, i2 + 1) : ids.slice(i2, i1 + 1).reverse();
   963→   963→    const names = slice.map((id) => getStation(id)!.name);
   964→   964→    legs.push({
   965→   965→      lineId: curLine,
   966→   966→      lineName: line.name,
   967→   967→      color: line.color,
   968→   968→      textOnColor: line.textOnColor,
   969→   969→      fromId: slice[0],
   970→   970→      fromName: names[0],
   971→   971→      toId: slice[slice.length - 1],
   972→   972→      toName: names[names.length - 1],
   973→   973→      stationIds: slice,
   974→   974→      stationNames: names,
   975→   975→      stops: slice.length - 1,
   976→   976→      minutes: Math.max(1, Math.round((slice.length - 1) * line.segmentMinutes)),
   977→   977→    });
   978→   978→  };
   979→   979→
   980→   980→  for (const step of path) {
   981→   981→    if (curLine === null) {
   982→   982→      curLine = step.line;
   983→   983→      legStart = prev;
   984→   984→    } else if (step.line !== curLine) {
   985→   985→      flush(prev);
   986→   986→      curLine = step.line;
   987→   987→      legStart = prev;
   988→   988→    }
   989→   989→    prev = step.station;
   990→   990→  }
   991→   991→  flush(prev);
   992→   992→
   993→   993→  const totalStops = legs.reduce((a, l) => a + l.stops, 0);
   994→   994→  // زمان نمایشی واقعی = جمع زمان لگ‌ها + هزینه تعویض خط + سوار شدن (بدون وزن بهینه‌سازی)
   995→   995→  const totalMinutes = Math.round(
   996→   996→    legs.reduce((a, l) => a + l.minutes, 0) + (legs.length - 1) * TRANSFER_MIN + BOARDING_MIN
   997→   997→  );
   998→   998→  const transferStations = legs.slice(0, -1).map((l) => l.toName);
   999→   999→  const busyLeg = legs.find((l) => crowdingNow(l.lineId, now) === "busy");
  1000→  1000→
  1001→  1001→  return {
  1002→  1002→    legs,
  1003→  1003→    totalMinutes,
  1004→  1004→    totalStations: totalStops + 1,
  1005→  1005→    transfers: legs.length - 1,
  1006→  1006→    transferStations,
  1007→  1007→    fare: fareFor(totalStops),
  1008→  1008→    pref,
  1009→  1009→    crowdNote: busyLeg
  1010→  1010→      ? `ازدحام بالا روی ${busyLeg.lineName} در ساعات اوج`
  1011→  1011→      : "سرویس‌دهی همه خطوط مسیر عادی و روان است",
  1012→  1012→  };
  1013→  1013→}
