   108→function StationField({
   109→  field,
   110→  sel,
   111→  text,
   112→  open,
   113→  onOpen,
   114→  onText,
   115→  onPick,
   116→  onClear,
   117→}: {
   118→  field: Field;
   119→  sel: StationDef | null;
   120→  text: string;
   121→  open: boolean;
   122→  onOpen: (f: Field) => void;
   123→  onText: (f: Field, v: string) => void;
   124→  onPick: (f: Field, s: StationDef) => void;
   125→  onClear: (f: Field) => void;
   126→}) {
   127→  const city = useShell((s) => s.city);
   128→  const isFrom = field === "from";
   129→
   130→  const suggestions = useMemo(() => {
   131→    const q = text.trim();
   132→    if (!q) {
   133→      const featured = searchStations({ cityId: city }).filter((s) => s.featured);
   134→      return (featured.length ? featured : searchStations({ cityId: city })).slice(0, 5);
   135→    }
   136→    const res = searchStations({ q, cityId: city });
   137→    return (res.length ? res : searchStations({ q })).slice(0, 5);
   138→  }, [text, city]);
   139→
   140→  const dotStyle = sel ? { backgroundColor: getLine(sel.lines[0]?.lineId ?? "")?.color } : undefined;
   141→  const chip = sel ? chipFor(sel) : null;
   142→
   143→  /* حالت نمایش (ایستگاه انتخاب‌شده) — دقیقاً مطابق کارت خلاصه طرح _10 */
   144→  if (!open && sel && chip) {
   145→    return (
   146→      <button
   147→        type="button"
   148→        onClick={() => onOpen(field)}
   149→        className="w-full flex items-center gap-2.5 bg-surface-container-low px-3 py-2 rounded-xl border border-outline-variant/20 text-right active:scale-[0.99] transition-all"
   150→        aria-label={isFrom ? `تغییر مبدأ: ${sel.name}` : `تغییر مقصد: ${sel.name}`}
   151→      >
   152→        <span className="w-2.5 h-2.5 rounded-full flex-shrink-0" style={dotStyle} />
   153→        <div className="flex-1 flex items-center justify-between min-w-0 gap-2">
   154→          <span className="text-body-md font-body-md font-medium text-on-surface truncate">{sel.name}</span>
   155→          <span className={`text-label-sm font-label-sm px-2 py-0.5 rounded font-bold flex-shrink-0 ${chip.cls}`}>
   156→            {chip.label}
   157→          </span>
   158→        </div>
   159→      </button>
   160→    );
   161→  }
   162→
   163→  /* حالت ورودی + فهرست پیشنهاد جستجو */
   164→  return (
   165→    <div className="relative flex items-center gap-2.5 bg-surface-container-low px-3 py-2 rounded-xl border border-outline-variant/20">
   166→      <span className="w-2.5 h-2.5 rounded-full bg-outline-variant flex-shrink-0" />
   167→      <input
   168→        type="text"
   169→        value={text}
   170→        onChange={(e) => onText(field, e.target.value)}
   171→        onFocus={() => onOpen(field)}
   172→        autoFocus={open}
   173→        placeholder={isFrom ? "ایستگاه مبدأ..." : "ایستگاه مقصد..."}
   174→        aria-label={isFrom ? "ایستگاه مبدأ" : "ایستگاه مقصد"}
   175→        className="flex-1 min-w-0 bg-transparent text-body-md font-body-md font-medium text-on-surface placeholder:text-outline focus:outline-none border-none p-0"
   176→      />
   177→      {text && (
   178→        <button
   179→          type="button"
   180→          aria-label="پاک کردن"
   181→          onClick={() => onClear(field)}
   182→          className="text-outline hover:text-on-surface p-1 rounded-full hover:bg-surface-container-high transition-colors shrink-0 active:scale-95"
   183→        >
   184→          <MI name="close" size={16} />
   185→        </button>
   186→      )}
   187→      {open && (
   188→        <div className="absolute top-full left-0 right-0 mt-1.5 bg-surface-container-lowest rounded-xl shadow-lg border border-outline-variant/40 py-1 z-[60] text-xs">
   189→          {suggestions.length === 0 && <div className="px-3 py-2 text-outline">ایستگاهی یافت نشد</div>}
   190→          {suggestions.map((s) => {
   191→            const c = chipFor(s);
   192→            return (
   193→              <button
   194→                key={s.id}
   195→                type="button"
   196→                onClick={() => onPick(field, s)}
   197→                className="w-full text-right px-3 py-2 hover:bg-surface-container text-on-surface flex items-center justify-between gap-2 active:bg-surface-container transition-colors"
   198→              >
   199→                <span className="truncate">{s.name}</span>
   200→                <span className={`text-[10px] font-bold px-1.5 py-0.5 rounded shrink-0 ${c.cls}`}>{c.label}</span>
   201→              </button>
   202→            );
   203→          })}
   204→        </div>
   205→      )}
   206→    </div>
   207→  );
   208→}

   327→/* --------------- نشان زنده قطار بعدی + شلوغی سکو هر بخش --------------- */
   328→function LegArrivalPill({ leg }: { leg: RouteLeg }) {
   329→  const [tick, setTick] = useState(0);
   330→  useEffect(() => {
   331→    const t = setInterval(() => setTick((x) => x + 1), 30000);
   332→    return () => clearInterval(t);
   333→  }, []);
   334→
   335→  const info = useMemo(() => {
   336→    const line = getLine(leg.lineId);
   337→    const list = stationsOfLine(leg.lineId);
   338→    const iFrom = list.findIndex((s) => s.id === leg.fromId);
   339→    const iTo = list.findIndex((s) => s.id === leg.toId);
   340→    if (!line || iFrom < 0 || iTo < 0) return null;
   341→    const terminal = line.terminals[iTo >= iFrom ? 1 : 0];
   342→    const dirs = nextArrivalsBoth(leg.lineId, leg.fromId);
   343→    const arr = dirs.find((d) => d.to === terminal) ?? dirs[0];
   344→    if (!arr) return null;
   345→    if (arr.outOfService || arr.minutes.length === 0) {
   346→      const d = new Date();
   347→      const fallback = arr.outOfService ? null : headwayAt(line, d.getHours() * 60 + d.getMinutes());
   348→      return { out: arr.outOfService, minutes: fallback, crowd: arr.crowd, crowdLabel: arr.crowdLabel };
   349→    }
   350→    return { out: false, minutes: arr.minutes[0], crowd: arr.crowd, crowdLabel: arr.crowdLabel };
   351→  }, [leg, tick]);
   352→
   353→  if (!info) return null;
   354→
   355→  return (
   356→    <div className="mt-2.5 bg-surface-container-low rounded-xl p-2.5 px-3 flex items-center justify-between gap-2">
   357→      <div className="flex items-center gap-1.5 text-on-surface-variant min-w-0">
   358→        <span className="inline-flex shrink-0" style={{ color: leg.color }}>
   359→          <MI name="schedule" size={16} />
   360→        </span>
   361→        <span className="text-body-sm font-body-sm shrink-0">قطار بعدی:</span>
   362→        <span className="text-body-sm font-body-sm font-bold text-on-surface truncate">
   363→          {info.out || info.minutes === null ? "خارج از ساعات سرویس" : `${toFa(info.minutes)} دقیقه دیگر`}
   364→        </span>
   365→      </div>
   366→      <div className="flex items-center gap-1 shrink-0">
   367→        <span className="text-label-sm font-label-sm text-outline">شلوغی سکو:</span>
   368→        <span className={`text-label-sm font-label-sm font-bold px-1.5 py-0.5 rounded ${CROWD_CHIP[info.crowd]}`}>
   369→          {info.crowdLabel}
   370→        </span>
   371→      </div>
   372→    </div>
   373→  );
   374→}

   413→/* --------------- گام‌های میانی: آکاردئون ایستگاه‌های عبوری --------------- */
   414→function LegAccordionRow({ leg }: { leg: RouteLeg }) {
   415→  const [open, setOpen] = useState(false);
   416→  const mid = leg.stationIds.slice(1, -1);
   417→  if (mid.length === 0) return null;
   418→  const perStop = leg.stops > 0 ? leg.minutes / leg.stops : 0;
   419→  const midMinutes = Math.round(perStop * (mid.length + 1));
   420→
   421→  const hintFor = (sid: string) => {
   422→    const others = interchangeLinesOf(sid)
   423→      .filter((id) => id !== leg.lineId)
   424→      .map((id) => getLine(id)?.number ?? 0)
   425→      .filter((n) => n > 0)
   426→      .sort((a, b) => a - b);
   427→    return others.length ? `تقاطع خط ${others.map((n) => toFa(n)).join(" و ")}` : null;
   428→  };
   429→
   430→  return (
   431→    <div className="relative flex gap-3 pb-6">
   432→      <div className="flex flex-col items-center flex-shrink-0 w-8">
   433→        <div className="w-1.5 h-full rounded-full" style={{ backgroundColor: leg.color }} />
   434→      </div>
   435→      <div className="flex-1 min-w-0">
   436→        <button
   437→          type="button"
   438→          onClick={() => setOpen((v) => !v)}
   439→          aria-expanded={open}
   440→          className="w-full bg-surface-container/60 hover:bg-surface-container rounded-xl px-3 py-2 flex items-center justify-between gap-2 transition-colors"
   441→        >
   442→          <div className="flex items-center gap-2 min-w-0">
   443→            <span className="w-2 h-2 rounded-full shrink-0" style={{ backgroundColor: leg.color }} />
   444→            <span className="text-body-sm font-body-sm text-on-surface-variant font-medium truncate">
   445→              {toFa(mid.length)} ایستگاه عبوری ({toFa(midMinutes)} دقیقه)
   446→            </span>

   654→/* ------------------------------ بدنه صفحه ------------------------------ */
   655→function PlanInner() {
   656→  const router = useRouter();
   657→  const searchParams = useSearchParams();
   658→  const { toast } = useToast();
   659→  const announcer = useAnnouncer();
   660→  const city = useShell((s) => s.city);
   661→
   662→  const [starting, setStarting] = useState(false);
   663→  const [openDrop, setOpenDrop] = useState<Field | null>(null);
   664→  const [fromText, setFromText] = useState("");
   665→  const [toText, setToText] = useState("");
   666→  const [fromSel, setFromSel] = useState<StationDef | null>(null);
   667→  const [toSel, setToSel] = useState<StationDef | null>(null);
   668→  const [pref, setPref] = useState<Pref>("fastest");
   669→
   670→  /* اعلام صوتی ایستگاه‌ها — ذخیره در حافظه محلی */
   671→  const [voiceOn, setVoiceOn] = useState(true);
   672→  useEffect(() => {
   673→    try {
   674→      setVoiceOn(localStorage.getItem("metrojo-plan-voice") !== "0");
   675→    } catch {}
   676→  }, []);
   677→  const toggleVoice = () => {
   678→    setVoiceOn((v) => {
   679→      const nv = !v;
   680→      try {
   681→        localStorage.setItem("metrojo-plan-voice", nv ? "1" : "0");
   682→      } catch {}
   683→      const j = useJourney.getState().journey;
   684→      if (j) useJourney.getState().setVoice(nv);
   685→      return nv;
   686→    });
   687→  };
   688→
   689→  /* بیدارباش مقصد — فقط حالت کلاینت + حافظه محلی */
   690→  const [alarm, setAlarm] = useState<{ on: boolean; minutes: number }>({ on: false, minutes: 2 });
   691→  useEffect(() => {
   692→    try {
   693→      const raw = localStorage.getItem("metrojo-plan-alarm");
   694→      if (raw) {
   695→        const p = JSON.parse(raw) as { on?: boolean; minutes?: number };
   696→        if (typeof p.on === "boolean") setAlarm({ on: p.on, minutes: p.minutes ?? 2 });
   697→      }
   698→    } catch {}
   699→  }, []);
   700→  const setAlarmState = (next: { on: boolean; minutes: number }) => {
   701→    setAlarm(next);
   702→    try {
   703→      localStorage.setItem("metrojo-plan-alarm", JSON.stringify(next));
   704→    } catch {}
   705→    // سفر فعال هم بلافاصله از بیدارباش جدید پیروی می‌کند
   706→    const j = useJourney.getState().journey;
   707→    if (j) useJourney.getState().setAlarm(next.on, next.minutes);
   708→  };
   709→
   710→  const qFrom = searchParams.get("from");
   711→  const qTo = searchParams.get("to");
   712→  const qPref = searchParams.get("pref");
   713→
   714→  const fromStation = qFrom ? getStation(qFrom) : undefined;
   715→  const toStation = qTo ? getStation(qTo) : undefined;
   716→  const hasParams = !!(fromStation && toStation && fromStation.id !== toStation.id);
   717→
   718→  /* همگام‌سازی انتخاب‌ها و اولویت از پارامترهای آدرس */
   719→  useEffect(() => {
   720→    if (fromStation) {
   721→      setFromSel(fromStation);
   722→      setFromText(fromStation.name);
   723→    }
   724→    if (toStation) {
   725→      setToSel(toStation);
   726→      setToText(toStation.name);
   727→    }
   728→    setPref(qPref === "least_transfers" || qPref === "quietest" ? qPref : "fastest");
   729→  }, [qFrom, qTo, qPref, fromStation, toStation]);
   730→
   731→  const plan = useMemo(
   732→    () =>
   733→      fromStation && toStation && fromStation.id !== toStation.id
   734→        ? computeRoute(fromStation.id, toStation.id, pref)
   735→        : null,
   736→    [fromStation, toStation, pref]
   737→  );
   738→
   739→  /* یادداشت خروجی‌های مقصد */
   740→  const exitsText = useMemo(() => {
   741→    if (!toStation) return "";
   742→    return exitsFor(toStation.id)
   743→      .slice(0, 2)
   744→      .map((e) => `${e.label}: ${e.landmarks[0] ?? e.direction}`)
   745→      .join(" • ");
   746→  }, [toStation]);
   747→
   748→  /* متن وضعیت خط اول مسیر */
   749→  const firstLine = plan ? getLine(plan.legs[0]?.lineId ?? "") : undefined;
   750→  const lineOk = !firstLine?.status || firstLine.status === "normal";
   751→  const statusText = (() => {
   752→    if (!firstLine) return "وضعیت خطوط";
   753→    if (firstLine.status === "closed") return `وضعیت خطوط: خط ${toFa(firstLine.number)} مسدود است`;
   754→    if (firstLine.status === "delay") return `وضعیت خطوط: ${firstLine.statusNote ?? "تأخیر در سرویس‌دهی"}`;
   755→    if (firstLine.status === "crowded" || crowdingNow(firstLine.id) === "busy")
   756→      return `وضعیت خطوط: ازدحام در خط ${toFa(firstLine.number)}`;
   757→    return `وضعیت خطوط: خط ${toFa(firstLine.number)} فعال و روان`;
   758→  })();
   759→
   760→  /* رفع نام آزاد به ایستگاه (مطابق منطق قبلی صفحه) */
   761→  const resolve = (text: string): StationDef | undefined => {
   762→    const q = text.trim().replace(/[«»"]/g, "");
   763→    if (!q) return undefined;
   764→    const candidates = [...searchStations({ q, cityId: city }), ...searchStations({ q })];
   765→    const exact = candidates.find((s) => s.name === q);
   766→    if (exact) return exact;
   767→    const incl = candidates.find((s) => s.name.includes(q) || q.includes(s.name));
   768→    if (incl) return incl;
   769→    const words = q.split(/[\s()]+/).filter((w) => w.length > 1);
   770→    return candidates.find((s) => {
   771→      const sn = s.name.replace(/[()]/g, " ");
   772→      return words.length > 0 && words.every((w) => sn.includes(w));
   773→    });
   774→  };
   775→
   776→  const onText = (field: Field, v: string) => {
   777→    if (field === "from") {
   778→      setFromText(v);
   779→      setFromSel(null);
   780→    } else {
   781→      setToText(v);
   782→      setToSel(null);
   783→    }
   784→  };
   785→
   786→  const onClearField = (field: Field) => {
   787→    if (field === "from") {
   788→      setFromText("");
   789→      setFromSel(null);
   790→    } else {
   791→      setToText("");
   792→      setToSel(null);
   793→    }
   794→    setOpenDrop(field);
   795→    /* در حالت نتیجه، حذف پارامتر → بازگشت به حالت ورودی با حفظ ایستگاه دیگر */
   796→    if (qFrom && qTo) {
   797→      const params = new URLSearchParams();
   798→      if (field === "from" && qTo) params.set("to", qTo);
   799→      if (field === "to" && qFrom) params.set("from", qFrom);
   800→      params.set("pref", pref);
   801→      router.replace(`/plan?${params.toString()}`, { scroll: false });
   802→    }
   803→  };
   804→
   805→  const onPick = (field: Field, s: StationDef) => {
   806→    if (field === "from") {
   807→      setFromSel(s);
   808→      setFromText(s.name);
   809→    } else {
   810→      setToSel(s);
   811→      setToText(s.name);
   812→    }
   813→    setOpenDrop(null);
   814→    /* در حالت نتیجه، انتخاب بلافاصله مسیر را دوباره محاسبه می‌کند */
   815→    if (qFrom && qTo) {
   816→      const f = field === "from" ? s : fromStation;
   817→      const t = field === "to" ? s : toStation;
   818→      if (f && t && f.id !== t.id) {
   819→        router.replace(`/plan?from=${f.id}&to=${t.id}&pref=${pref}`, { scroll: false });
   820→      }
   821→    }
   822→  };
   823→
   824→  const onSwap = () => {
   825→    if (qFrom && qTo && fromStation && toStation) {
   826→      router.replace(`/plan?from=${toStation.id}&to=${fromStation.id}&pref=${pref}`, { scroll: false });
   827→      return;
   828→    }
   829→    const f = fromSel ?? resolve(fromText);
   830→    const t = toSel ?? resolve(toText);
   831→    setFromSel(t ?? null);
   832→    setFromText(t?.name ?? "");
   833→    setToSel(f ?? null);
   834→    setToText(f?.name ?? "");
   835→  };
   836→
   837→  const onPref = (p: Pref) => {
   838→    setPref(p);
   839→    if (qFrom && qTo) {
   840→      router.replace(`/plan?from=${qFrom}&to=${qTo}&pref=${p}`, { scroll: false });
   841→    }
   842→  };
   843→
   844→  const submit = () => {
   845→    const f = fromSel ?? resolve(fromText);
   846→    const t = toSel ?? resolve(toText);
   847→    if (!f || !t) {
   848→      toast({
   849→        title: "مبدأ و مقصد را انتخاب کنید",
   850→        description: "نام ایستگاه را از فهرست پیشنهاد برگزینید",
   851→        variant: "destructive",
   852→      });
   853→      return;
   854→    }
   855→    if (f.id === t.id) {
   856→      toast({ title: "مبدأ و مقصد یکسان هستند", variant: "destructive" });
   857→      return;
   858→    }
   859→    router.replace(`/plan?from=${f.id}&to=${t.id}&pref=${pref}`);
   860→  };
   861→
   862→  const startTrip = async () => {
   863→    if (!plan || !fromStation || !toStation || starting) return;
   864→    setStarting(true);
   865→    try {
   866→      const res = await fetch("/api/history", {
   867→        method: "POST",
   868→        headers: { "Content-Type": "application/json" },
   869→        body: JSON.stringify({
   870→          fromStationId: fromStation.id,
   871→          toStationId: toStation.id,
   872→          pref: plan.pref,
   873→        }),
   874→      });
   875→      if (res.ok) {
   876→        toast({
   877→          title: "سفر در تاریخچه ثبت شد",
   878→          description: `${fromStation.name} → ${toStation.name}`,
   879→        });
   880→      }
   881→    } catch {
   882→      toast({
   883→        title: "ثبت سفر ناموفق بود",
   884→        description: "به ناوبری روی نقشه ادامه می‌دهیم",
   885→        variant: "destructive",
   886→      });
   887→    } finally {
   888→      // سفر فعال مستقل از ثبت تاریخچه شروع می‌شود:
   889→      // کارت ایستگاه‌به‌ایستگاه در صفحه اصلی + پایش GPS + اعلام صوتی
   890→      const args = buildJourneyFromPlan(plan, fromStation.id, toStation.id, alarm.on, alarm.minutes, voiceOn);
   891→      if (args) {
   892→        useJourney.getState().start(args);
   893→        toast({
   894→          title: "سفر فعال شد",
   895→          description: "در صفحه اصلی ایستگاه‌به‌ایستگاه نمایش داده می‌شود",
   896→        });
   897→        announcer
   898→          .announce(
   899→            `ناوبری آغاز شد. مدت سفر حدود ${toFa(Math.round(plan.totalMinutes))} دقیقه. رسیدن به هر ایستگاه اعلام می‌شود`
   900→          )
   901→          .catch(() => {});
   902→      }
   903→      setStarting(false);
