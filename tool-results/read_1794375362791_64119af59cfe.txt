   501→   501→    async _rawSSMLRequestToFile(dirPath, requestSSML) {
   502→   502→        const { audioStream, metadataStream, requestId } = this._rawSSMLRequest(requestSSML);
   503→   503→        const audioFilePath = (0, utils_1.joinPath)(dirPath, "audio." + Output_1.OUTPUT_EXTENSIONS[this._outputFormat]);
   504→   504→        const metadataFilePath = !!metadataStream ? (0, utils_1.joinPath)(dirPath, "metadata.json") : null;
   505→   505→        try {
   506→   506→            await Promise.all([
   507→   507→                new Promise((resolve, reject) => {
   508→   508→                    const writableAudioFile = audioStream.pipe(fs.createWriteStream(audioFilePath));
   509→   509→                    // .pipe() doesn't forward source errors to the writable, so a truncated stream
   510→   510→                    // would hang here — watch the audio stream directly and reject on its error
   511→   511→                    audioStream.once("error", (e) => {
   512→   512→                        writableAudioFile.destroy();
   513→   513→                        reject(e);
   514→   514→                    });
   515→   515→                    writableAudioFile.once("close", async () => {
   516→   516→                        if (writableAudioFile.bytesWritten > 0) {
   517→   517→                            resolve(audioFilePath);
   518→   518→                        }
   519→   519→                        else {
   520→   520→                            reject(new Error("No audio data received"));
   521→   521→                            fs.rmSync(audioFilePath, { force: true });
   522→   522→                        }
   523→   523→                    });
   524→   524→                    writableAudioFile.once("error", reject);
   525→   525→                }),
   526→   526→                new Promise((resolve, reject) => {
   527→   527→                    if (!metadataFilePath) {
   528→   528→                        return resolve(null);
   529→   529→                    }
   530→   530→                    // get metadata from buffer and combine all MetaData root elements
   531→   531→                    const metadataItems = { Metadata: [] };
   532→   532→                    metadataStream.on("data", (chunk) => {
   533→   533→                        const chunkObj = JSON.parse(chunk.toString());
   534→   534→                        // .Metadata is an array of objects, just combine them
   535→   535→                        metadataItems.Metadata.push(...chunkObj["Metadata"]);
   536→   536→                    });
   537→   537→                    metadataStream.once("close", () => {
   538→   538→                        if (metadataItems.Metadata.length > 0) {
   539→   539→                            // create file if not exists
   540→   540→                            fs.writeFileSync(metadataFilePath, JSON.stringify(metadataItems, null, 2));
   541→   541→                            resolve(metadataFilePath);
   542→   542→                        }
   543→   543→                        else {
   544→   544→                            reject(new Error("No metadata received"));
   545→   545→                            fs.rmSync(metadataFilePath, { force: true });
   546→   546→                        }
   547→   547→                    });
   548→   548→                    metadataStream.once("error", reject);
   549→   549→                }),
   550→   550→            ]);
   551→   551→            return { audioFilePath, metadataFilePath, requestId };
   552→   552→        }
   553→   553→        catch (e) {
   554→   554→            audioStream.destroy();
   555→   555→            metadataStream?.destroy();
   556→   556→            throw e;
   557→   557→        }
   558→   558→    }
