    50→/** چیپ نام خط — با رنگ رسمی خط */
    51→function LineChips({ stationId }: { stationId: string }) {
    52→  const st = getStation(stationId);
    53→  if (!st) return null;
    54→  return (
    55→    <span className="flex gap-1 shrink-0">
    56→      {st.lines.slice(0, 2).map((l) => {
    57→        const line = getLine(l.lineId);
    58→        if (!line) return null;
    59→        return (
    60→          <span
    61→            key={l.lineId}
    62→            className="text-[10px] font-bold px-1.5 py-0.5 rounded whitespace-nowrap"
    63→            style={{ backgroundColor: line.color, color: line.textOnColor }}
    64→          >
    65→            {line.name}
    66→          </span>
    67→        );
    68→      })}
    69→    </span>
    70→  );
    71→}

    75→  const router = useRouter();

    77→
    78→  /** محتوای وابسته به زمان (قطار بعدی) فقط بعد از mount رندر می‌شود */

   129→    const sid = near.id;

   141→      stationName: st.name,
   142→      text: `ایستگاه بعدی: ${adj?.name ?? best.to} (${toFa(best.minutes[0])} دقیقه)`,

   269→
   270→  return (
   271→    <>
   272→      <AppBar />
   273→      <main className="flex-1 w-full max-w-md mx-auto px-4 pt-3 pb-28">

   292→            <div className="relative w-28 h-28 flex items-center justify-center my-1.5">

   295→              <button

   297→                  setVoiceResult(null);
   298→                  if (voice.listening) voice.stop();
   299→                  else voice.start();
   300→                }}
   301→                aria-label="جستجوی صوتی"
