     6→export type CityId = "tehran" | "karaj" | "mashhad" | "isfahan";
     7→
     8→export interface City {
     9→  id: CityId;
    10→  name: string;
    11→  subtitle: string;
    12→  lineIds: string[];
    13→}
    14→
    15→export interface LineDef {
    16→  id: string;
    17→  cityId: CityId;
    18→  number: number;
    19→  name: string; // خط ۱
    20→  color: string;
    21→  textOnColor: string;
    22→  terminals: [string, string]; // [south/west side, north/east side] by array order
    23→  direction: string; // شمال - جنوب
    24→  headway: number; // minutes
    25→  firstTrain: string; // 05:30
    26→  lastTrain: string; // 23:00
    27→  segmentMinutes: number; // avg travel time per adjacent hop
    28→  badges?: string[]; // e.g. فرودگاه
    29→  status?: "normal" | "crowded" | "delay" | "closed";
    30→  statusNote?: string;
    31→}
    32→
    33→export interface StationDef {
    34→  id: string;
    35→  name: string;
    36→  cityId: CityId;
    37→  lines: { lineId: string; order: number }[]; // order: index along the line array
    38→  amenities: AmenityKey[];
    39→  featured?: boolean;
    40→}

   518→/* derive stations: merge same-id entries across lines (interchanges are separate ids) */
   519→function hash(s: string): number {
   520→  let h = 0;
   521→  for (let i = 0; i < s.length; i++) h = (h * 31 + s.charCodeAt(i)) >>> 0;
   522→  return h;
   523→}
   524→
   525→const AMENITY_POOL: AmenityKey[] = [
   526→  "escalator", "elevator", "wc", "atm", "bus", "taxi", "parking", "baby", "bike", "wifi",
   527→];
   528→
   529→export const STATIONS: StationDef[] = RAW.map((r) => {
   530→  const h = hash(r.id);
   531→  const pool = AMENITY_POOL.filter((_, i) => (h >> i) % 3 !== 0);
   532→  const amenities = r.amenities ?? (pool.length >= 3 ? pool.slice(0, 3 + (h % 3)) : ["escalator" as AmenityKey]);
   533→  return {
   534→    id: r.id,
   535→    name: r.name,
   536→    cityId: LINES.find((l) => l.id === r.line)!.cityId,
   537→    lines: [{ lineId: r.line, order: r.order }],
   538→    amenities,
   539→    featured: r.featured,
   540→  };
   541→}).reduce<StationDef[]>((acc, s) => {
   542→  const found = acc.find((x) => x.id === s.id);
   543→  if (found) found.lines.push(...s.lines);
   544→  else acc.push(s);
   545→  return acc;
   546→}, []);

   562→    (a, b) =>
   563→      a.lines.find((l) => l.lineId === lineId)!.order -

   565→  );

   594→export function isWithinService(line: { firstTrain: string; lastTrain: string }, d = new Date()): boolean {
   595→  const [fh, fm] = line.firstTrain.split(":").map(Number);
   596→  const [lh, lm] = line.lastTrain.split(":").map(Number);
   597→  const mins = d.getHours() * 60 + d.getMinutes();
   598→  return mins >= fh * 60 + fm && mins <= lh * 60 + lm;
   599→}

   601→export function nextArrivals(lineId: string, stationId: string, now = Date.now()): number[] {
   602→  const line = getLine(lineId);
   603→  const headwaySec = Math.round(line.headway * 60);
   604→  const phase = hash(stationId + lineId + new Date(now).getHours()) % headwaySec;
   605→  const elapsed = Math.floor(now / 1000) % headwaySec;
   606→  let next = phase - elapsed;
   607→  while (next <= 0) next += headwaySec;
   608→  const late = line.status === "delay" ? 8 * 60 : 0;
   609→  return [next + late, next + headwaySec + late];
   610→}

   612→export type CrowdLevel = "quiet" | "moderate" | "busy";
   613→
   614→export function crowdingNow(lineId: string, now = new Date()): CrowdLevel {
   615→  const h = now.getHours();
   616→  const peak = (h >= 7 && h <= 9) || (h >= 16.5 && h <= 19);
   617→  const mid = (h >= 10 && h <= 16) || (h >= 19.5 && h <= 22);
   618→  const line = getLine(lineId);
   619→  if (line.status === "crowded") return peak ? "busy" : mid ? "busy" : "moderate";
   620→  if (peak) return hash(lineId + h) % 4 === 0 ? "quiet" : "busy";
   621→  if (mid) return hash(lineId + h) % 3 === 0 ? "busy" : "moderate";
   622→  return hash(lineId + h) % 3 === 0 ? "moderate" : "quiet";
   623→}

   646→export function timetableFor(lineId: string): TimetableRow[] {
   647→  const line = getLine(lineId);
   648→  const fmt = (t: string, add = 0) => {
   649→    const [h, m] = t.split(":").map(Number);
   650→    const mm = m + add;
   651→    return toFa(`${String(h + Math.floor(mm / 60)).padStart(2, "0")}:${String(mm % 60).padStart(2, "0")}`);
   652→  };
   653→  return [
   654→    {
   655→      direction: `به سمت ${line.terminals[1]}`,
   656→      first: fmt(line.firstTrain),
   657→      last: fmt(line.lastTrain),
   658→      headwayPeak: toFa(`${line.headway} دقیقه`),
   659→      headwayOff: toFa(`${Math.round(line.headway * 1.6 * 2) / 2} دقیقه`),
   660→    },
   661→    {
   662→      direction: `به سمت ${line.terminals[0]}`,
   663→      first: fmt(line.firstTrain, 5),
   664→      last: fmt(line.lastTrain, -6),
   665→      headwayPeak: toFa(`${Math.round(line.headway * 1.15 * 2) / 2} دقیقه`),
   666→      headwayOff: toFa(`${Math.round(line.headway * 1.8 * 2) / 2} دقیقه`),

   844→export interface RoutePlan {
   845→  legs: RouteLeg[];

   855→const TRANSFER_MIN = 4.5;
   856→const BOARDING_MIN = 1.5;

   858→export function fareFor(stops: number): number {
   859→  if (stops <= 6) return 2800;
   860→  if (stops <= 12) return 3200;
   861→  if (stops <= 20) return 3800;
   862→  return 4500;
   863→}
   864→
   865→interface State {
   866→  station: string;
   867→  line: string | null;
   868→  time: number;
   869→  transfers: number;
   870→  path: { station: string; line: string | null }[];
   871→}
   872→
   873→/** Dijkstra over (station, line) states. */
   874→export function computeRoute(
   875→  fromId: string,
   876→  toId: string,
   877→  pref: "fastest" | "least_transfers" | "quietest" = "fastest",
   878→  now = new Date()
   879→): RoutePlan | null {
   880→  const from = getStation(fromId);
   881→  const to = getStation(toId);
   882→  if (!from || !to || fromId === toId) return null;
   883→
   884→  const seg = (lineId: string) => {
   885→    const base = getLine(lineId).segmentMinutes;
   886→    const crowd = crowdingNow(lineId, now);
   887→    const factor = pref === "quietest" ? (crowd === "busy" ? 1.35 : crowd === "moderate" ? 1.1 : 0.95) : 1;
   888→    return base * factor;
   889→  };
   890→
   891→  const start: State = { station: fromId, line: null, time: 0, transfers: 0, path: [{ station: fromId, line: null }] };
   892→  const best = new Map<string, number>();
   893→  const key = (s: State) => `${s.station}|${s.line ?? "*"}`;
   894→  const queue: State[] = [start];
   895→  best.set(key(start), 0);
   896→
   897→  let done: State | null = null;
   898→  const visited = new Set<string>();
   899→
   900→  while (queue.length) {
   901→    // pick min
   902→    let idx = 0;
   903→    for (let i = 1; i < queue.length; i++) {
   904→      if (queue[i].time < queue[idx].time) idx = i;
   905→    }
   906→    const cur = queue.splice(idx, 1)[0];
   907→    if (visited.has(key(cur))) continue;
   908→    visited.add(key(cur));
   909→
   910→    if (cur.station === toId) {
   911→      done = cur;
   912→      break;
   913→    }
   914→
   915→    const st = getStation(cur.station)!;
   916→    for (const ls of st.lines) {
   917→      // board / continue on line ls.lineId
   918→      const isTransfer = cur.line !== null && cur.line !== ls.lineId;
   919→      const penalty = cur.line === null ? BOARDING_MIN : isTransfer ? TRANSFER_MIN : 0;
   920→      const transferCost =
   921→        pref === "least_transfers" ? (isTransfer ? 40 : 0) : 0;
   922→
   923→      // neighbors along the line
   924→      const list = stationsOfLine(ls.lineId);
   925→      const myIdx = list.findIndex((x) => x.id === cur.station);
   926→      for (const delta of [-1, 1]) {
   927→        const nIdx = myIdx + delta;
   928→        if (nIdx < 0 || nIdx >= list.length) continue;
   929→        const nb = list[nIdx];
   930→        const t = cur.time + seg(ls.lineId) + penalty + transferCost;
   931→        const ns: State = {
   932→          station: nb.id,
   933→          line: ls.lineId,
   934→          time: t,
   935→          transfers: cur.transfers + (isTransfer ? 1 : 0),
   936→          path: [...cur.path, { station: nb.id, line: ls.lineId }],
   937→        };
   938→        const k = key(ns);
   939→        if (!best.has(k) || t < best.get(k)! - 1e-9) {
   940→          best.set(k, t);
   941→          queue.push(ns);
   942→        }
   943→      }
   944→    }
   945→  }
   946→
   947→  if (!done) return null;
   948→
   949→  // compress path into legs
   950→  const legs: RouteLeg[] = [];
   951→  const path = done.path.slice(1);
   952→  let curLine: string | null = null;
   953→  let legStart = path[0].station;
   954→  let prev = fromId;
   955→
   956→  const flush = (endStation: string) => {
   957→    if (!curLine) return;
   958→    const line = getLine(curLine);
   959→    const ids = stationsOfLine(curLine).map((s) => s.id);
   960→    const i1 = ids.indexOf(legStart);
   961→    const i2 = ids.indexOf(endStation);
   962→    const slice = i1 <= i2 ? ids.slice(i1, i2 + 1) : ids.slice(i2, i1 + 1).reverse();
   963→    const names = slice.map((id) => getStation(id)!.name);
   964→    legs.push({
   965→      lineId: curLine,
   966→      lineName: line.name,
   967→      color: line.color,
   968→      textOnColor: line.textOnColor,
   969→      fromId: slice[0],
   970→      fromName: names[0],
   971→      toId: slice[slice.length - 1],
   972→      toName: names[names.length - 1],
   973→      stationIds: slice,
   974→      stationNames: names,
   975→      stops: slice.length - 1,
   976→      minutes: Math.max(1, Math.round((slice.length - 1) * line.segmentMinutes)),
   977→    });
   978→  };
   979→
   980→  for (const step of path) {
   981→    if (curLine === null) {
   982→      curLine = step.line;
   983→      legStart = prev;
   984→    } else if (step.line !== curLine) {
   985→      flush(prev);
   986→      curLine = step.line;
   987→      legStart = prev;
   988→    }
   989→    prev = step.station;
   990→  }
   991→  flush(prev);
   992→
   993→  const totalStops = legs.reduce((a, l) => a + l.stops, 0);
   994→  // زمان نمایشی واقعی = جمع زمان لگ‌ها + هزینه تعویض خط + سوار شدن (بدون وزن بهینه‌سازی)
   995→  const totalMinutes = Math.round(
   996→    legs.reduce((a, l) => a + l.minutes, 0) + (legs.length - 1) * TRANSFER_MIN + BOARDING_MIN
   997→  );
   998→  const transferStations = legs.slice(0, -1).map((l) => l.toName);
   999→  const busyLeg = legs.find((l) => crowdingNow(l.lineId, now) === "busy");
  1000→
  1001→  return {
  1002→    legs,
  1003→    totalMinutes,
  1004→    totalStations: totalStops + 1,
  1005→    transfers: legs.length - 1,
  1006→    transferStations,
  1007→    fare: fareFor(totalStops),
  1008→    pref,
  1009→    crowdNote: busyLeg
  1010→      ? `ازدحام بالا روی ${busyLeg.lineName} در ساعات اوج`
  1011→      : "سرویس‌دهی همه خطوط مسیر عادی و روان است",
  1012→  };
  1013→}

  1024→  facility?: AmenityKey;
  1025→  cityId?: CityId;
  1026→}): StationDef[] {

  1035→      (s) => s.name.toLowerCase().includes(nq) || s.name.replace(/ /g, "").includes(nq.replace(/ /g, ""))
  1036→    );
  1037→  }
  1038→  return res;
