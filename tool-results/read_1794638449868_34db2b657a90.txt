    47→    47→
    48→    48→function getSession(req: Request): AdminRow | null {
    49→    49→  const auth = req.headers.get("Authorization") || "";
    50→    50→  const token = auth.startsWith("Bearer ") ? auth.slice(7) : "";
    51→    51→  if (!token) return null;
    52→    52→  const row = db
    53→    53→    .query(
    54→    54→      `SELECT a.* FROM sessions s JOIN admins a ON a.id = s.admin_id
    55→    55→       WHERE s.token = ? AND s.expires_at > ? AND a.is_active = 1`
    56→    56→    )
    57→    57→    .get(token, now()) as AdminRow | undefined;
    58→    58→  return row ?? null;
    59→    59→}
    60→    60→
    61→    61→function requireRole(admin: AdminRow, role: Role): Response | null {
    62→    62→  if (RANK[admin.role] < RANK[role]) {
    63→    63→    return fail("FORBIDDEN", "سطح دسترسی شما برای این عملیات کافی نیست", 403);
    64→    64→  }
    65→    65→  return null;
    66→    66→}
    67→    67→
    68→    68→function touchSession(adminId: number, token: string, req: Request) {
    69→    69→  const expires = new Date(Date.now() + SESSION_HOURS * 3600 * 1000).toISOString();
    70→    70→  db.prepare(
    71→    71→    "INSERT INTO sessions (token, admin_id, created_at, expires_at, ip, user_agent) VALUES (?,?,?,?,?,?) ON CONFLICT(token) DO UPDATE SET expires_at = excluded.expires_at"
    72→    72→  ).run(token, adminId, now(), expires, clientIp(req), req.headers.get("User-Agent") ?? "");
    73→    73→  db.prepare("DELETE FROM sessions WHERE expires_at < ?").run(now());
    74→    74→}

    78→    78→export async function handleApi(req: Request, url: URL): Promise<Response> {
    79→    79→  bumpApiStats();
    80→    80→  const path = url.pathname.replace(/^\/api\/v1\/?/, "").replace(/\/$/, "");
    81→    81→  const method = req.method.toUpperCase();
    82→    82→  const ip = clientIp(req);
    83→    83→
    84→    84→  let body: any = {};
    85→    85→  if (method === "POST" || method === "PUT" || method === "PATCH") {
    86→    86→    try {
    87→    87→      body = await req.json();
    88→    88→    } catch {
    89→    89→      body = {};
    90→    90→    }
    91→    91→  }
    92→    92→
    93→    93→  try {
    94→    94→    return await route(path, method, url, req, body, ip);
    95→    95→  } catch (e: any) {
    96→    96→    console.error("[darmasir-admin][api]", e?.message ?? e);
    97→    97→    return fail("INTERNAL", "خطای داخلی سرور پنل", 500);
    98→    98→  }
    99→    99→}
   100→   100→
   101→   101→async function route(
   102→   102→  path: string,
   103→   103→  method: string,
   104→   104→  url: URL,
   105→   105→  req: Request,
   106→   106→  body: any,
   107→   107→  ip: string

   111→   111→
   112→   112→  /* ---------- auth ---------- */
   113→   113→  if (path === "auth/login" && method === "POST") {
   114→   114→    if (!loginAllowed(ip))
   115→   115→      return fail("RATE_LIMIT", "تلاش‌های ناموفق زیاد است؛ چند دقیقه بعد دوباره امتحان کنید", 429);

   123→   123→      audit(null, username || "ناشناس", "login_failed", `نام کاربری: ${username}`, ip);

   125→   125→    }

   132→   132→      token,

   139→   139→  if (path === "auth/me" && method === "GET") {

   143→   143→  if (path === "auth/change-password" && method === "POST") {

   149→   149→    db.prepare("UPDATE admins SET password_hash = ? WHERE id = ?").run(hashPassword(next), admin.id);
   150→   150→    audit(admin.id, admin.display_name, "password_change", "تغییر گذرواژه شخصی", ip);
   151→   151→    return ok({ changed: true });
   152→   152→  }
   153→   153→
   154→   154→  if (path === "auth/logout" && method === "POST") {
   155→   155→    const auth = req.headers.get("Authorization") || "";
   156→   156→    const token = auth.startsWith("Bearer ") ? auth.slice(7) : "";
   157→   157→    db.prepare("DELETE FROM sessions WHERE token = ?").run(token);
   158→   158→    audit(admin.id, admin.display_name, "logout", "خروج از پنل", ip);
   159→   159→    return ok({ logged_out: true });
   160→   160→  }
   161→   161→
   162→   162→  /* ---------- dashboard ---------- */

   164→   164→    const q = (sql: string) => (db.query(sql).get() as any).c ?? 0;
   165→   165→    const days: { day: string; count: number }[] = [];
   166→   166→    for (let i = 6; i >= 0; i--) {
   167→   167→      const d = new Date(Date.now() - i * 86400000).toISOString().slice(0, 10);
   168→   168→      days.push({ day: d, count: 0 });
   169→   169→    }
   170→   170→    (db.query("SELECT day, count FROM api_stats WHERE day >= ?").all(
   171→   171→      new Date(Date.now() - 6 * 86400000).toISOString().slice(0, 10)
   172→   172→    ) as any[]).forEach((r) => {
   173→   173→      const f = days.find((x) => x.day === r.day);
   174→   174→      if (f) f.count = r.count;
   175→   175→    });
   176→   176→    return ok({
   177→   177→      users_total: q("SELECT COUNT(*) AS c FROM app_users"),
   178→   178→      users_active: q("SELECT COUNT(*) AS c FROM app_users WHERE status = 'active'"),
   179→   179→      trips_total: q("SELECT COALESCE(SUM(trips_count),0) AS c FROM app_users"),
   180→   180→      lines_active: q("SELECT COUNT(*) AS c FROM lines WHERE is_active = 1"),

   183→   183→        db

   185→   185→          .get(now()) as any
   186→   186→      ).c,
   187→   187→      api_last7: days,
   188→   188→      audit_recent: db
   189→   189→        .query("SELECT action, admin_name, details, created_at FROM audit_log ORDER BY id DESC LIMIT 8")
   190→   190→        .all(),
   191→   191→      services: [

   193→   193→        {
   194→   194→          name: "سرویس صوتی",

   196→   196→          note: (db.query("SELECT value FROM settings WHERE key = 'voice_service'").get() as any)?.value ?? "-",
   197→   197→        },
   198→   198→        { name: "دروازه API پنل", status: "ok", note: "پورت ۳۰۳۵" },
   199→   199→      ],
   200→   200→    });
   201→   201→  }
   202→   202→
   203→   203→  /* ---------- users (کاربران اپ) ---------- */
   204→   204→  if (seg[0] === "users") {
   205→   205→    if (method === "GET") {
   206→   206→      if (seg[1] && !isNaN(Number(seg[1]))) {
   207→   207→        const u = db
   208→   208→          .query("SELECT id, phone, display_name, status, trips_count, created_at, last_seen FROM app_users WHERE id = ?")
   209→   209→          .get(Number(seg[1]));
   210→   210→        if (!u) return fail("NOT_FOUND", "کاربر یافت نشد", 404);
   211→   211→        return ok(u);
   212→   212→      }
   213→   213→      const q = str(url.searchParams.get("q") ?? "", 60);
   214→   214→      const status = str(url.searchParams.get("status") ?? "", 20);
   215→   215→      let sql = "SELECT id, phone, display_name, status, trips_count, created_at, last_seen FROM app_users WHERE 1=1";
   216→   216→      const args: any[] = [];
   217→   217→      if (q) {
   218→   218→        sql += " AND (display_name LIKE ? OR phone LIKE ?)";
   219→   219→        args.push(`%${q}%`, `%${q}%`);
   220→   220→      }
   221→   221→      if (status) {
   222→   222→        sql += " AND status = ?";
   223→   223→        args.push(status);
   224→   224→      }
   225→   225→      sql += " ORDER BY id DESC LIMIT 200";
   226→   226→      return ok(db.prepare(sql).all(...args));
   227→   227→    }
   228→   228→    const guard = requireRole(admin, "operator");
   229→   229→    if (guard) return guard;
   230→   230→    if (path === "users/bulk-status" && method === "POST") {
   231→   231→      const status = ["active", "limited", "disabled"].includes(body.status) ? body.status : null;
   232→   232→      const ids: number[] = Array.isArray(body.ids) ? body.ids.map(Number).filter(Boolean) : [];
   233→   233→      if (!status || !ids.length) return fail("BAD_INPUT", "وضعیت و فهرست کاربران الزامی است");
   234→   234→      const stmt = db.prepare("UPDATE app_users SET status = ? WHERE id = ?");
   235→   235→      let n = 0;
   236→   236→      for (const id of ids) { stmt.run(status, id); n++; }
   237→   237→      changeLog("app_user", null, "bulk_status", admin.display_name, `${n} کاربر → ${status}`);
   238→   238→      audit(admin.id, admin.display_name, "user_bulk_status", `${n} کاربر → ${status}`, ip);
   239→   239→      return ok({ updated: n });
   240→   240→    }
   241→   241→    if (seg[1] === "trips" && seg[2] && method === "POST") {
   242→   242→      const id = Number(seg[2]);
   243→   243→      const u = db.prepare("SELECT * FROM app_users WHERE id = ?").get(id) as any;
   244→   244→      if (!u) return fail("NOT_FOUND", "کاربر یافت نشد", 404);
   245→   245→      const delta = Math.round(Number(body.delta) || 0);
   246→   246→      if (!delta) return fail("BAD_INPUT", "مقدار تغییر سفر الزامی است");
   247→   247→      const next = Math.max(0, (u.trips_count || 0) + delta);

   264→   264→        return ok({ id: Number(r.lastInsertRowid) }, 201);
   265→   265→      } catch {
   266→   266→        return fail("DUPLICATE", "این شماره قبلاً ثبت شده است", 409);
   267→   267→      }
   268→   268→    }
   269→   269→    const id = Number(seg[1]);

   285→   285→        patch.push("status = ?");
   286→   286→        args.push(body.status);

   293→   293→      return ok({ updated: true });
   294→   294→    }

   300→   300→    }

   312→   312→      return ok(rows);
   313→   313→    }

   315→   315→    if (guard) return guard;

   348→   348→    if (method === "POST") {

   710→   710→      for (const s of stations) {
   711→   711→        if (exists.has(s.name)) continue;
   712→   712→        ins.run(s.name, s.name, 5);
   713→   713→        added++;
   714→   714→      }
   715→   715→      changeLog("voice_lexicon", null, "bulk_create", admin.display_name, `${added} ایستگاه وارد شد`);
   716→   716→      audit(admin.id, admin.display_name, "lexicon_import", `${added} ایستگاه`, ip);
   717→   717→      return ok({ added, total_stations: stations.length });
   718→   718→    }
   719→   719→    if (method === "POST") {
   720→   720→      const spoken = str(body.spoken, 120);
   721→   721→      const written = str(body.written, 120);
   722→   722→      if (!spoken || !written) return fail("BAD_INPUT", "هر دو مقدار گفتاری و نوشتاری الزامی است");
   723→   723→      const r = db
   724→   724→        .prepare("INSERT INTO voice_lexicon (spoken, written, priority) VALUES (?,?,?)")
   725→   725→        .run(spoken, written, Number(body.priority) || 0);
   726→   726→      changeLog("voice_lexicon", Number(r.lastInsertRowid), "create", admin.display_name, `${spoken} → ${written}`);
   727→   727→      audit(admin.id, admin.display_name, "lexicon_create", spoken, ip);
   728→   728→      return ok({ id: Number(r.lastInsertRowid) }, 201);
   729→   729→    }
   730→   730→    const id = Number(seg[1]);
   731→   731→    if (!id) return fail("NOT_FOUND", "مدخل یافت نشد", 404);
   732→   732→    if (method === "PUT") {
   733→   733→      const cur = db.prepare("SELECT * FROM voice_lexicon WHERE id = ?").get(id) as any;
   734→   734→      if (!cur) return fail("NOT_FOUND", "مدخل یافت نشد", 404);
   735→   735→      const next = { ...cur, ...body };
   736→   736→      db.prepare("UPDATE voice_lexicon SET spoken=?, written=?, priority=? WHERE id=?").run(
   737→   737→        str(next.spoken, 120) || cur.spoken,
   738→   738→        str(next.written, 120) || cur.written,
   739→   739→        Number(next.priority) || 0,
   740→   740→        id
   741→   741→      );
   742→   742→      changeLog("voice_lexicon", id, "update", admin.display_name, next.spoken);
   743→   743→      return ok({ updated: true });
   744→   744→    }
   745→   745→    if (method === "DELETE") {
   746→   746→      db.prepare("DELETE FROM voice_lexicon WHERE id = ?").run(id);
   747→   747→      changeLog("voice_lexicon", id, "delete", admin.display_name);
   748→   748→      return ok({ deleted: true });
   749→   749→    }
   750→   750→  }
   751→   751→
   752→   752→  /* ---------- settings ---------- */
   753→   753→  if (path === "settings") {
   754→   754→    if (method === "GET") {
   755→   755→      const rows = db.query("SELECT key, value, updated_at FROM settings").all();
   756→   756→      return ok(Object.fromEntries(rows.map((r: any) => [r.key, r])));
   757→   757→    }
   758→   758→    const guard = requireRole(admin, "super");
   759→   759→    if (guard) return guard;
   760→   760→    if (method === "PUT") {
   761→   761→      const allowed = [
   762→   762→        "app_name",
   763→   763→        "app_version",
   764→   764→        "welcome_message",
   765→   765→        "maintenance_mode",
   766→   766→        "voice_enabled",
   767→   767→        "voice_service",
   768→   768→        "support_message",
   769→   769→        "support_phone",
   770→   770→        "ads_rotation_seconds",
   771→   771→        "max_ads_per_placement",
   772→   772→        "force_update_version",
   773→   773→        "store_link",
   774→   774→        "default_city",
   775→   775→      ];
   776→   776→      let changed = 0;
   777→   777→      for (const [k, v] of Object.entries(body)) {
   778→   778→        if (!allowed.includes(k)) continue;
   779→   779→        db.prepare("INSERT INTO settings (key, value, updated_at) VALUES (?,?,?) ON CONFLICT(key) DO UPDATE SET value = excluded.value, updated_at = excluded.updated_at").run(
   780→   780→          k,
   781→   781→          String(v).slice(0, 300),
   782→   782→          now()
   783→   783→        );
   784→   784→        changed++;
   785→   785→      }
   786→   786→      if (!changed) return fail("BAD_INPUT", "هیچ کلید مجازی ارسال نشده است");
   787→   787→      audit(admin.id, admin.display_name, "settings_update", JSON.stringify(Object.keys(body)), ip);
   788→   788→      return ok({ updated: changed });
   789→   789→    }
   790→   790→  }
   791→   791→
   792→   792→  /* ---------- reports ---------- */
   793→   793→  if (path === "reports" && method === "GET") {
   794→   794→    const range = Math.min(Math.max(Number(url.searchParams.get("days")) || 14, 7), 90);
   795→   795→    const since = new Date(Date.now() - (range - 1) * 86400000).toISOString().slice(0, 10);
   796→   796→    const stats = db.query("SELECT day, count FROM api_stats WHERE day >= ? ORDER BY day DESC LIMIT ?").all(since, range);
   797→   797→    const adsDaily = db
   798→   798→      .query(
   799→   799→        "SELECT day, SUM(impressions) AS impressions, SUM(clicks) AS clicks FROM ad_stats_daily WHERE day >= ? GROUP BY day ORDER BY day"
   800→   800→      )
   801→   801→      .all(since);
   802→   802→    const adsRanking = db
   803→   803→      .query(
   804→   804→        `SELECT id, title, placement, is_active, impressions, clicks,
   805→   805→                CASE WHEN impressions > 0 THEN ROUND(clicks * 100.0 / impressions, 2) ELSE 0 END AS ctr
   806→   806→         FROM ads ORDER BY ctr DESC, impressions DESC`
   807→   807→      )
   808→   808→      .all();
   809→   809→    const changes = db.query("SELECT entity, action, admin_name, details, created_at FROM change_log ORDER BY id DESC LIMIT 30").all();
   810→   810→    const sessions = (db.query("SELECT COUNT(*) AS c FROM sessions WHERE expires_at > ?").get(now()) as any).c;
   811→   811→    const usersByStatus = db.query("SELECT status, COUNT(*) AS count FROM app_users GROUP BY status").all();
   812→   812→    const topStations = db
   813→   813→      .query("SELECT s.name, l.name AS line_name, l.color FROM stations s JOIN lines l ON l.id = s.line_id WHERE s.is_active = 1 LIMIT 8")
   814→   814→      .all();
   815→   815→    const totals = (db.query("SELECT COALESCE(SUM(impressions),0) AS imp, COALESCE(SUM(clicks),0) AS clk FROM ad_stats_daily WHERE day >= ?").get(since) as any);
   816→   816→    return ok({
   817→   817→      range,
   818→   818→      api_stats: stats,
   819→   819→      ads_daily: adsDaily,
   820→   820→      ads_ranking: adsRanking,
   821→   821→      users_by_status: usersByStatus,
   822→   822→      top_stations: topStations,
   823→   823→      ads_totals: { impressions: totals.imp, clicks: totals.clk },
   824→   824→      changes,
   825→   825→      active_sessions: sessions,
   826→   826→    });
   827→   827→  }
   828→   828→
   829→   829→  /* ---------- security (فقط مدیر کل) ---------- */
   830→   830→  if (seg[0] === "security" && method === "GET") {
   831→   831→    const guard = requireRole(admin, "super");
   832→   832→    if (guard) return guard;
   833→   833→    return ok({

   835→   835→        .query("SELECT id, username, display_name, role, is_active, created_at, last_login FROM admins ORDER BY id")
   836→   836→        .all(),
   837→   837→      sessions: db
   838→   838→        .query(
   839→   839→          `SELECT s.token, s.created_at, s.expires_at, s.ip, s.user_agent, a.username, a.display_name
   840→   840→           FROM sessions s JOIN admins a ON a.id = s.admin_id

   842→   842→        )

   845→   845→    });
   846→   846→  }
   847→   847→
   848→   848→  /* ---------- ابطال نشست (فقط مدیر کل) ---------- */

   870→   870→        return fail("BAD_INPUT", "نام کاربری، نام نمایشی و گذرواژه (حداقل ۸ کاراکتر) الزامی است");

   872→   872→        const r = db
   873→   873→          .prepare("INSERT INTO admins (username, password_hash, display_name, role, created_at) VALUES (?,?,?,?,?)")
   874→   874→          .run(username, hashPassword(password), displayName, role, now());
   875→   875→        audit(admin.id, admin.display_name, "admin_create", username, ip);
