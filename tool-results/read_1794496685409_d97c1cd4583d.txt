    23→/* رنگ روشن برای نشان خط داخل پیشنهادها (مثل دیزاین: bg-red-50/text-red-800) */
    24→const LINE_CHIP: Record<string, string> = {
    25→  l1: "bg-red-50 text-red-800",
    26→  l2: "bg-blue-50 text-blue-800",
    27→  l3: "bg-cyan-50 text-cyan-800",
    28→  l4: "bg-yellow-50 text-yellow-800",
    29→  l5: "bg-green-50 text-green-800",
    30→  l6: "bg-pink-50 text-pink-800",
    31→  l7: "bg-purple-50 text-purple-800",
    32→  k1: "bg-sky-50 text-sky-800",
    33→  m1: "bg-orange-50 text-orange-800",
    34→  m2: "bg-teal-50 text-teal-800",
    35→  e1: "bg-lime-50 text-lime-800",
    36→};
    37→
    38→function chipClass(station: StationDef) {
    39→  if (station.lines.length > 1) return "bg-cyan-50 text-cyan-800";
    40→  return LINE_CHIP[station.lines[0]?.lineId] ?? "bg-surface-container text-on-surface-variant";
    41→}
    42→
    43→function chipLabel(station: StationDef) {
    44→  if (station.lines.length > 1)
    45→    return `تقاطع ${station.lines.map((l) => toFa(getLine(l.lineId)?.number ?? 0)).join(" و ")}`;
    46→  return `خط ${toFa(getLine(station.lines[0]?.lineId)?.number ?? 0)}`;
    47→}

    82→  /* ---------- resolve + محاسبه زنده مسیر ---------- */

    89→      const candidates: StationDef[] = [];
    90→      for (const s of [

    92→        ...searchStations({ q }),
    93→        ...STATIONS.filter((st) => st.cityId === city),

   108→      const words = q.split(/[\s()]+/).filter((w) => w.length > 1);
   109→      const token = candidates.find((s) => {
   110→        const sn = s.name.replace(/[()]/g, " ");
   111→        return words.length > 0 && words.every((w) => sn.includes(w));
   112→      });

   333→      });
   334→      setOriginText(from.name);
   335→      setDestText(to.name);
   336→      setRouteDialog(null);
   337→      toast({
   338→        title: "مسیر ذخیره شد",
   339→        description: `${routeDialog === "home" ? "مسیر خانه" : "مسیر محل کار"}: به سمت ${to.name}`,
