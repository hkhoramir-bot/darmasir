    15→export interface LineDef {
    16→  id: string;
    17→  cityId: CityId;
    18→  number: number;
    19→  name: string; // خط ۱
    20→  color: string;
    21→  textOnColor: string;
    22→  terminals: [string, string]; // [south/west side, north/east side] by array order
    23→  direction: string; // شمال - جنوب
    24→  headway: number; // minutes — فاصله پیش‌فرض (fallback)
    25→  headwayPeak?: number; // فاصله حرکت در ساعات اوج — جدول رسمی ۱۴۰۴
    26→  headwayOff?: number; // فاصله حرکت در ساعات غیراوج — جدول رسمی ۱۴۰۴
    27→  firstTrain: string; // 04:30
    28→  lastTrain: string; // 22:30
    29→  segmentMinutes: number; // avg travel time per adjacent hop
    30→  badges?: string[]; // e.g. فرودگاه
    31→  status?: "normal" | "crowded" | "delay" | "closed";
    32→  statusNote?: string;
    33→}
    34→
    35→export interface StationDef {
    36→  id: string;
    37→  name: string;
    38→  cityId: CityId;
    39→  lines: { lineId: string; order: number }[]; // order: index along the line array
    40→  amenities: AmenityKey[];
    41→  featured?: boolean;
    42→}

   543→/* derive stations: merge same-id entries across lines (interchanges are separate ids) */
   544→function hash(s: string): number {
   545→  let h = 0;
   546→  for (let i = 0; i < s.length; i++) h = (h * 31 + s.charCodeAt(i)) >>> 0;
   547→  return h;
   548→}

   550→const AMENITY_POOL: AmenityKey[] = [
   551→  "escalator", "elevator", "wc", "atm", "bus", "taxi", "parking", "baby", "bike", "wifi",
   552→];
   553→
   554→export const STATIONS: StationDef[] = RAW.map((r) => {
   555→  const h = hash(r.id);
   556→  const pool = AMENITY_POOL.filter((_, i) => (h >> i) % 3 !== 0);
   557→  const amenities = r.amenities ?? (pool.length >= 3 ? pool.slice(0, 3 + (h % 3)) : ["escalator" as AmenityKey]);
   558→  return {
   559→    id: r.id,
   560→    name: r.name,
   561→    cityId: LINES.find((l) => l.id === r.line)!.cityId,
   562→    lines: [{ lineId: r.line, order: r.order }],
   563→    amenities,
   564→    featured: r.featured,
   565→  };
   566→}).reduce<StationDef[]>((acc, s) => {
   567→  const found = acc.find((x) => x.id === s.id);
   568→  if (found) found.lines.push(...s.lines);
   569→  else acc.push(s);
   570→  return acc;
   571→}, []);

   616→/* ------------------------------ موتور جدول حرکت رسمی (۱۴۰۴) ------------------------------ */
   617→/**
   618→ * مترو تهران فید لحظه‌ای عمومی ندارد؛ زمان ورود قطارها بر اساس
   619→ * جدول حرکت رسمی شرکت مترو تهران (سامانه metro.tehran.ir) محاسبه می‌شود:
   620→ * ساعات سرویس: شنبه تا چهارشنبه ۵:۳۰ تا ۲۲:۳۰ — پنجشنبه ۶:۰۰ — جمعه ۶:۳۰
   621→ * و سرفاصله اوج/غیراوج رسمی هر خط.
   622→ */
   623→
   624→/** بازه‌های ساعات اوج (دقیقه از نیمه‌شب) — صبح و عصر */
   625→const PEAK_WINDOWS: [number, number][] = [
   626→  [420, 570], // ۷:۰۰ تا ۹:۳۰
   627→  [990, 1140], // ۱۶:۳۰ تا ۱۹:۰۰
   628→];
   629→
   630→/** فاصله حرکت قطارها در یک لحظه از روز — اوج / غیراوج */
   631→export function headwayAt(line: LineDef, mins: number): number {
   632→  const peak = PEAK_WINDOWS.some(([a, b]) => mins >= a && mins < b);
   633→  return peak ? (line.headwayPeak ?? line.headway) : (line.headwayOff ?? line.headway);
   634→}
   635→
   636→/**
   637→ * ساعت شروع سرویس بر اساس روز هفته (الگوی رسمی ۱۴۰۴):
   638→ * شنبه تا چهارشنبه: ۵:۳۰ | پنجشنبه: ۶:۰۰ | جمعه: ۶:۳۰
   639→ */
   640→export function serviceStartMin(line: LineDef, d = new Date()): number {
   641→  const [fh, fm] = line.firstTrain.split(":").map(Number);
   642→  let start = fh * 60 + fm;
   643→  const day = d.getDay(); // ۰ یکشنبه … ۳ چهارشنبه، ۴ پنجشنبه، ۵ جمعه، ۶ شنبه
   644→  if (day === 4) start += 30; // پنجشنبه ۶:۰۰
   645→  else if (day === 5) start += 60; // جمعه ۶:۳۰
   646→  return start;
   647→}
   648→
   649→/** ساعت حرکت آخرین قطار از سرخط */
   650→export function serviceEndMin(line: LineDef): number {
   651→  const [lh, lm] = line.lastTrain.split(":").map(Number);
   652→  return lh * 60 + lm;
   653→}

   655→/** آیا الان داخل ساعات سرویس‌دهی خط هستیم؟ */
   656→export function isWithinService(line: { firstTrain: string; lastTrain: string }, d = new Date()): boolean {
   657→  const mins = d.getHours() * 60 + d.getMinutes();
   658→  return mins >= serviceStartMin(line as LineDef, d) && mins <= serviceEndMin(line as LineDef);
   659→}

   661→/** زمان‌های حرکت قطارها از سرخط با فاصله اوج/غیراوج */
   662→function departureGrid(line: LineDef, fromMin: number, toMin: number): number[] {
   663→  const out: number[] = [];
   664→  let t = fromMin;
   665→  while (t <= toMin + 1e-9) {
   666→    out.push(t);
   667→    t += headwayAt(line, t);
   668→  }
   669→  return out;
   670→}

   688→  lineId: string,
   689→  stationId: string,
   690→  dir: 0 | 1,
   691→  now: number
   692→): ArrivalDirDetail {
   693→  const line = getLine(lineId);
   694→  const list = stationsOfLine(lineId);
   695→  const idx = list.findIndex((s) => s.id === stationId);
   696→  if (idx < 0) return { minutes: [], at: [], outOfService: true };
   697→
   698→  const d = new Date(now);
   699→  const nowMin = d.getHours() * 60 + d.getMinutes() + d.getSeconds() / 60;
   700→  const start = serviceStartMin(line, d);
   701→  const end = serviceEndMin(line);
   702→  // تا ۴۵ دقیقه پس از پایان، رسیدن آخرین قطارها به ایستگاه‌های انتهایی ادامه دارد
   703→  if (nowMin < start || nowMin > end + 45) {
   704→    return { minutes: [], at: [], outOfService: true };
   705→  }
   706→
   707→  const hops = dir === 0 ? idx : list.length - 1 - idx;
   708→  const travel = hops * line.segmentMinutes;
   709→  // ۴۰ ثانیه توقف در ایستگاه برای سوار و پیاده شدن
   710→  const deps = departureGrid(line, start, end).filter((dep) => dep + travel >= nowMin + 0.6);
   711→  const next2 = deps.slice(0, 2);
   712→  return {
   713→    minutes: next2.map((dep) => Math.max(1, Math.round(dep + travel - nowMin))),
   714→    at: next2.map((dep) => fmtClock(dep + travel)),
   715→    outOfService: false,
   716→  };
   717→}

   738→ * ورودهای بعدی به هر دو جهت — با نام درستِ مقصد هر جهت و بر اساس جدول حرکت رسمی.
   739→ * ایستگاه سرخط فقط یک جهت دارد؛ ایستگاه میانی هر دو جهت.

   741→export function nextArrivalsBoth(
   742→  lineId: string,
   743→  stationId: string,
   744→  now = Date.now()
   745→): DirectionalArrival[] {
   746→  const line = getLine(lineId);
   747→  const list = stationsOfLine(lineId);
   748→  const idx = list.findIndex((s) => s.id === stationId);
   749→  if (idx < 0) return [];
   750→
   751→  const out: DirectionalArrival[] = [];
   752→  const mk = (to: string, dir: 0 | 1): DirectionalArrival => {
   753→    const r = scheduleArrivalsDir(lineId, stationId, dir, now);
   754→    const crowd = crowdingNow(lineId, new Date(now));
   755→    return {
   756→      to,
   757→      minutes: r.minutes,
   758→      at: r.at,
   759→      outOfService: r.outOfService,
   760→      crowd,
   761→      crowdLabel: CROWD_LABEL[crowd],
   762→    };
   763→  };
   764→
   765→  // به سمت terminals[1] — مگر این ایستگاه خودش همان سرخط باشد
   766→  if (idx < list.length - 1) out.push(mk(line.terminals[1], 0));
   767→  // به سمت terminals[0] — مگر این ایستگاه خودش همان سرخط باشد
   768→  if (idx > 0) out.push(mk(line.terminals[0], 1));
   769→  return out;
   770→}

   772→export type CrowdLevel = "quiet" | "moderate" | "busy";
   773→
   774→export function crowdingNow(lineId: string, now = new Date()): CrowdLevel {
   775→  const h = now.getHours();
   776→  const peak = (h >= 7 && h <= 9) || (h >= 16.5 && h <= 19);
   777→  const mid = (h >= 10 && h <= 16) || (h >= 19.5 && h <= 22);
   778→  const line = getLine(lineId);
   779→  if (line.status === "crowded") return peak ? "busy" : mid ? "busy" : "moderate";
   780→  if (peak) return hash(lineId + h) % 4 === 0 ? "quiet" : "busy";
   781→  if (mid) return hash(lineId + h) % 3 === 0 ? "busy" : "moderate";
   782→  return hash(lineId + h) % 3 === 0 ? "moderate" : "quiet";
   783→}

  1032→/** Dijkstra over (station, line) states. */
  1033→export function computeRoute(
  1034→  fromId: string,
  1035→  toId: string,
  1036→  pref: "fastest" | "least_transfers" | "quietest" = "fastest",
  1037→  now = new Date()
  1038→): RoutePlan | null {
  1039→  const from = getStation(fromId);
  1040→  const to = getStation(toId);
  1041→  if (!from || !to || fromId === toId) return null;
  1042→
  1043→  const seg = (lineId: string) => {
  1044→    const base = getLine(lineId).segmentMinutes;
  1045→    const crowd = crowdingNow(lineId, now);
  1046→    const factor = pref === "quietest" ? (crowd === "busy" ? 1.35 : crowd === "moderate" ? 1.1 : 0.95) : 1;
  1047→    return base * factor;
  1048→  };
  1049→
  1050→  const start: State = { station: fromId, line: null, time: 0, transfers: 0, path: [{ station: fromId, line: null }] };
  1051→  const best = new Map<string, number>();
  1052→  const key = (s: State) => `${s.station}|${s.line ?? "*"}`;
  1053→  const queue: State[] = [start];
  1054→  best.set(key(start), 0);
  1055→
  1056→  let done: State | null = null;
  1057→  const visited = new Set<string>();
  1058→
  1059→  while (queue.length) {
  1060→    // pick min
  1061→    let idx = 0;
  1062→    for (let i = 1; i < queue.length; i++) {
  1063→      if (queue[i].time < queue[idx].time) idx = i;
  1064→    }
  1065→    const cur = queue.splice(idx, 1)[0];
  1066→    if (visited.has(key(cur))) continue;
  1067→    visited.add(key(cur));
  1068→
  1069→    if (cur.station === toId) {
  1070→      done = cur;
  1071→      break;
  1072→    }
  1073→
  1074→    const st = getStation(cur.station)!;
  1075→    for (const ls of st.lines) {
  1076→      // board / continue on line ls.lineId
  1077→      const isTransfer = cur.line !== null && cur.line !== ls.lineId;
  1078→      const penalty = cur.line === null ? BOARDING_MIN : isTransfer ? TRANSFER_MIN : 0;
  1079→      const transferCost =
  1080→        pref === "least_transfers" ? (isTransfer ? 40 : 0) : 0;
  1081→
  1082→      // neighbors along the line
  1083→      const list = stationsOfLine(ls.lineId);
  1084→      const myIdx = list.findIndex((x) => x.id === cur.station);
  1085→      for (const delta of [-1, 1]) {
  1086→        const nIdx = myIdx + delta;
  1087→        if (nIdx < 0 || nIdx >= list.length) continue;
  1088→        const nb = list[nIdx];
  1089→        const t = cur.time + seg(ls.lineId) + penalty + transferCost;
  1090→        const ns: State = {
  1091→          station: nb.id,
  1092→          line: ls.lineId,
  1093→          time: t,
  1094→          transfers: cur.transfers + (isTransfer ? 1 : 0),
  1095→          path: [...cur.path, { station: nb.id, line: ls.lineId }],
  1096→        };
  1097→        const k = key(ns);
  1098→        if (!best.has(k) || t < best.get(k)! - 1e-9) {
  1099→          best.set(k, t);
  1100→          queue.push(ns);
  1101→        }
  1102→      }
  1103→    }
  1104→  }
  1105→
  1106→  if (!done) return null;
  1107→
  1108→  // compress path into legs
  1109→  const legs: RouteLeg[] = [];
  1110→  const path = done.path.slice(1);
  1111→  let curLine: string | null = null;
  1112→  let legStart = path[0].station;
  1113→  let prev = fromId;
  1114→
  1115→  const flush = (endStation: string) => {
  1116→    if (!curLine) return;
  1117→    const line = getLine(curLine);
  1118→    const ids = stationsOfLine(curLine).map((s) => s.id);
  1119→    const i1 = ids.indexOf(legStart);
  1120→    const i2 = ids.indexOf(endStation);
  1121→    const slice = i1 <= i2 ? ids.slice(i1, i2 + 1) : ids.slice(i2, i1 + 1).reverse();
  1122→    const names = slice.map((id) => getStation(id)!.name);
  1123→    legs.push({
  1124→      lineId: curLine,
  1125→      lineName: line.name,
  1126→      color: line.color,
  1127→      textOnColor: line.textOnColor,
  1128→      fromId: slice[0],
  1129→      fromName: names[0],
  1130→      toId: slice[slice.length - 1],
  1131→      toName: names[names.length - 1],
  1132→      stationIds: slice,
  1133→      stationNames: names,
  1134→      stops: slice.length - 1,
  1135→      minutes: Math.max(1, Math.round((slice.length - 1) * line.segmentMinutes)),
  1136→    });
  1137→  };
  1138→
  1139→  for (const step of path) {
  1140→    if (curLine === null) {
  1141→      curLine = step.line;
  1142→      legStart = prev;
  1143→    } else if (step.line !== curLine) {
  1144→      flush(prev);
  1145→      curLine = step.line;
  1146→      legStart = prev;
  1147→    }
  1148→    prev = step.station;
  1149→  }
  1150→  flush(prev);
  1151→
  1152→  const totalStops = legs.reduce((a, l) => a + l.stops, 0);
  1153→  // زمان نمایشی واقعی = جمع زمان لگ‌ها + هزینه تعویض خط + سوار شدن (بدون وزن بهینه‌سازی)
  1154→  const totalMinutes = Math.round(
  1155→    legs.reduce((a, l) => a + l.minutes, 0) + (legs.length - 1) * TRANSFER_MIN + BOARDING_MIN
  1156→  );
  1157→  const transferStations = legs.slice(0, -1).map((l) => l.toName);
  1158→  const busyLeg = legs.find((l) => crowdingNow(l.lineId, now) === "busy");
  1159→
  1160→  return {
  1161→    legs,
  1162→    totalMinutes,
  1163→    totalStations: totalStops + 1,
  1164→    transfers: legs.length - 1,
  1165→    transferStations,
  1166→    fare: fareFor(totalStops),
  1167→    pref,
  1168→    crowdNote: busyLeg
  1169→      ? `ازدحام بالا روی ${busyLeg.lineName} در ساعات اوج`
  1170→      : "سرویس‌دهی همه خطوط مسیر عادی و روان است",
  1171→  };
  1172→}

  1190→  if (opts.facility) res = res.filter((s) => s.amenities.includes(opts.facility!));

  1194→      (s) => s.name.toLowerCase().includes(nq) || s.name.replace(/ /g, "").includes(nq.replace(/ /g, ""))
