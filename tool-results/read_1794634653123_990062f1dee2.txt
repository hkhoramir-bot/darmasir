    21→function esc(s) {
    22→  return String(s ?? "").replace(/[&<>"']/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]));
    23→}

    25→function faTime(iso) {
    26→  if (!iso) return "—";
    27→  try {
    28→    return new Date(iso).toLocaleString("fa-IR", { dateStyle: "short", timeStyle: "short" });
    29→  } catch { return "—"; }
    30→}

    32→async function api(path, opts = {}) {
    33→  const headers = { "Content-Type": "application/json", ...(opts.headers || {}) };
    34→  const token = localStorage.getItem(TOKEN_KEY);
    35→  if (token) headers.Authorization = `Bearer ${token}`;
    36→  const res = await fetch(apiPath(path), {
    37→    method: opts.method || "GET",
    38→    headers,
    39→    body: opts.body ? JSON.stringify(opts.body) : undefined,
    40→  });
    41→  const json = await res.json().catch(() => ({ ok: false, error: { code: "PARSE", message: "پاسخ نامعتبر سرور" } }));
    42→  if (!json.ok) {
    43→    if (res.status === 401 && !path.startsWith("auth/")) { logoutLocal(); throw new Error(json.error?.message || "نشست منقضی شده"); }
    44→    throw new Error(json.error?.message || "خطای ناشناخته");
    45→  }
    46→  return json.data;
    47→}
