    59→    59→let measureCtx: CanvasRenderingContext2D | null = null;
    60→    60→const widthCache = new Map<string, number>();
    61→    61→
    62→    62→function textWidthPx(text: string, weight: number, px: number, epoch: number): number {
    63→    63→  const key = `${epoch}|${weight}|${px}|${text}`;
    64→    64→  const hit = widthCache.get(key);
    65→    65→  if (hit !== undefined) return hit;
    66→    66→  let w: number;
    67→    67→  try {
    68→    68→    if (!measureCtx) {
    69→    69→      measureCtx = document.createElement("canvas").getContext("2d");
    70→    70→    }
    71→    71→    if (measureCtx) {
    72→    72→      measureCtx.font = `${weight} ${px}px Vazirmatn, sans-serif`;
    73→    73→      w = measureCtx.measureText(text).width;
    74→    74→    } else {
    75→    75→      w = text.length * px * 0.6;
    76→    76→    }
    77→    77→  } catch {
    78→    78→    w = text.length * px * 0.6;
    79→    79→  }
    80→    80→  widthCache.set(key, w);
    81→    81→  return w;
    82→    82→}

   116→   116→  | { kind: "idle" }

   121→   121→
   122→   122→export default function MetroMap({
   123→   123→  onStationSelect,
   124→   124→  highlightStations,
   125→   125→  className = "",
   126→   126→}: MetroMapProps) {
   127→   127→  const [t, setT] = useState<Transform>({ scale: 1, tx: 0, ty: 0 });
   128→   128→  const [legendOpen, setLegendOpen] = useState(false);
   129→   129→  const [locate, setLocate] = useState<LocateState>({ kind: "idle" });
   130→   130→  const [box, setBox] = useState({ w: 0, h: 0 });
   131→   131→  const [fontsTick, setFontsTick] = useState(0);
   132→   132→
   133→   133→  /* ---------- آینه همزمان ترنسفورم — جلوگیری از رقابت با React ---------- */
   134→   134→  const tRef = useRef<Transform>({ scale: 1, tx: 0, ty: 0 });
   135→   135→  const svgRef = useRef<SVGSVGElement>(null);
   136→   136→
   137→   137→  /* اشاره‌گرهای فعال — پایه ژست چندلمسی */
   138→   138→  const pointersRef = useRef<Map<number, { x: number; y: number }>>(new Map());
   139→   139→  const panRef = useRef<{ startX: number; startY: number; tx: number; ty: number } | null>(null);
   140→   140→  const pinchRef = useRef<{
   141→   141→    dist0: number;
   142→   142→    vx0: number;
   143→   143→    vy0: number;
   144→   144→    s0: number;
   145→   145→    tx0: number;
   146→   146→    ty0: number;
   147→   147→  } | null>(null);
   148→   148→  const movedRef = useRef(false); // بعد از جابجایی، تپ ایستگاه نادیده گرفته می‌شود
   149→   149→
   150→   150→  /* ---------- اندازه واقعی بوم برای مقیاس ثابت متن ---------- */
   151→   151→  useEffect(() => {
   152→   152→    const el = svgRef.current;
   153→   153→    if (!el) return;
   154→   154→    const update = () => {
   155→   155→      const r = el.getBoundingClientRect();
   156→   156→      if (r.width < 2 || r.height < 2) return;
   157→   157→      setBox((b) =>
   158→   158→        Math.abs(b.w - r.width) > 0.5 || Math.abs(b.h - r.height) > 0.5
   159→   159→          ? { w: r.width, h: r.height }
   160→   160→          : b
   161→   161→      );
   162→   162→    };
   163→   163→    update();
   164→   164→    const ro = new ResizeObserver(update);
   165→   165→    ro.observe(el);
   166→   166→    return () => ro.disconnect();
   167→   167→  }, []);
   168→   168→
   169→   169→  /* ---------- بعد از لود کامل فونت، عرض متن‌ها دوباره اندازه‌گیری می‌شود ---------- */
   170→   170→  useEffect(() => {
   171→   171→    let alive = true;
   172→   172→    try {
   173→   173→      document.fonts?.ready
   174→   174→        .then(() => {
   175→   175→          if (!alive) return;
   176→   176→          widthCache.clear();
   177→   177→          setFontsTick((v) => v + 1);
   178→   178→        })
   179→   179→        .catch(() => {});
   180→   180→    } catch {
   181→   181→      /* مرورگرهای قدیمی */
   182→   182→    }
   183→   183→    return () => {
   184→   184→      alive = false;
   185→   185→    };
   186→   186→  }, []);
   187→   187→
   188→   188→  const clamp = (v: number, min: number, max: number) => Math.min(max, Math.max(min, v));
   189→   189→
   190→   190→  /** ثبت ترنسفورم جدید — همیشه از طریق این تابع */
   191→   191→  const commit = useCallback((next: Transform) => {
   192→   192→    const scale = clamp(next.scale, MIN_SCALE, MAX_SCALE);
   193→   193→    const half = MAP_CANVAS.w / 2;
   194→   194→    const halfH = MAP_CANVAS.h / 2;
   195→   195→    const tx = clamp(next.tx, half - scale * half - 60, -(half - scale * half) + 60);
   196→   196→    const ty = clamp(next.ty, halfH - scale * halfH - 60, -(halfH - scale * halfH) + 60);
   197→   197→    const fixed = { scale, tx, ty };
   198→   198→    tRef.current = fixed;
   199→   199→    setT(fixed);
   200→   200→  }, []);
   201→   201→
   202→   202→  /** تبدیل مختصات صفحه به مختصات viewBox (با احتساب letterbox وضعیت meet) */
   203→   203→  const viewPoint = useCallback((clientX: number, clientY: number) => {
   204→   204→    const svg = svgRef.current;
   205→   205→    if (!svg) return { vx: 0, vy: 0 };
   206→   206→    const rect = svg.getBoundingClientRect();
   207→   207→    const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   208→   208→    const ox = (rect.width - MAP_CANVAS.w * fit) / 2;
   209→   209→    const oy = (rect.height - MAP_CANVAS.h * fit) / 2;
   210→   210→    return {
   211→   211→      vx: (clientX - rect.left - ox) / (fit * tRef.current.scale) - tRef.current.tx / tRef.current.scale,
   212→   212→      vy: (clientY - rect.top - oy) / (fit * tRef.current.scale) - tRef.current.ty / tRef.current.scale,
   213→   213→    };
   214→   214→  }, []);
   215→   215→
   216→   216→  /** بزرگ‌نمایی حول یک نقطه صفحه (پیش‌فرض: مرکز) — لنگر ثابت می‌ماند */
   217→   217→  const zoomAt = useCallback(
   218→   218→    (factor: number, clientX?: number, clientY?: number) => {
   219→   219→      const t0 = tRef.current;
   220→   220→      const svg = svgRef.current;
   221→   221→      if (!svg) {
   222→   222→        commit({ ...t0, scale: t0.scale * factor });
   223→   223→        return;
   224→   224→      }
   225→   225→      const rect = svg.getBoundingClientRect();
   226→   226→      const cx = clientX ?? rect.left + rect.width / 2;
   227→   227→      const cy = clientY ?? rect.top + rect.height / 2;
   228→   228→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   229→   229→      const ox = (rect.width - MAP_CANVAS.w * fit) / 2;
   230→   230→      const oy = (rect.height - MAP_CANVAS.h * fit) / 2;
   231→   231→      // نقطه لنگر در مختصات viewBox فعلی
   232→   232→      const ax = (cx - rect.left - ox) / fit;
   233→   233→      const ay = (cy - rect.top - oy) / fit;
   234→   234→      const scale = clamp(t0.scale * factor, MIN_SCALE, MAX_SCALE);
   235→   235→      commit({
   236→   236→        scale,
   237→   237→        tx: ax - ((ax - t0.tx) * scale) / t0.scale,
   238→   238→        ty: ay - ((ay - t0.ty) * scale) / t0.scale,
   239→   239→      });
   240→   240→    },
   241→   241→    [commit]
   242→   242→  );
   243→   243→
   244→   244→  /* ---------- چرخ ماوس (غیر-Passive تا preventDefault کار کند) ---------- */
   245→   245→  useEffect(() => {
   246→   246→    const el = svgRef.current;
   247→   247→    if (!el) return;
   248→   248→    const onWheel = (e: WheelEvent) => {
   249→   249→      e.preventDefault();
   250→   250→      zoomAt(e.deltaY < 0 ? 1.15 : 0.87, e.clientX, e.clientY);
   251→   251→    };
   252→   252→    el.addEventListener("wheel", onWheel, { passive: false });
   253→   253→    return () => el.removeEventListener("wheel", onWheel);
   254→   254→  }, [zoomAt]);
   255→   255→
   256→   256→  /* ---------- اشاره‌گرها ---------- */
   257→   257→  const onPointerDown = (e: React.PointerEvent) => {
   258→   258→    try {
   259→   259→      (e.target as Element).setPointerCapture?.(e.pointerId);
   260→   260→    } catch {
   261→   261→      /* برخی محیط‌ها (رویدادهای ساختگی/WebView) خطای InvalidPointerId می‌دهند */
   262→   262→    }
   263→   263→    pointersRef.current.set(e.pointerId, { x: e.clientX, y: e.clientY });
   264→   264→
   265→   265→    if (pointersRef.current.size === 1) {
   266→   266→      // شروع جابجایی تک‌انگشتی
   267→   267→      movedRef.current = false;
   268→   268→      panRef.current = { startX: e.clientX, startY: e.clientY, tx: tRef.current.tx, ty: tRef.current.ty };
   269→   269→      pinchRef.current = null;
   270→   270→    } else if (pointersRef.current.size === 2) {
   271→   271→      // شروع پینچ — فاصله و نقطه میانی دو انگشت
   272→   272→      const [p1, p2] = [...pointersRef.current.values()];
   273→   273→      const t0 = tRef.current;
   274→   274→      const svg = svgRef.current;
   275→   275→      if (!svg) return;
   276→   276→      const rect = svg.getBoundingClientRect();
   277→   277→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   278→   278→      const ox = (rect.width - MAP_CANVAS.w * fit) / 2;
   279→   279→      const oy = (rect.height - MAP_CANVAS.h * fit) / 2;
   280→   280→      const mx = (p1.x + p2.x) / 2;
   281→   281→      const my = (p1.y + p2.y) / 2;
   282→   282→      pinchRef.current = {
   283→   283→        dist0: Math.hypot(p1.x - p2.x, p1.y - p2.y) || 1,
   284→   284→        vx0: (mx - rect.left - ox) / fit,
   285→   285→        vy0: (my - rect.top - oy) / fit,
   286→   286→        s0: t0.scale,
   287→   287→        tx0: t0.tx,
   288→   288→        ty0: t0.ty,
   289→   289→      };
   290→   290→      panRef.current = null;
   291→   291→    }
   292→   292→  };
   293→   293→
   294→   294→  const onPointerMove = (e: React.PointerEvent) => {
   295→   295→    if (!pointersRef.current.has(e.pointerId)) return;
   296→   296→    pointersRef.current.set(e.pointerId, { x: e.clientX, y: e.clientY });
   297→   297→
   298→   298→    if (pinchRef.current && pointersRef.current.size >= 2) {
   299→   299→      // پینچ: مقیاس + حفظ لنگر زیر انگشتان
   300→   300→      const [p1, p2] = [...pointersRef.current.values()];
   301→   301→      const p0 = pinchRef.current;
   302→   302→      const dist = Math.hypot(p1.x - p2.x, p1.y - p2.y) || 1;
   303→   303→      const svg = svgRef.current;
   304→   304→      if (!svg) return;
   305→   305→      const rect = svg.getBoundingClientRect();
   306→   306→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   307→   307→      const ox = (rect.width - MAP_CANVAS.w * fit) / 2;
   308→   308→      const oy = (rect.height - MAP_CANVAS.h * fit) / 2;
   309→   309→      const mx = (p1.x + p2.x) / 2;
   310→   310→      const my = (p1.y + p2.y) / 2;
   311→   311→      const vxm = (mx - rect.left - ox) / fit;
   312→   312→      const vym = (my - rect.top - oy) / fit;
   313→   313→      const scale = clamp((p0.s0 * dist) / p0.dist0, MIN_SCALE, MAX_SCALE);
   314→   314→      // نقطه میانی اولیه باید زیر انگشتان بماند
   315→   315→      const cx = (p0.vx0 - p0.tx0) / p0.s0;
   316→   316→      const cy = (p0.vy0 - p0.ty0) / p0.s0;
   317→   317→      commit({
   318→   318→        scale,
   319→   319→        tx: vxm - scale * cx,
   320→   320→        ty: vym - scale * cy,
   321→   321→      });
   322→   322→      movedRef.current = true;
   323→   323→      return;
   324→   324→    }
   325→   325→
   326→   326→    if (panRef.current && pointersRef.current.size === 1) {
   327→   327→      const p0 = panRef.current;
   328→   328→      const svg = svgRef.current;
   329→   329→      if (!svg) return;
   330→   330→      const rect = svg.getBoundingClientRect();
   331→   331→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   332→   332→      const dx = (e.clientX - p0.startX) / fit;
   333→   333→      const dy = (e.clientY - p0.startY) / fit;
   334→   334→      if (Math.abs(e.clientX - p0.startX) > TAP_SLOP_PX || Math.abs(e.clientY - p0.startY) > TAP_SLOP_PX) {
   335→   335→        movedRef.current = true;
   336→   336→      }
   337→   337→      commit({ scale: tRef.current.scale, tx: p0.tx + dx, ty: p0.ty + dy });
   338→   338→    }
   339→   339→  };
   340→   340→
   341→   341→  const endPointer = (e: React.PointerEvent) => {
   342→   342→    try {
   343→   343→      (e.target as Element).releasePointerCapture?.(e.pointerId);
   344→   344→    } catch {
   345→   345→      /* قبلاً آزاد شده */
   346→   346→    }
   347→   347→    pointersRef.current.delete(e.pointerId);
   348→   348→    if (pointersRef.current.size === 1) {
   349→   349→      // یک انگشت باقی مانده — پینچ تمام شده، ادامه با جابجایی
   350→   350→      pinchRef.current = null;
   351→   351→      const [p] = [...pointersRef.current.values()];
   352→   352→      panRef.current = { startX: p.x, startY: p.y, tx: tRef.current.tx, ty: tRef.current.ty };
   353→   353→    } else if (pointersRef.current.size === 0) {
   354→   354→      panRef.current = null;
   355→   355→      pinchRef.current = null;
   356→   356→    }
   357→   357→  };
   358→   358→
   359→   359→  /* ---------- مرکز کردن دید روی یک ایستگاه ---------- */
   360→   360→  const zoomToStation = useCallback(
   361→   361→    (sx: number, sy: number) => {
   362→   362→      const svg = svgRef.current;
   363→   363→      if (!svg) return;
   364→   364→      const rect = svg.getBoundingClientRect();
   365→   365→      const fit = Math.min(rect.width / MAP_CANVAS.w, rect.height / MAP_CANVAS.h);
   366→   366→      const scale = 2.4;
   367→   367→      commit({
   368→   368→        scale,
   369→   369→        tx: rect.width / (2 * fit) - scale * sx,
   370→   370→        ty: rect.height / (2 * fit) - scale * sy,
   371→   371→      });
   372→   372→    },
   373→   373→    [commit]
   374→   374→  );
   375→   375→
   376→   376→  /* ---------- موقعیت‌یاب GPS ---------- */
   377→   377→  const locateMe = useCallback(() => {
   378→   378→    if (locate.kind === "busy") return;
   379→   379→    if (!("geolocation" in navigator)) {
   380→   380→      setLocate({ kind: "error", message: "مرورگر از موقعیت‌یابی پشتیبانی نمی‌کند" });
   381→   381→      return;
   382→   382→    }
   383→   383→    setLocate({ kind: "busy" });
   384→   384→    navigator.geolocation.getCurrentPosition(
   385→   385→      (pos) => {
   386→   386→        const { latitude, longitude } = pos.coords;
   387→   387→        const near = nearestStationTo(latitude, longitude);
   388→   388→        if (!near) {
   389→   389→          setLocate({ kind: "far" });
   390→   390→          return;
   391→   391→        }
   392→   392→        const st = MAP_STATION_BY_ID.get(near.id);
   393→   393→        if (!st) {
   394→   394→          setLocate({ kind: "far" });
   395→   395→          return;
   396→   396→        }
   397→   397→        setLocate({ kind: "ok", stationId: near.id, stationName: near.name, distanceM: near.distanceM });
   398→   398→        // بزرگ‌نمایی روی ایستگاه
   399→   399→        zoomToStation(st.x, st.y);
   400→   400→      },
   401→   401→      (err) => {
   402→   402→        setLocate({
   403→   403→          kind: "error",
   404→   404→          message:
   405→   405→            err.code === err.PERMISSION_DENIED
   406→   406→              ? "دسترسی موقعیت مکانی داده نشد"
