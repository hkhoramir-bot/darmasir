    53→/** چیپ نام خط — با رنگ رسمی خط */
    54→function LineChips({ stationId }: { stationId: string }) {
    55→  const st = getStation(stationId);
    56→  if (!st) return null;
    57→  return (
    58→    <span className="flex gap-1 shrink-0">
    59→      {st.lines.slice(0, 2).map((l) => {
    60→        const line = getLine(l.lineId);
    61→        if (!line) return null;
    62→        return (
    63→          <span
    64→            key={l.lineId}
    65→            className="text-[10px] font-bold px-1.5 py-0.5 rounded whitespace-nowrap"
    66→            style={{ backgroundColor: line.color, color: line.textOnColor }}
    67→          >
    68→            {line.name}
    69→          </span>
    70→        );
    71→      })}
    72→    </span>
    73→  );
    74→}

    78→  const router = useRouter();
    79→  const city = useShell((s) => s.city);

    82→  /** محتوای وابسته به زمان (قطار بعدی) فقط بعد از mount رندر می‌شود */

    93→  const [destId, setDestId] = useState<string | null>("tajrish");
    94→  const [dropFor, setDropFor] = useState<"origin" | "destination" | null>(null);

    97→  /* ---- ایستگاه جاری (GPS) ----

   101→  const [near, setNear] = useState<{ id: string; name: string } | null>(null);
   102→  const [nearAlt, setNearAlt] = useState<{ name: string; distanceM: number } | null>(null);
   103→  const [fixAcc, setFixAcc] = useState<number | null>(null);
   104→  const [geoNote, setGeoNote] = useState<string | null>(null);

   106→  /* ---- مسیرهای روزمره ---- */
   107→  const [daily, setDaily] = useState<DailyRouteRow[]>([]);
   108→
   109→  /* ---- وضعیت صوت ---- */
   110→  const [voiceResult, setVoiceResult] = useState<string | null>(null);
   111→  const originRef = useRef<HTMLInputElement>(null);
   112→  const destRef = useRef<HTMLInputElement>(null);
   113→
   114→  const stations = useMemo(
   115→    () => searchStations({ cityId: city === "isfahan" ? "isfahan" : city === "mashhad" ? "mashhad" : "tehran" }),

   118→
   119→  const suggestions = useMemo(() => {
   120→    const q = dropFor === "origin" ? originText : destText;
   121→    if (!dropFor) return [];
   122→    const base = q.trim() ? stations.filter((s) => s.name.includes(q.trim())) : stations;
   123→    const featured = base.filter((s) => s.featured);
   124→    return (featured.length >= 3 ? featured : base).slice(0, 6);
   125→  }, [dropFor, originText, destText, stations]);
   126→
   127→  const plan = useMemo(
   128→    () => (originId && destId && originId !== destId ? computeRoute(originId, destId, "fastest") : null),
   129→    [originId, destId]
   130→  );
   131→
   132→  /* ---- ایستگاه بعدی برای نوار وضعیت زنده — فقط با موقعیت واقعی ---- */
   133→  const liveInfo = useMemo(() => {
   134→    if (!near) return null;
   135→    const sid = near.id;
   136→    const st = getStation(sid);
   137→    if (!st || st.lines.length === 0) return null;
   138→    const lineId = st.lines[0].lineId;
   139→    const idx = st.lines[0].order;
   140→    const list = stationsOfLine(lineId);
   141→    const dirs = nextArrivalsBoth(lineId, sid);
   142→    const withMin = dirs.map((d, k) => ({ ...d, k })).filter((d) => d.minutes.length > 0);
   143→    if (!withMin.length) return { stationName: st.name, text: null, out: dirs.length > 0 && dirs[0].outOfService };

   147→      stationName: st.name,
   148→      text: `ایستگاه بعدی: ${adj?.name ?? best.to} (${toFa(best.minutes[0])} دقیقه)`,

   150→    };
   151→  }, [near]);

   157→      .then((d) => setDaily(d.routes ?? []))

   167→      const fix = await getCurrentGeoBest(3, 8000, 450, 25000);

   254→            ? (near?.name ?? null)
   255→            : null;

   257→      const toName = intent.toId ? (stations.find((s) => s.id === intent.toId)?.name ?? null) : null;
   258→
   259→      if (toId) {
   260→        setDestText(toName ?? "");
   261→        setDestId(toId);
   262→      }
   263→      if (fromId) {
   264→        setOriginText(fromName ?? "");
   265→        setOriginId(fromId);
   266→      }
   267→
   268→      if (fromId && toId) {
   269→        toast({
   270→          title: "مبدأ و مقصد صوتی شناسایی شد",
   271→          description: `${fromName} → ${toName} — مسیریابی باز می‌شود`,

   286→      }

   312→      toast({
   313→        title: "سفر فعالی در جریان است",
   314→        description: "برای مسیریابی جدید، ابتدا سفر فعلی را از پایین صفحه پایان دهید",
   315→      });

   319→    else if (originId) router.push(`/plan?from=${originId}`);
   320→    else router.push("/plan");
   321→  }, [originId, destId, router, toast]);
   322→
   323→  const voiceStatusText = voice.listening
   324→    ? "در حال گوش دادن..."
   325→    : voice.busy

   328→        ? voiceResult

   330→

   332→    <>
   333→      <AppBar />

   335→        <div className="space-y-4">
   336→          {/* 1. بنر تبلیغات */}

   338→            <div className="flex items-center justify-center py-2 text-center text-xs font-medium text-outline">

   346→            {/* هاله محیطی */}

   348→            <div className="absolute -bottom-10 -right-10 w-44 h-44 bg-tertiary-fixed/30 rounded-full blur-2xl pointer-events-none"></div>
   349→
   350→            <div className="relative w-28 h-28 flex items-center justify-center my-1.5">
   351→              <div className="pulse-wave-1 absolute inset-0 rounded-full bg-primary-fixed/60"></div>
   352→              <div className="pulse-wave-2 absolute inset-2 rounded-full bg-primary-fixed/40"></div>

   359→                aria-label="جستجوی صوتی"

   367→            <div className="space-y-1.5 mt-1">

   400→                <h3 className="font-bold text-xs text-on-surface">دسترسی سریع مسافر</h3>
   401→              </div>

   411→                type="button"
   412→              >
   413→                <div className="w-12 h-12 rounded-2xl bg-primary/10 group-hover:bg-primary/20 text-primary flex items-center justify-center transition-colors shadow-sm">
